   241→// ─── API helper ───────────────────────────────────────────────
   242→async function api(path, opts = {}) {
   243→  const headers = { 'Content-Type': 'application/json', ...(opts.headers || {}) };
   244→  if (token) headers['Authorization'] = 'Bearer ' + token;
   245→  const res = await fetch('/api' + path, { ...opts, headers });
   246→  const data = await res.json();
   247→  if (!res.ok) throw new Error(data.error || 'Gagal memuat data');
   248→  return data;
   249→}

   251→// ─── Format helpers ───────────────────────────────────────────
   252→function fmtRp(n) { return 'Rp ' + Number(n||0).toLocaleString('id-ID'); }
   253→function fmtDate(d) { if (!d) return '—'; return new Date(d).toLocaleDateString('id-ID', {day:'numeric',month:'short',year:'numeric'}); }
   254→function fmtRelTime(d) {
   255→  if (!d) return '—';
   256→  const diff = Date.now() - new Date(d).getTime();
   257→  const days = Math.floor(diff / 86400000);
   258→  if (days < 1) return 'Hari ini';
   259→  if (days < 7) return days + ' hari lalu';
   260→  return fmtDate(d);
   261→}

   263→// ─── Login ────────────────────────────────────────────────────
   264→async function doLogin() {
   265→  const email = document.getElementById('loginEmail').value.trim();
   266→  const pass = document.getElementById('loginPass').value;
   267→  const errEl = document.getElementById('loginError');
   268→  const btn = document.getElementById('loginBtn');
   269→  errEl.classList.remove('show');
   270→  btn.disabled = true; btn.textContent = 'Memproses...';
   271→
   272→  try {
   273→    // Step 1: Login to get token
   274→    const loginRes = await fetch('/api/auth/login', {
   275→      method: 'POST',
   276→      headers: { 'Content-Type': 'application/json' },
   277→      body: JSON.stringify({ email, password: pass })
   278→    });
   279→    const loginData = await loginRes.json();
   280→    if (!loginRes.ok) throw new Error(loginData.error_description || loginData.error || 'Login gagal');
   281→
   282→    token = loginData.access_token;
   283→    if (!token) throw new Error('Token tidak ditemukan');
   284→
   285→    // Step 2: Verify user with /api/auth/me
   286→    const meRes = await fetch('/api/auth/me', {
   287→      headers: { 'Authorization': 'Bearer ' + token }
   288→    });
   289→    const meData = await meRes.json();
   290→    if (!meRes.ok) throw new Error('Verifikasi gagal');
   291→
   292→    // Step 3: Check admin email
   293→    const userEmail = meData.data?.email || meData.email;
   294→    if (userEmail !== ADMIN_EMAIL) {
   295→      throw new Error('Akses ditolak. Halaman ini hanya untuk admin.');
   296→    }
   297→
   298→    // Success
   299→    sessionStorage.setItem('arus_admin_token', token);
   300→    document.getElementById('adminEmail').textContent = userEmail;
   301→    document.getElementById('loginScreen').style.display = 'none';
   302→    document.getElementById('dashScreen').style.display = 'flex';
   303→    loadSection('dashboard');
   304→
   305→  } catch (e) {
   306→    errEl.textContent = e.message;
   307→    errEl.classList.add('show');
   308→    token = '';
   309→  } finally {
   310→    btn.disabled = false; btn.textContent = 'Masuk';
   311→  }
   312→}
