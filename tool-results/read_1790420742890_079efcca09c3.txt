    76→    76→  const authHeader = req.headers.get("Authorization");

    79→    79→
    80→    80→  // Check KV session cache first
    81→    81→  const cached = await getCachedSession(env, token);
    82→    82→  if (cached) {
    83→    83→    const adminEmail = env.ADMIN_EMAIL || "riskiakbarp123@gmail.com";
    84→    84→    if (cached.email !== adminEmail) return null;
    85→    85→    return cached;

    89→    89→  const res = await fetch(`${env.SUPABASE_URL}/auth/v1/user`, {
    90→    90→    headers: { Authorization: `Bearer ${token}`, apikey: env.SUPABASE_ANON_KEY },
    91→    91→  });
    92→    92→  if (!res.ok) return null;
    93→    93→  const user = await res.json() as { id: string; email: string };
    94→    94→
    95→    95→  const adminEmail = env.ADMIN_EMAIL || "riskiakbarp123@gmail.com";

   100→   100→  await cacheSession(env, token, sessionData);

   102→   102→  return sessionData;
   103→   103→}

   514→   514→  ]);
   515→   515→
   516→   516→  if (txnsRes.error) throw txnsRes.error;
   517→   517→  if (debtsRes.error) throw debtsRes.error;
   518→   518→
   519→   519→  const transactions = txnsRes.data || [];
   520→   520→  const debts = debtsRes.data || [];
   521→   521→
   522→   522→  const totalIncome = transactions.filter((t: { type: string }) => t.type === "income").reduce((s: number, t: { amount: number }) => s + t.amount, 0);
   523→   523→  const totalExpenses = transactions.filter((t: { type: string }) => t.type === "expense").reduce((s: number, t: { amount: number }) => s + t.amount, 0);
   524→   524→  const activeBalance = totalIncome - totalExpenses;
   525→   525→  const totalDebt = debts.filter((d: { type: string; status: string }) => d.type === "debt" && d.status !== "paid").reduce((s: number, d: { amount: number; paid_amount: number }) => s + (d.amount - d.paid_amount), 0);
   526→   526→  const totalReceivable = debts.filter((d: { type: string; status: string }) => d.type === "receivable" && d.status !== "paid").reduce((s: number, d: { amount: number; paid_amount: number }) => s + (d.amount - d.paid_amount), 0);
   527→   527→
   528→   528→  const now = new Date();
   529→   529→  const monthlyData = [];
   530→   530→  for (let i = 11; i >= 0; i--) {
   531→   531→    const mStart = new Date(now.getFullYear(), now.getMonth() - i, 1);
   532→   532→    const mEnd = new Date(now.getFullYear(), now.getMonth() - i + 1, 1);
   533→   533→    const mTxns = transactions.filter((t: { occurred_at: string }) => { const d = new Date(t.occurred_at); return d >= mStart && d < mEnd; });
   534→   534→    const mIncome = mTxns.filter((t: { type: string }) => t.type === "income").reduce((s: number, t: { amount: number }) => s + t.amount, 0);
   535→   535→    const mExpenses = mTxns.filter((t: { type: string }) => t.type === "expense").reduce((s: number, t: { amount: number }) => s + t.amount, 0);
   536→   536→    monthlyData.push({ month: mStart.toLocaleDateString("en-US", { month: "short", year: "2-digit" }), income: mIncome, expenses: mExpenses, net: mIncome - mExpenses });
   537→   537→  }
   538→   538→
   539→   539→  return json({ activeBalance, totalIncome, totalExpenses, totalDebt, totalReceivable, monthlyData });
   540→   540→}

  1269→  1269→      await setFeatureFlag(env, flag, value);
  1270→  1270→      return json({ ok: true, flag: `ff:${flag}`, value });

  1272→  1272→  }
  1273→  1273→
  1274→  1274→  // ── Generic KV Get/Put/Delete ──

  1296→  1296→  }

  1298→  1298→  if (req.method === "DELETE") {
