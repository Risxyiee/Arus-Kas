    67→    67→// ─── SHA-512 helper (for Midtrans signature verification) ────────
    68→    68→async function sha512(message: string): Promise<string> {
    69→    69→  const data = new TextEncoder().encode(message);
    70→    70→  const hash = await crypto.subtle.digest("SHA-512", data);
    71→    71→  return Array.from(new Uint8Array(hash)).map(b => b.toString(16).padStart(2, "0")).join("");
    72→    72→}

    74→    74→// ─── Admin Auth Verification (with KV session cache) ───────────
    75→    75→async function verifyAdmin(req: Request, env: Env): Promise<{ userId: string; email: string } | null> {
    76→    76→  const authHeader = req.headers.get("Authorization");
    77→    77→  if (!authHeader || !authHeader.startsWith("Bearer ")) return null;
    78→    78→  const token = authHeader.replace("Bearer ", "");
    79→    79→
    80→    80→  // Check KV session cache first
    81→    81→  const cached = await getCachedSession(env, token);
    82→    82→  if (cached) {
    83→    83→    const adminEmail = env.ADMIN_EMAIL || "riskiakbarp123@gmail.com";
    84→    84→    if (cached.email !== adminEmail) return null;
    85→    85→    return cached;
    86→    86→  }
    87→    87→
    88→    88→  // Verify token+cache session
    89→    89→  const res = await fetch(`${env.SUPABASE_URL}/auth/v1/user`, {
    90→    90→    headers: { Authorization: `Bearer ${token}`, apikey: env.SUPABASE_ANON_KEY },
    91→    91→  });
    92→    92→  if (!res.ok) return null;
    93→    93→  const user = await res.json() as { id: string; email: string };
    94→    94→
    95→    95→  const adminEmail = env.ADMIN_EMAIL || "riskiakbarp123@gmail.com";
    96→    96→  if (user.email !== adminEmail) return null;
    97→    97→
    98→    98→  // Cache the session
    99→    99→  const sessionData = { userId: user.id, email: user.email };
   100→   100→  await cacheSession(env, token, sessionData);
   101→   101→
   102→   102→  return sessionData;
   103→   103→}

   636→   636→  const debts = debtsRes.data || [];
   637→   637→

   641→   641→  const totalDebt = debts.filter((d: { type: string; status: string }) => d.type === "debt" && d.status !== "paid").reduce((s: number, d: { amount: number; paid_amount: number }) => s + (d.amount - d.paid_amount), 0);
   642→   642→  const totalReceivable = debts.filter((d: { type: string; status: string }) => d.type === "receivable" && d.status !== "paid").reduce((s: number, d: { amount: number; paid_amount: number }) => s + (d.amount - d.paid_amount), 0);
   643→   643→
   644→   644→  const now = new Date();
   645→   645→  const monthlyData = [];
   646→   646→  for (let i = 11; i >= 0; i--) {
   647→   647→    const mStart = new Date(now.getFullYear(), now.getMonth() - i, 1);
   648→   648→    const mEnd = new Date(now.getFullYear(), now.getMonth() - i + 1, 1);
   649→   649→    const mTxns = transactions.filter((t: { occurred_at: string }) => { const d = new Date(t.occurred_at); return d >= mStart && d < mEnd; });
   650→   650→    const mIncome = mTxns.filter((t: { type: string }) => t.type === "income").reduce((s: number, t: { amount: number }) => s + t.amount, 0);
   651→   651→    const mExpenses = mTxns.filter((t: { type: string }) => t.type === "expense").reduce((s: number, t: { amount: number }) => s + t.amount, 0);
   652→   652→    monthlyData.push({ month: mStart.toLocaleDateString("en-US", { month: "short", year: "2-digit" }), income: mIncome, expenses: mExpenses, net: mIncome - mExpenses });
   653→   653→  }
   654→   654→
   655→   655→  return json({ activeBalance, totalIncome, totalExpenses, totalDebt, totalReceivable, monthlyData });
   656→   656→}

   966→   966→async function handleMidtransWebhook(req: Request, env: Env): Promise<Response> {
   967→   967→  const body = await getBody(req);
   968→   968→
   969→   969→  const transactionStatus = body.transaction_status as string;
   970→   970→  const orderId = body.order_id as string;
   971→   971→  const paymentType = body.payment_type as string;
   972→   972→  const metadata = body.metadata as { user_id?: string; plan?: string } | undefined;
   973→   973→  const userId = metadata?.user_id || (body.custom_field1 as string);
   974→   974→
   975→   975→  if (!orderId || !transactionStatus) {
   976→   976→    return json({ error: "Invalid webhook payload" }, 400);
   977→   977→  }
   978→   978→
   979→   979→  // Verify signature to prevent fake webhook attacks
   980→   980→  const signatureKey = body.signature_key as string;
   981→   981→  const grossAmount = body.gross_amount as string;
   982→   982→  if (signatureKey && env.MIDTRANS_SERVER_KEY) {
   983→   983→    const expectedSig = await sha512(orderId + transactionStatus + grossAmount + env.MIDTRANS_SERVER_KEY);
   984→   984→    if (signatureKey !== expectedSig) {
   985→   985→      return json({ error: "Invalid signature" }, 403);
   986→   986→    }
   987→   987→  }
   988→   988→
   989→   989→  const admin = getAdmin(env);
   990→   990→
   991→   991→  // Determine if payment is successful
   992→   992→  const isSuccess =
   993→   993→    transactionStatus === "capture" ||
   994→   994→    transactionStatus === "settlement";
   995→   995→
   996→   996→  const isPending = transactionStatus === "pending";
   997→   997→  const isFailure =
   998→   998→    transactionStatus === "deny" ||
   999→   999→    transactionStatus === "expire" ||
  1000→  1000→    transactionStatus === "cancel";
  1001→  1001→
  1002→  1002→  if (isSuccess && userId) {
  1003→  1003→    // Activate Pro plan — determine duration from gross_amount
  1004→  1004→    const amount = parseFloat(grossAmount || "29000");
  1005→  1005→    const isAnnual = amount >= 278400;
  1006→  1006→    const durationDays = isAnnual ? 365 : 30;
  1007→  1007→    const billingSuffix = isAnnual ? "_annual" : "_monthly";
  1008→  1008→    const expiresAt = new Date(Date.now() + durationDays * 86400000).toISOString();
  1009→  1009→    // Try update with billing column; if it doesn't exist, fall back to without
  1010→  1010→    let updateData: Record<string, unknown> = {
  1011→  1011→      plan: "pro",
  1012→  1012→      started_at: new Date().toISOString(),
  1013→  1013→      expires_at: expiresAt,
  1014→  1014→      billing: isAnnual ? "annual" : "monthly",
  1015→  1015→      payment_method: `midtrans_${paymentType}${billingSuffix}`,
  1016→  1016→      transaction_id: orderId,
  1017→  1017→    };
  1018→  1018→    let { error } = await admin
  1019→  1019→      .from("subscriptions")
  1020→  1020→      .update(updateData)
  1021→  1021→      .eq("user_id", userId);
  1022→  1022→
  1023→  1023→    // If billing column doesn't exist, retry without it
  1024→  1024→    if (error && (error.message || "").includes("billing")) {
  1025→  1025→      const { billing: _, ...fallbackData } = updateData;
  1026→  1026→      const retryResult = await admin.from("subscriptions").update(fallbackData).eq("user_id", userId);
  1027→  1027→      error = retryResult.error;
  1028→  1028→    }
  1029→  1029→
  1030→  1030→    if (error) console.error("Failed to update subscription:", error);
  1031→  1031→    else await invalidateSubscriptionCache(env, userId);
  1032→  1032→  } else if (isFailure && userId) {
  1033→  1033→    // Revert to free
  1034→  1034→    await admin
  1035→  1035→      .from("subscriptions")
  1036→  1036→      .update({ plan: "free", transaction_id: orderId })
  1037→  1037→      .eq("user_id", userId);
  1038→  1038→    await invalidateSubscriptionCache(env, userId);
  1039→  1039→  }
  1040→  1040→
  1041→  1041→  // Always return 200 to Midtrans
  1042→  1042→  return json({ ok: true });
  1043→  1043→}

  1147→  1147→async function handleCron(env: Env): Promise<void> {

  1149→  1149→  const now = new Date().toISOString();
  1150→  1150→
  1151→  1151→  // Find all Pro subscriptions that have expired

  1154→  1154→    .select("user_id, expires_at")

  1158→  1158→  if (error) {

  1160→  1160→    return;

  1162→  1162→
  1163→  1163→  if (!expired || expired.length === 0) {
  1164→  1164→    console.log(`Cron: no expired subscriptions at ${now}`);
  1165→  1165→    return;
  1166→  1166→  }
  1167→  1167→
  1168→  1168→  // Downgrade expired Pro users to Free
  1169→  1169→  const userIds = expired.map((s: { user_id: string }) => s.user_id);
  1170→  1170→  const { error: updateError } = await admin
  1171→  1171→    .from("subscriptions")
  1172→  1172→    .update({ plan: "free" })
  1173→  1173→    .in("user_id", userIds);
  1174→  1174→
  1175→  1175→  if (updateError) {
  1176→  1176→    console.error("Cron: failed to downgrade:", updateError);
  1177→  1177→  } else {
  1178→  1178→    // Invalidate KV subscription caches for downgraded users
  1179→  1179→    for (const uid of userIds) {

  1278→  1278→async function handleCache(req: Request, env: Env): Promise<Response> {
  1279→  1279→  const url = new URL(req.url);
  1280→  1280→  const subPath = url.pathname.replace("/api/cache", "").replace("//", "/");
  1281→  1281→
  1282→  1282→  // ── Feature Flags ──
  1283→  1283→  if (subPath === "/flags" || subPath === "/flags/") {
  1284→  1284→    if (req.method === "GET") {
  1285→  1285→      // Get all feature flags (merged with defaults)
  1286→  1286→      const flags: Record<string, string> = { ...DEFAULT_FEATURE_FLAGS };
  1287→  1287→      try {
  1288→  1288→        const list = await env.arus_kv.list({ prefix: "ff:" });
  1289→  1289→        for (const k of list.keys) {
  1290→  1290→          const val = await env.arus_kv.get(k.name);
  1291→  1291→          if (val !== null) flags[k.name] = val;
  1292→  1292→        }
  1293→  1293→      } catch {}
  1294→  1294→      return json({ flags });
  1295→  1295→    }
  1296→  1296→
  1297→  1297→    if (req.method === "PUT") {
  1298→  1298→      // Admin: set a feature flag
  1299→  1299→      const body = await getBody(req);
  1300→  1300→      const flag = body.flag as string;
  1301→  1301→      const value = body.value as string;
  1302→  1302→      if (!flag || value === undefined) return json({ error: "flag dan value required" }, 400);
  1303→  1303→      await setFeatureFlag(env, flag, value);
  1304→  1304→      return json({ ok: true, flag: `ff:${flag}`, value });
  1305→  1305→    }
  1306→  1306→  }
  1307→  1307→
  1308→  1308→  // ── Generic KV Get/Put/Delete ──
  1309→  1309→  if (req.method === "GET") {
  1310→  1310→    const key = url.searchParams.get("key");
  1311→  1311→    if (key) {
  1312→  1312→      const value = await env.arus_kv.get(key);
  1313→  1313→      return json({ key, value });
  1314→  1314→    }
  1315→  1315→    // List all keys with prefix
  1316→  1316→    const prefix = url.searchParams.get("prefix") || "";
  1317→  1317→    const list = await env.arus_kv.list({ prefix });
  1318→  1318→    return json({ keys: list.keys.map((k) => ({ name: k.name, expiration: k.expiration })) });
  1319→  1319→  }
  1320→  1320→
  1321→  1321→  if (req.method === "PUT") {
  1322→  1322→    const body = await getBody(req);
  1323→  1323→    const key = body.key as string;
  1324→  1324→    const value = body.value as string;
  1325→  1325→    const ttl = body.ttl as number | undefined;
  1326→  1326→    if (!key || value === undefined) return json({ error: "key dan value required" }, 400);
  1327→  1327→    const opts = ttl ? { expirationTtl: ttl } : undefined;
  1328→  1328→    await env.arus_kv.put(key, value, opts);
  1329→  1329→    return json({ ok: true, key });
  1330→  1330→  }
  1331→  1331→
  1332→  1332→  if (req.method === "DELETE") {
  1333→  1333→    const key = url.searchParams.get("key");
  1334→  1334→    if (!key) return json({ error: "key required" }, 400);
  1335→  1335→    await env.arus_kv.delete(key);
  1336→  1336→    return json({ ok: true });
  1337→  1337→  }
  1338→  1338→
  1339→  1339→  return json({ error: "Method not allowed" }, 405);
  1340→  1340→}
