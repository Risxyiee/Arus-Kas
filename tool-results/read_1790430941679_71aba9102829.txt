    67→// ─── Admin Auth Verification (with KV session cache) ───────────
    68→async function verifyAdmin(req: Request, env: Env): Promise<{ userId: string; email: string } | null> {
    69→  const authHeader = req.headers.get("Authorization");
    70→  if (!authHeader || !authHeader.startsWith("Bearer ")) return null;
    71→  const token = authHeader.replace("Bearer ", "");
    72→
    73→  // Check KV session cache first
    74→  const cached = await getCachedSession(env, token);
    75→  if (cached) {
    76→    const adminEmail = env.ADMIN_EMAIL || "riskiakbarp123@gmail.com";
    77→    if (cached.email !== adminEmail) return null;
    78→    return cached;
    79→  }
    80→
    81→  // Verify token+cache session
    82→  const res = await fetch(`${env.SUPABASE_URL}/auth/v1/user`, {
    83→    headers: { Authorization: `Bearer ${token}`, apikey: env.SUPABASE_ANON_KEY },
    84→  });
    85→  if (!res.ok) return null;
    86→  const user = await res.json() as { id: string; email: string };
    87→
    88→  const adminEmail = env.ADMIN_EMAIL || "riskiakbarp123@gmail.com";
    89→  if (user.email !== adminEmail) return null;
    90→
    91→  // Cache the session
    92→  const sessionData = { userId: user.id, email: user.email };
    93→  await cacheSession(env, token, sessionData);
    94→
    95→  return sessionData;
    96→}

    98→// ─── API: /api/admin/* ────────────────────────────────────────
    99→async function handleAdmin(req: Request, env: Env, path: string): Promise<Response> {
   100→  // Verify admin access for all admin routes
   101→  const admin = await verifyAdmin(req, env);
   102→  if (!admin) return json({ error: "Akses ditolak. Halaman ini hanya untuk admin." }, 403);
   103→
   104→  const sb = getAdmin(env);
   105→  const url = new URL(req.url);
   106→
   107→  // GET /api/admin/stats — Dashboard overview stats
   108→  if (path === "/admin/stats" && req.method === "GET") {
   109→    const sevenDaysAgo = new Date(Date.now() - 7 * 86400000).toISOString();
   110→
   111→    const [profilesRes, subsRes, recentRes] = await Promise.all([
   112→      sb.from("profiles").select("id, created_at", { count: "exact" }),
   113→      sb.from("subscriptions").select("plan, expires_at"),
   114→      sb.from("profiles").select("id").gte("created_at", sevenDaysAgo),
   115→    ]);
   116→
   117→    if (profilesRes.error) throw profilesRes.error;
   118→    if (subsRes.error) throw subsRes.error;
   119→    if (recentRes.error) throw recentRes.error;
   120→
   121→    const totalUsers = profilesRes.count ?? (profilesRes.data || []).length;
   122→    const subs = subsRes.data || [];
   123→    const proCount = subs.filter((s: { plan: string }) => s.plan === "pro").length;
   124→    const freeCount = totalUsers - proCount;
   125→    const now = new Date();
   126→    const activePro = subs.filter((s: { plan: string; expires_at: string | null }) => s.plan === "pro" && s.expires_at && new Date(s.expires_at) >= now).length;
   127→    const expiredPro = subs.filter((s: { plan: string; expires_at: string | null }) => s.plan === "pro" && s.expires_at && new Date(s.expires_at) < now).length;
   128→    const recentSignups = (recentRes.data || []).length;
   129→    const monthlyRevenue = proCount * 29000;
   130→
   131→    return json({
   132→      data: { totalUsers, proCount, freeCount, monthlyRevenue, recentSignups, activePro, expiredPro },
   133→    });
   134→  }
   135→
   136→  // GET /api/admin/users — List all users with profile + subscription info
   137→  if (path === "/admin/users" && req.method === "GET") {
   138→    const [profilesRes, subsRes] = await Promise.all([
   139→      sb.from("profiles").select("*").order("created_at", { ascending: false }),
   140→      sb.from("subscriptions").select("user_id, plan, started_at, expires_at"),
   141→    ]);
   142→
   143→    if (profilesRes.error) throw profilesRes.error;
   144→    if (subsRes.error) throw subsRes.error;
   145→
   146→    const profiles = profilesRes.data || [];
   147→    const subMap = new Map<string, { plan: string; started_at: string | null; expires_at: string | null }>();
   148→    for (const s of (subsRes.data || [])) {
   149→      subMap.set(s.user_id, s);
   150→    }
   151→
   152→    // Try to get emails from auth.users via admin API (list users)
   153→    let emailMap = new Map<string, string>();
   154→    try {
   155→      const authRes = await fetch(`${env.SUPABASE_URL}/auth/v1/admin/users?limit=1000`, {
   156→        headers: { Authorization: `Bearer ${env.SUPABASE_SERVICE_KEY}`, apikey: env.SUPABASE_ANON_KEY },
   157→      });
   158→      if (authRes.ok) {
   159→        const authData = await authRes.json() as { users: { id: string; email: string }[] };
   160→        for (const u of (authData.users || [])) {
   161→          emailMap.set(u.id, u.email);
   162→        }
   163→      }
   164→    } catch {
   165→      // If admin API not available, proceed without emails
   166→    }
   167→
   168→    const users = profiles.map((p: { id: string; user_id?: string; name?: string; created_at: string }) => {
   169→      const uid = p.user_id || p.id;
   170→      const sub = subMap.get(uid);
   171→      return {
   172→        user_id: uid,
   173→        id: p.id,
   174→        name: p.name || "—",
   175→        email: emailMap.get(uid) || "—",
   176→        plan: sub?.plan || "free",
   177→        created_at: p.created_at,
   178→      };
   179→    });
   180→
   181→    return json({ data: users });
   182→  }
   183→
   184→  // GET /api/admin/subscriptions/update — should be POST, reject GET
   185→  // POST /api/admin/subscription/update — Manually update a user's plan by email
   186→  if (path === "/admin/subscription/update" && req.method === "POST") {
   187→    const body = await getBody(req);
   188→    const email = (body.email as string || "").trim().toLowerCase();
   189→    const plan = body.plan as string;
   190→    const expiresAt = body.expires_at as string | undefined;
   191→
   192→    if (!email || !plan) return json({ error: "email dan plan required" }, 400);
   193→    if (plan !== "free" && plan !== "pro") return json({ error: "Plan harus 'free' atau 'pro'" }, 400);
   194→
   195→    // Resolve user_id from email via Supabase Auth admin API
   196→    let userId = body.user_id as string | undefined; // optional override
   197→    if (!userId) {
   198→      try {
   199→        const authRes = await fetch(`${env.SUPABASE_URL}/auth/v1/admin/users?limit=1000`, {
   200→          headers: { Authorization: `Bearer ${env.SUPABASE_SERVICE_KEY}`, apikey: env.SUPABASE_ANON_KEY },
   201→        });
   202→        if (authRes.ok) {
   203→          const authData = await authRes.json() as { users: { id: string; email: string }[] };
   204→          const found = (authData.users || []).find(u => u.email.toLowerCase() === email);
   205→          if (found) userId = found.id;
   206→        }
   207→      } catch {}
   208→    }
   209→    if (!userId) return json({ error: `User dengan email "${email}" tidak ditemukan` }, 404);
   210→
   211→    // Check if subscription exists
   212→    const { data: existing } = await sb.from("subscriptions").select("id").eq("user_id", userId).single();
   213→
   214→    let result;

   216→      const updateData: Record<string, unknown> = { plan, started_at: new Date().toISOString() };

   218→        updateData.expires_at = expiresAt || new Date(Date.now() + 30 * 86400000).toISOString();
   219→      } else {
   220→        updateData.expires_at = expiresAt || null;
   221→      }
   222→      result = await sb.from("subscriptions").update(updateData).eq("user_id", userId).select().single();
   223→    } else {
   224→      const insertData: Record<string, unknown> = { user_id: userId, plan };
   225→      if (plan === "pro") {
   226→        insertData.expires_at = expiresAt || new Date(Date.now() + 30 * 86400000).toISOString();
   227→      }
   228→      result = await sb.from("subscriptions").insert(insertData).select().single();
   229→    }
   230→
   231→    if (result.error) throw result.error;
   232→    return json({ data: result.data, email, user_id: userId });

   235→  // GET /api/admin/subscriptions/update — method not allowed

   242→  // GET /api/admin/subscriptions — List all subscriptions

   244→    if (req.method !== "GET") return json({ error: "Method not allowed" }, 405);
   245→
   246→    const { data, error } = await sb.from("subscriptions").select("*").order("started_at", { ascending: false, nullsFirst: "last" });
   247→    if (error) throw error;
   248→
   249→    // Get emails
   250→    let emailMap = new Map<string, string>();
   251→    try {
   252→      const authRes = await fetch(`${env.SUPABASE_URL}/auth/v1/admin/users?limit=1000`, {
   253→        headers: { Authorization: `Bearer ${env.SUPABASE_SERVICE_KEY}`, apikey: env.SUPABASE_ANON_KEY },
   254→      });
   255→      if (authRes.ok) {
   256→        const authData = await authRes.json() as { users: { id: string; email: string }[] };
   257→        for (const u of (authData.users || [])) {
   258→          emailMap.set(u.id, u.email);

   262→      // proceed without emails
   263→    }
   264→
   265→    const subs = (data || []).map((s: { user_id: string; [key: string]: unknown }) => ({
   266→      ...s,
   267→      email: emailMap.get(s.user_id) || "—",

   273→  // GET /api/admin/transactions — Recent transactions across all users
   274→  if (path === "/admin/transactions" && req.method === "GET") {

   288→    if (to) query = query.lte("occurred_at", to);

   290→
   291→    const { data, error } = await query;
   292→    if (error) throw error;
   293→
   294→    // Get emails for the user_ids in the transactions
   295→    const userIds = [...new Set((data || []).map((t: { user_id: string }) => t.user_id))];
   296→    let emailMap = new Map<string, string>();
   297→    if (userIds.length > 0) {
   298→      try {
   299→        const authRes = await fetch(`${env.SUPABASE_URL}/auth/v1/admin/users?limit=1000`, {
   300→          headers: { Authorization: `Bearer ${env.SUPABASE_SERVICE_KEY}`, apikey: env.SUPABASE_ANON_KEY },

   302→        if (authRes.ok) {
   303→          const authData = await authRes.json() as { users: { id: string; email: string }[] };

   389→      headers: { Authorization: `Bearer ${env.SUPABASE_SERVICE_KEY}`, apikey: env.SUPABASE_ANON_KEY },

   399→      body: JSON.stringify({ email: userData.email }),
   400→    });
   401→
   402→    return json({ ok: true, message: `Link reset password dikirim ke ${userData.email}` });
   403→  }
   404→
   405→  return json({ error: "Not found" }, 404);
   406→}

   408→// ─── API: /api/wallets ──────────────────────────────────────────
   409→async function handleWallets(req: Request, env: Env): Promise<Response> {
   410→  const url = new URL(req.url);
   411→  const admin = getAdmin(env);
   412→
   413→  if (req.method === "GET") {
   414→    const userId = url.searchParams.get("user_id");
   415→    if (!userId) return json({ error: "user_id required" }, 400);
   416→    const { data, error } = await admin
   417→      .from("wallets")
   418→      .select("*")
   419→      .eq("user_id", userId)
   420→      .order("created_at", { ascending: true });
   421→    if (error) throw error;
   422→    return json({ data });
   423→  }
   424→
   425→  if (req.method === "POST") {
   426→    const body = await getBody(req);
   427→    const userId = body.user_id as string;
   428→
   429→    // Check plan limit (with KV cache)
   430→    const isPro = await isProUser(env, userId);
   431→    if (!isPro) {
   432→      const { count } = await admin
   433→        .from("wallets")
   434→        .select("*", { count: "exact", head: true })
   435→        .eq("user_id", userId);
   436→      if ((count || 0) >= 2) {
   437→        return json(
   438→          { error: "Limit dompet tercapai (2 untuk Gratis). Upgrade ke Pro untuk unlimited.", limit: "wallets" },
   439→          403
   440→        );
   441→      }
   442→    }
   443→
   444→    const { data, error } = await admin.from("wallets").insert(body).select().single();
   445→    if (error) throw error;
   446→    return json({ data }, 201);
   447→  }
   448→
   449→  if (req.method === "DELETE") {
   450→    const id = url.searchParams.get("id");
   451→    if (!id) return json({ error: "id required" }, 400);
   452→    const { error } = await admin.from("wallets").delete().eq("id", id);
   453→    if (error) throw error;
   454→    return json({ ok: true });
   455→  }
   456→
   457→  return json({ error: "Method not allowed" }, 405);
   458→}

   460→// ─── API: /api/transactions ─────────────────────────────────────
   461→async function handleTransactions(req: Request, env: Env): Promise<Response> {
   462→  const url = new URL(req.url);
   463→  const admin = getAdmin(env);
   464→
   465→  if (req.method === "GET") {
   466→    const userId = url.searchParams.get("user_id");
   467→    if (!userId) return json({ error: "user_id required" }, 400);
   468→
   469→    let query = admin
   470→      .from("transactions")
   471→      .select("*")
   472→      .eq("user_id", userId)
   473→      .order("occurred_at", { ascending: false });
   474→
   475→    const from = url.searchParams.get("from");
   476→    const to = url.searchParams.get("to");
   477→    const type = url.searchParams.get("type");
   478→    const category = url.searchParams.get("category");
   479→    const wallet = url.searchParams.get("wallet");
   480→
   481→    if (from) query = query.gte("occurred_at", from);
   482→    if (to) query = query.lte("occurred_at", to);
   483→    if (type) query = query.eq("type", type);
   484→    if (category) query = query.eq("category", category);
   485→    if (wallet) query = query.eq("wallet_id", wallet);
   486→
   487→    const { data, error } = await query;
   488→    if (error) throw error;
   489→    return json({ data });

   598→async function handleSummary(req: Request, env: Env): Promise<Response> {
   599→  const url = new URL(req.url);
   600→  const userId = url.searchParams.get("user_id");
   601→  if (!userId) return json({ error: "user_id required" }, 400);
   602→
   603→  const admin = getAdmin(env);
   604→  const [txnsRes, debtsRes] = await Promise.all([
   605→    admin.from("transactions").select("*").eq("user_id", userId),
   606→    admin.from("debts").select("*").eq("user_id", userId),
   607→  ]);
   608→
   609→  if (txnsRes.error) throw txnsRes.error;
   610→  if (debtsRes.error) throw debtsRes.error;
   611→
   612→  const transactions = txnsRes.data || [];
   613→  const debts = debtsRes.data || [];
   614→
   615→  const totalIncome = transactions.filter((t: { type: string }) => t.type === "income").reduce((s: number, t: { amount: number }) => s + t.amount, 0);
   616→  const totalExpenses = transactions.filter((t: { type: string }) => t.type === "expense").reduce((s: number, t: { amount: number }) => s + t.amount, 0);
   617→  const activeBalance = totalIncome - totalExpenses;
   618→  const totalDebt = debts.filter((d: { type: string; status: string }) => d.type === "debt" && d.status !== "paid").reduce((s: number, d: { amount: number; paid_amount: number }) => s + (d.amount - d.paid_amount), 0);
   619→  const totalReceivable = debts.filter((d: { type: string; status: string }) => d.type === "receivable" && d.status !== "paid").reduce((s: number, d: { amount: number; paid_amount: number }) => s + (d.amount - d.paid_amount), 0);
   620→
   621→  const now = new Date();
   622→  const monthlyData = [];
   623→  for (let i = 11; i >= 0; i--) {
   624→    const mStart = new Date(now.getFullYear(), now.getMonth() - i, 1);
   625→    const mEnd = new Date(now.getFullYear(), now.getMonth() - i + 1, 1);
   626→    const mTxns = transactions.filter((t: { occurred_at: string }) => { const d = new Date(t.occurred_at); return d >= mStart && d < mEnd; });
   627→    const mIncome = mTxns.filter((t: { type: string }) => t.type === "income").reduce((s: number, t: { amount: number }) => s + t.amount, 0);
   628→    const mExpenses = mTxns.filter((t: { type: string }) => t.type === "expense").reduce((s: number, t: { amount: number }) => s + t.amount, 0);
   629→    monthlyData.push({ month: mStart.toLocaleDateString("en-US", { month: "short", year: "2-digit" }), income: mIncome, expenses: mExpenses, net: mIncome - mExpenses });
   630→  }
   631→
   632→  return json({ activeBalance, totalIncome, totalExpenses, totalDebt, totalReceivable, monthlyData });
   633→}

   653→    }
   654→  }

   656→}

   725→

   728→        id: t.id, user_id: userId, wallet_id: t.wallet_id, type: t.type, category: t.category,

   733→      if (error) throw error;
   734→      results.transactions = r;
   735→    }
   736→
   737→    if ((syncData.debts as unknown[])?.length) {
   738→      const debts = (syncData.debts as Record<string, unknown>[]).map((d) => ({

   764→    ]);

   768→        wallets: wallets.data || [],
   769→        transactions: transactions.data || [],
   770→        debts: debts.data || [],

   774→      },
   775→    });
   776→  }

   942→  const userId = metadata?.user_id || (body.custom_field1 as string);
   943→
   944→  if (!orderId || !transactionStatus) {
   945→    return json({ error: "Invalid webhook payload" }, 400);
   946→  }
   947→
   948→  // Verify signature (optional but recommended for production)
   949→  // const signatureKey = body.signature_key;
   950→  // ... verify with SHA512(order_id + status + gross_amount + server_key)
   951→
   952→  const admin = getAdmin(env);
   953→
   954→  // Determine if payment is successful
   955→  const isSuccess =
   956→    transactionStatus === "capture" ||
   957→    transactionStatus === "settlement";
   958→
   959→  const isPending = transactionStatus === "pending";
   960→  const isFailure =
   961→    transactionStatus === "deny" ||
   962→    transactionStatus === "expire" ||
   963→    transactionStatus === "cancel";
   964→
   965→  if (isSuccess && userId) {
   966→    // Activate Pro plan
   967→    const expiresAt = new Date(Date.now() + 30 * 86400000).toISOString();
   968→    const { error } = await admin
   969→      .from("subscriptions")
   970→      .update({
   971→        plan: "pro",
   972→        started_at: new Date().toISOString(),
   973→        expires_at: expiresAt,
   974→        payment_method: `midtrans_${paymentType}`,
   975→        transaction_id: orderId,
   976→      })
   977→      .eq("user_id", userId);
   978→
   979→    if (error) console.error("Failed to update subscription:", error);
   980→  } else if (isFailure && userId) {
   981→    // Revert to free
   982→    await admin
   983→      .from("subscriptions")
   984→      .update({ plan: "free", transaction_id: orderId })
   985→      .eq("user_id", userId);
   986→  }
   987→
   988→  // Always return 200 to Midtrans
   989→  return json({ ok: true });
   990→}

   993→async function handleApi(request: Request, env: Env): Promise<Response> {
   994→  const url = new URL(request.url);
   995→  const path = url.pathname.replace("/api", "").replace("//", "/");
   996→
   997→  try {
   998→    // Check critical env vars are set
   999→    if (!env.SUPABASE_URL || !env.SUPABASE_ANON_KEY || !env.SUPABASE_SERVICE_KEY) {
  1000→      const missing: string[] = [];

  1003→      if (!env.SUPABASE_SERVICE_KEY) missing.push("SUPABASE_SERVICE_KEY");

  1005→        error: `Worker secrets belum dikonfigurasi: ${missing.join(", ")}. Set via dashboard → Worker → Settings → Variables and Secrets.`,
  1006→        missing,
  1007→        hint: "Pastikan nama secret persis: SUPABASE_ANON_KEY dan SUPABASE_SERVICE_KEY (bukan NEXT_PUBLIC_...)",
  1008→      }, 503);
  1009→    }
  1010→
  1011→    // Route matching
  1012→    if (path === "/" || path === "") return json({ message: "Arus Kas API v3.1", status: "ok" });
  1013→
  1014→    if (path === "/wallets" || path.startsWith("/wallets")) return await handleWallets(request, env);
  1015→    if (path === "/transactions" || path.startsWith("/transactions")) return await handleTransactions(request, env);
  1016→    if (path === "/debts" || path.startsWith("/debts")) return await handleDebts(request, env);
  1017→    if (path === "/subscription" || path.startsWith("/subscription")) return await handleSubscription(request, env);
  1018→    if (path === "/summary" || path.startsWith("/summary")) return await handleSummary(request, env);
  1019→    if (path === "/categorize" || path.startsWith("/categorize")) return await handleCategorize(request);
  1020→    if (path === "/auth/login") return await handleLogin(request, env);
  1021→    if (path === "/auth/signup") return await handleSignup(request, env);
  1022→    if (path === "/auth/me") return await handleAuthMe(request, env);
  1023→    if (path === "/auth/logout") return await handleLogout(request, env);
  1024→    if (path === "/auth/forgot-password") return await handleForgotPassword(request, env);
  1025→    if (path === "/auth/reset-password") return await handleResetPassword(request, env);
  1026→    if (path === "/sync" || path.startsWith("/sync")) return await handleSync(request, env);
  1027→    if (path === "/payment/create") return await handleMidtransCreate(request, env);
  1028→    if (path === "/payment/webhook") return await handleMidtransWebhook(request, env);
  1029→    // R2 storage routes — disabled (no R2 binding)
  1030→    if (path === "/storage/backup" || path.startsWith("/storage/backup")) return r2Disabled();
  1031→    if (path === "/storage/restore" || path.startsWith("/storage/restore")) return r2Disabled();
  1032→    if (path === "/storage" || path.startsWith("/storage")) return r2Disabled();
  1033→    if (path === "/cache" || path.startsWith("/cache")) return await handleCache(request, env);
  1034→
  1035→    // Admin routes
  1036→    if (path.startsWith("/admin")) return await handleAdmin(request, env, path);
  1037→
  1038→    return json({ error: "Not found" }, 404);
  1039→  } catch (err: unknown) {
  1040→    const msg = err instanceof Error ? err.message : "Server error";
  1041→    return json({ error: msg }, 500);
  1042→  }
  1043→}

  1045→// ─── Cron: Check expired Pro subscriptions ────────────────────
  1046→async function handleCron(env: Env): Promise<void> {
  1047→  const admin = getAdmin(env);
  1048→  const now = new Date().toISOString();
  1049→
  1050→  // Find all Pro subscriptions that have expired
  1051→  const { data: expired, error } = await admin
  1052→    .from("subscriptions")
  1053→    .select("user_id, expires_at")
  1054→    .eq("plan", "pro")
  1055→    .lt("expires_at", now);
  1056→
  1057→  if (error) {
  1058→    console.error("Cron: failed to query expired subscriptions:", error);
  1059→    return;
  1060→  }
  1061→
  1062→  if (!expired || expired.length === 0) {
  1063→    console.log(`Cron: no expired subscriptions at ${now}`);
  1064→    return;
  1065→  }
  1066→
  1067→  // Downgrade expired Pro users to Free
  1068→  const userIds = expired.map((s: { user_id: string }) => s.user_id);
  1069→  const { error: updateError } = await admin
  1070→    .from("subscriptions")
  1071→    .update({ plan: "free" })
  1072→    .in("user_id", userIds);
  1073→
  1074→  if (updateError) {
  1075→    console.error("Cron: failed to downgrade:", updateError);
  1076→  } else {
  1077→    // Invalidate KV subscription caches for downgraded users
  1078→    for (const uid of userIds) {
  1079→      await invalidateSubscriptionCache(env, uid);
  1080→    }
  1081→    console.log(`Cron: downgraded ${userIds.length} expired Pro → Free`);
  1082→  }
  1083→}

  1176→// ─── API: /api/cache — KV Feature Flags, Cache & Sessions ──────
  1177→async function handleCache(req: Request, env: Env): Promise<Response> {
  1178→  const url = new URL(req.url);
  1179→  const subPath = url.pathname.replace("/api/cache", "").replace("//", "/");
  1180→
  1181→  // ── Feature Flags ──
  1182→  if (subPath === "/flags" || subPath === "/flags/") {
  1183→    if (req.method === "GET") {
  1184→      // Get all feature flags (merged with defaults)
  1185→      const flags: Record<string, string> = { ...DEFAULT_FEATURE_FLAGS };
  1186→      try {
  1187→        const list = await env.arus_kv.list({ prefix: "ff:" });
  1188→        for (const k of list.keys) {
  1189→          const val = await env.arus_kv.get(k.name);
  1190→          if (val !== null) flags[k.name] = val;
  1191→        }
  1192→      } catch {}
  1193→      return json({ flags });
  1194→    }
  1195→
  1196→    if (req.method === "PUT") {
  1197→      // Admin: set a feature flag
  1198→      const body = await getBody(req);
  1199→      const flag = body.flag as string;
  1200→      const value = body.value as string;
  1201→      if (!flag || value === undefined) return json({ error: "flag dan value required" }, 400);
  1202→      await setFeatureFlag(env, flag, value);
  1203→      return json({ ok: true, flag: `ff:${flag}`, value });
  1204→    }
  1205→  }
  1206→
  1207→  // ── Generic KV Get/Put/Delete ──
  1208→  if (req.method === "GET") {
  1209→    const key = url.searchParams.get("key");
  1210→    if (key) {
  1211→      const value = await env.arus_kv.get(key);
  1212→      return json({ key, value });
  1213→    }
  1214→    // List all keys with prefix
  1215→    const prefix = url.searchParams.get("prefix") || "";
  1216→    const list = await env.arus_kv.list({ prefix });
  1217→    return json({ keys: list.keys.map((k) => ({ name: k.name, expiration: k.expiration })) });
  1218→  }
  1219→
  1220→  if (req.method === "PUT") {
  1221→    const body = await getBody(req);
  1222→    const key = body.key as string;
  1223→    const value = body.value as string;
  1224→    const ttl = body.ttl as number | undefined;
  1225→    if (!key || value === undefined) return json({ error: "key dan value required" }, 400);
  1226→    const opts = ttl ? { expirationTtl: ttl } : undefined;
  1227→    await env.arus_kv.put(key, value, opts);
  1228→    return json({ ok: true, key });
  1229→  }
  1230→
  1231→  if (req.method === "DELETE") {
  1232→    const key = url.searchParams.get("key");
  1233→    if (!key) return json({ error: "key required" }, 400);
  1234→    await env.arus_kv.delete(key);
  1235→    return json({ ok: true });
  1236→  }
  1237→
  1238→  return json({ error: "Method not allowed" }, 405);
  1239→}
