   244→// ─── API helper ───────────────────────────────────────────────
   245→async function api(path, opts = {}) {
   246→  const headers = { 'Content-Type': 'application/json', ...(opts.headers || {}) };
   247→  if (token) headers['Authorization'] = 'Bearer ' + token;
   248→  try {
   249→    const res = await fetch('/api' + path, { ...opts, headers });
   250→    const text = await res.text();
   251→    try {
   252→      const data = JSON.parse(text);
   253→      if (!res.ok) throw new Error(data.error || 'Gagal memuat data');
   254→      return data;
   255→    } catch(e) {
   256→      if (!res.ok) throw new Error('Server error (HTTP ' + res.status + ')');
   257→      throw e;
   258→    }
   259→  } catch(e) {
   260→    throw e;
   261→  }
   262→}

   264→// ─── Format helpers ───────────────────────────────────────────
   265→function fmtRp(n) { return 'Rp ' + Number(n||0).toLocaleString('id-ID'); }
   266→function fmtDate(d) { if (!d) return '—'; return new Date(d).toLocaleDateString('id-ID', {day:'numeric',month:'short',year:'numeric'}); }
   267→function fmtRelTime(d) {
   268→  if (!d) return '—';
   269→  const diff = Date.now() - new Date(d).getTime();
   270→  const days = Math.floor(diff / 86400000);
   271→  if (days < 1) return 'Hari ini';
   272→  if (days < 7) return days + ' hari lalu';
   273→  return fmtDate(d);
   274→}

   276→// ─── Login ────────────────────────────────────────────────────
   277→async function doLogin() {
   278→  const email = document.getElementById('loginEmail').value.trim();
   279→  const pass = document.getElementById('loginPass').value;
   280→  const errEl = document.getElementById('loginError');
   281→  const btn = document.getElementById('loginBtn');
   282→  errEl.classList.remove('show');
   283→  btn.disabled = true; btn.textContent = 'Memproses...';
   284→
   285→  try {
   286→    // Step 1: Login to get token
   287→    const loginRes = await fetch('/api/auth/login', {
   288→      method: 'POST',
   289→      headers: { 'Content-Type': 'application/json' },
   290→      body: JSON.stringify({ email, password: pass })
   291→    });
   292→    const loginData = await loginRes.json();
   293→    if (!loginRes.ok) throw new Error(loginData.error_description || loginData.error || 'Login gagal');
   294→
   295→    token = loginData.access_token;
   296→    if (!token) throw new Error('Token tidak ditemukan');
   297→
   298→    // Step 2: Verify user with /api/auth/me
   299→    const meRes = await fetch('/api/auth/me', {
   300→      headers: { 'Authorization': 'Bearer ' + token }
   301→    });
   302→    const meData = await meRes.json();
   303→    if (!meRes.ok) throw new Error('Verifikasi gagal');
   304→
   305→    // Step 3: Check admin email
   306→    const userEmail = meData.data?.email || meData.email;
   307→    if (userEmail !== ADMIN_EMAIL) {
   308→      throw new Error('Akses ditolak. Halaman ini hanya untuk admin.');
   309→    }
   310→
   311→    // Success
   312→    sessionStorage.setItem('arus_admin_token', token);
   313→    document.getElementById('adminEmail').textContent = userEmail;
   314→    document.getElementById('loginScreen').style.display = 'none';
   315→    document.getElementById('dashScreen').style.display = 'flex';
   316→    loadSection('dashboard');
   317→
   318→  } catch (e) {
   319→    errEl.textContent = e.message;
   320→    errEl.classList.add('show');
   321→    token = '';
   322→  } finally {
   323→    btn.disabled = false; btn.textContent = 'Masuk';
   324→  }
   325→}

   907→async function loadKvKeys() {
   908→  const prefix = document.getElementById('kvPrefix')?.value || '';
   909→  const list = document.getElementById('kvKeysList');
   910→  if (!list) return;
   911→  list.innerHTML = '<span class="muted">Memuat...</span>';
   912→  try {
   913→    const res = await api('/cache?prefix=' + encodeURIComponent(prefix));
   914→    const keys = res.keys || [];
   915→    if (!keys.length) {
   916→      list.innerHTML = '<span class="muted">Tidak ada keys dengan prefix "' + esc(prefix) + '"</span>';
   917→    } else {
   918→      list.innerHTML = '<table style="width:100%"><thead><tr><th>Key</th><th>Expiry</th><th>Aksi</th></tr></thead><tbody>' +
   919→        keys.slice(0, 50).map(k => `<tr><td class="mono xs">${esc(k.name)}</td><td class="xs muted">${k.expiration ? new Date(k.expiration * 1000).toLocaleString('id') : 'Permanen'}</td><td><button class="btn mini" onclick="deleteKvKey('${esc(k.name)}')">Hapus</button></td></tr>`).join('') +
   920→        '</tbody></table>';
   921→    }
   922→  } catch (e) {
   923→    list.innerHTML = '<span class="neg">Error: ' + esc(e.message) + '</span>';
   924→  }
   925→}
