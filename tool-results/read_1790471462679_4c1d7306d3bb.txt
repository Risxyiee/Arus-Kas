    67→// ─── SHA-512 helper (for Midtrans signature verification) ────────
    68→async function sha512(message: string): Promise<string> {
    69→  const data = new TextEncoder().encode(message);
    70→  const hash = await crypto.subtle.digest("SHA-512", data);
    71→  return Array.from(new Uint8Array(hash)).map(b => b.toString(16).padStart(2, "0")).join("");
    72→}

    74→// ─── Admin Auth Verification (with KV session cache) ───────────
    75→async function verifyAdmin(req: Request, env: Env): Promise<{ userId: string; email: string } | null> {
    76→  const authHeader = req.headers.get("Authorization");
    77→  if (!authHeader || !authHeader.startsWith("Bearer ")) return null;
    78→  const token = authHeader.replace("Bearer ", "");
    79→
    80→  // Check KV session cache first
    81→  const cached = await getCachedSession(env, token);
    82→  if (cached) {
    83→    const adminEmail = env.ADMIN_EMAIL || "riskiakbarp123@gmail.com";
    84→    if (cached.email !== adminEmail) return null;
    85→    return cached;
    86→  }
    87→
    88→  // Verify token+cache session
    89→  const res = await fetch(`${env.SUPABASE_URL}/auth/v1/user`, {
    90→    headers: { Authorization: `Bearer ${token}`, apikey: env.SUPABASE_ANON_KEY },
    91→  });
    92→  if (!res.ok) return null;
    93→  const user = await res.json() as { id: string; email: string };
    94→
    95→  const adminEmail = env.ADMIN_EMAIL || "riskiakbarp123@gmail.com";
    96→  if (user.email !== adminEmail) return null;
    97→
    98→  // Cache the session
    99→  const sessionData = { userId: user.id, email: user.email };
   100→  await cacheSession(env, token, sessionData);
   101→
   102→  return sessionData;
   103→}

   420→async function handleWallets(req: Request, env: Env): Promise<Response> {
   421→  const url = new URL(req.url);
   422→  const admin = getAdmin(env);
   423→
   424→  if (req.method === "GET") {
   425→    const userId = url.searchParams.get("user_id");
   426→    if (!userId) return json({ error: "user_id required" }, 400);
   427→    const { data, error } = await admin
   428→      .from("wallets")
   429→      .select("*")
   430→      .eq("user_id", userId)
   431→      .order("created_at", { ascending: true });
   432→    if (error) throw error;
   433→    return json({ data });
   434→  }
   435→
   436→  if (req.method === "POST") {
   437→    const body = await getBody(req);
   438→    const userId = body.user_id as string;
   439→
   440→    // Check plan limit (with KV cache)
   441→    const isPro = await isProUser(env, userId);
   442→    if (!isPro) {
   443→      const { count } = await admin
   444→        .from("wallets")
   445→        .select("*", { count: "exact", head: true })
   446→        .eq("user_id", userId);
   447→      if ((count || 0) >= 2) {
   448→        return json(
   449→          { error: "Limit dompet tercapai (2 untuk Gratis). Upgrade ke Pro untuk unlimited.", limit: "wallets" },
   450→          403
   451→        );
   452→      }
   453→    }
   454→
   455→    const { data, error } = await admin.from("wallets").insert(body).select().single();
   456→    if (error) throw error;
   457→    return json({ data }, 201);
   458→  }
   459→
   460→  if (req.method === "DELETE") {
   461→    const id = url.searchParams.get("id");
   462→    if (!id) return json({ error: "id required" }, 400);
   463→    const { error } = await admin.from("wallets").delete().eq("id", id);
   464→    if (error) throw error;
   465→    return json({ ok: true });
   466→  }
   467→
   468→  return json({ error: "Method not allowed" }, 405);
   469→}

   471→// ─── API: /api/transactions ─────────────────────────────────────
   472→async function handleTransactions(req: Request, env: Env): Promise<Response> {
   473→  const url = new URL(req.url);

   480→    let query = admin
   481→      .from("transactions")
   482→      .select("*")
   483→      .eq("user_id", userId)
   484→      .order("occurred_at", { ascending: false });
   485→
   486→    const from = url.searchParams.get("from");
   487→    const to = url.searchParams.get("to");

   609→async function handleSummary(req: Request, env: Env): Promise<Response> {
   610→  const url = new URL(req.url);
   611→  const userId = url.searchParams.get("user_id");
   612→  if (!userId) return json({ error: "user_id required" }, 400);
   613→
   614→  const admin = getAdmin(env);
   615→  const [txnsRes, debtsRes] = await Promise.all([
   616→    admin.from("transactions").select("*").eq("user_id", userId),
   617→    admin.from("debts").select("*").eq("user_id", userId),
   618→  ]);
   619→
   620→  if (txnsRes.error) throw txnsRes.error;
   621→  if (debtsRes.error) throw debtsRes.error;
   622→
   623→  const transactions = txnsRes.data || [];
   624→  const debts = debtsRes.data || [];
   625→
   626→  const totalIncome = transactions.filter((t: { type: string }) => t.type === "income").reduce((s: number, t: { amount: number }) => s + t.amount, 0);
   627→  const totalExpenses = transactions.filter((t: { type: string }) => t.type === "expense").reduce((s: number, t: { amount: number }) => s + t.amount, 0);
   628→  const activeBalance = totalIncome - totalExpenses;
   629→  const totalDebt = debts.filter((d: { type: string; status: string }) => d.type === "debt" && d.status !== "paid").reduce((s: number, d: { amount: number; paid_amount: number }) => s + (d.amount - d.paid_amount), 0);
   630→  const totalReceivable = debts.filter((d: { type: string; status: string }) => d.type === "receivable" && d.status !== "paid").reduce((s: number, d: { amount: number; paid_amount: number }) => s + (d.amount - d.paid_amount), 0);
   631→
   632→  const now = new Date();
   633→  const monthlyData = [];
   634→  for (let i = 11; i >= 0; i--) {
   635→    const mStart = new Date(now.getFullYear(), now.getMonth() - i, 1);
   636→    const mEnd = new Date(now.getFullYear(), now.getMonth() - i + 1, 1);
   637→    const mTxns = transactions.filter((t: { occurred_at: string }) => { const d = new Date(t.occurred_at); return d >= mStart && d < mEnd; });
   638→    const mIncome = mTxns.filter((t: { type: string }) => t.type === "income").reduce((s: number, t: { amount: number }) => s + t.amount, 0);
   639→    const mExpenses = mTxns.filter((t: { type: string }) => t.type === "expense").reduce((s: number, t: { amount: number }) => s + t.amount, 0);
   640→    monthlyData.push({ month: mStart.toLocaleDateString("en-US", { month: "short", year: "2-digit" }), income: mIncome, expenses: mExpenses, net: mIncome - mExpenses });
   641→  }
   642→
   643→  return json({ activeBalance, totalIncome, totalExpenses, totalDebt, totalReceivable, monthlyData });
   644→}

   733→      if (error) throw error;
   734→      results.wallets = r;

   736→
   737→    if ((syncData.transactions as unknown[])?.length) {
   738→      const txns = (syncData.transactions as Record<string, unknown>[]).map((t) => ({
   739→        id: t.id, user_id: userId, wallet_id: t.wallet_id, type: t.type, category: t.category,

   743→      const { data: r, error } = await admin.from("transactions").upsert(txns, { onConflict: "id" });
   744→      if (error) throw error;
   745→      results.transactions = r;
   746→    }
   747→
   748→    if ((syncData.debts as unknown[])?.length) {
   749→      const debts = (syncData.debts as Record<string, unknown>[]).map((d) => ({
   750→        id: d.id, user_id: userId, name: d.name, type: d.type, amount: d.amount,

   772→      admin.from("budgets").select("*").eq("user_id", userId),
   773→      admin.from("custom_categories").select("*").eq("user_id", userId),
   774→      admin.from("subscriptions").select("*").eq("user_id", userId).single(),
   775→    ]);

   779→        wallets: wallets.data || [],
   780→        transactions: transactions.data || [],
   781→        debts: debts.data || [],
   782→        budgets: (budgets.data || []).reduce((acc: Record<string, number>, b: { category: string; amount: number }) => { acc[b.category] = b.amount; return acc; }, {}),

   784→        subscription: subscription.data || { plan: "free" },
   785→      },
   786→    });
   787→  }
   788→
   789→  return json({ error: "Method not allowed" }, 405);

   953→// ─── Midtrans: Webhook Handler ──────────────────────────────────
   954→async function handleMidtransWebhook(req: Request, env: Env): Promise<Response> {
   955→  const body = await getBody(req);
   956→
   957→  const transactionStatus = body.transaction_status as string;
   958→  const orderId = body.order_id as string;
   959→  const paymentType = body.payment_type as string;
   960→  const metadata = body.metadata as { user_id?: string; plan?: string } | undefined;
   961→  const userId = metadata?.user_id || (body.custom_field1 as string);
   962→
   963→  if (!orderId || !transactionStatus) {
   964→    return json({ error: "Invalid webhook payload" }, 400);
   965→  }
   966→
   967→  // Verify signature to prevent fake webhook attacks
   968→  const signatureKey = body.signature_key as string;
   969→  const grossAmount = body.gross_amount as string;
   970→  if (signatureKey && env.MIDTRANS_SERVER_KEY) {
   971→    const expectedSig = await sha512(orderId + transactionStatus + grossAmount + env.MIDTRANS_SERVER_KEY);
   972→    if (signatureKey !== expectedSig) {
   973→      return json({ error: "Invalid signature" }, 403);
   974→    }
   975→  }
   976→
   977→  const admin = getAdmin(env);
   978→
   979→  // Determine if payment is successful
   980→  const isSuccess =
   981→    transactionStatus === "capture" ||
   982→    transactionStatus === "settlement";
   983→
   984→  const isPending = transactionStatus === "pending";
   985→  const isFailure =
   986→    transactionStatus === "deny" ||
   987→    transactionStatus === "expire" ||
   988→    transactionStatus === "cancel";
   989→
   990→  if (isSuccess && userId) {
   991→    // Activate Pro plan
   992→    const expiresAt = new Date(Date.now() + 30 * 86400000).toISOString();
   993→    const { error } = await admin
   994→      .from("subscriptions")
   995→      .update({
   996→        plan: "pro",
   997→        started_at: new Date().toISOString(),
   998→        expires_at: expiresAt,
   999→        payment_method: `midtrans_${paymentType}`,
  1000→        transaction_id: orderId,
  1001→      })
  1002→      .eq("user_id", userId);
  1003→
  1004→    if (error) console.error("Failed to update subscription:", error);
  1005→    else await invalidateSubscriptionCache(env, userId);
  1006→  } else if (isFailure && userId) {
  1007→    // Revert to free
  1008→    await admin
  1009→      .from("subscriptions")
  1010→      .update({ plan: "free", transaction_id: orderId })
  1011→      .eq("user_id", userId);
  1012→    await invalidateSubscriptionCache(env, userId);
  1013→  }
  1014→
  1015→  // Always return 200 to Midtrans
  1016→  return json({ ok: true });
  1017→}

  1020→async function handleApi(request: Request, env: Env): Promise<Response> {
  1021→  const url = new URL(request.url);
  1022→  const path = url.pathname.replace("/api", "").replace("//", "/");
  1023→
  1024→  try {
  1025→    // Check critical env vars are set
  1026→    if (!env.SUPABASE_URL || !env.SUPABASE_ANON_KEY || !env.SUPABASE_SERVICE_KEY) {
  1027→      const missing: string[] = [];
  1028→      if (!env.SUPABASE_URL) missing.push("SUPABASE_URL");
  1029→      if (!env.SUPABASE_ANON_KEY) missing.push("SUPABASE_ANON_KEY");
  1030→      if (!env.SUPABASE_SERVICE_KEY) missing.push("SUPABASE_SERVICE_KEY");
  1031→      return json({
  1032→        error: `Worker secrets belum dikonfigurasi: ${missing.join(", ")}. Set via dashboard → Worker → Settings → Variables and Secrets.`,
  1033→        missing,
  1034→        hint: "Pastikan nama secret persis: SUPABASE_ANON_KEY dan SUPABASE_SERVICE_KEY (bukan NEXT_PUBLIC_...)",
  1035→      }, 503);
  1036→    }
  1037→
  1038→    // Route matching
  1039→    if (path === "/" || path === "") return json({ message: "Arus Kas API v3.1", status: "ok" });
  1040→
  1041→    if (path === "/wallets" || path.startsWith("/wallets")) return await handleWallets(request, env);
  1042→    if (path === "/transactions" || path.startsWith("/transactions")) return await handleTransactions(request, env);
  1043→    if (path === "/debts" || path.startsWith("/debts")) return await handleDebts(request, env);
  1044→    if (path === "/subscription" || path.startsWith("/subscription")) return await handleSubscription(request, env);
  1045→    if (path === "/summary" || path.startsWith("/summary")) return await handleSummary(request, env);
  1046→    if (path === "/categorize" || path.startsWith("/categorize")) return await handleCategorize(request);
  1047→    if (path === "/auth/login") return await handleLogin(request, env);
  1048→    if (path === "/auth/signup") return await handleSignup(request, env);
  1049→    if (path === "/auth/me") return await handleAuthMe(request, env);
  1050→    if (path === "/auth/logout") return await handleLogout(request, env);
  1051→    if (path === "/auth/forgot-password") return await handleForgotPassword(request, env);
  1052→    if (path === "/auth/reset-password") return await handleResetPassword(request, env);
  1053→    if (path === "/sync" || path.startsWith("/sync")) return await handleSync(request, env);
  1054→    if (path === "/payment/create") return await handleMidtransCreate(request, env);
  1055→    if (path === "/payment/webhook") return await handleMidtransWebhook(request, env);
  1056→    // R2 storage routes — disabled (no R2 binding)
  1057→    if (path === "/storage/backup" || path.startsWith("/storage/backup")) return r2Disabled();
  1058→    if (path === "/storage/restore" || path.startsWith("/storage/restore")) return r2Disabled();
  1059→    if (path === "/storage" || path.startsWith("/storage")) return r2Disabled();
  1060→    if (path === "/cache" || path.startsWith("/cache")) return await handleCache(request, env);
  1061→
  1062→    // Admin routes
  1063→    if (path.startsWith("/admin")) return await handleAdmin(request, env, path);
  1064→
  1065→    return json({ error: "Not found" }, 404);
  1066→  } catch (err: unknown) {
  1067→    const msg = err instanceof Error ? err.message : "Server error";
  1068→    return json({ error: msg }, 500);
  1069→  }
  1070→}

  1072→// ─── Cron: Check expired Pro subscriptions ────────────────────
  1073→async function handleCron(env: Env): Promise<void> {
  1074→  const admin = getAdmin(env);
  1075→  const now = new Date().toISOString();
  1076→
  1077→  // Find all Pro subscriptions that have expired
  1078→  const { data: expired, error } = await admin
  1079→    .from("subscriptions")
  1080→    .select("user_id, expires_at")
  1081→    .eq("plan", "pro")
  1082→    .lt("expires_at", now);
  1083→
  1084→  if (error) {
  1085→    console.error("Cron: failed to query expired subscriptions:", error);
  1086→    return;
  1087→  }
  1088→
  1089→  if (!expired || expired.length === 0) {
  1090→    console.log(`Cron: no expired subscriptions at ${now}`);
  1091→    return;
  1092→  }
  1093→
  1094→  // Downgrade expired Pro users to Free
  1095→  const userIds = expired.map((s: { user_id: string }) => s.user_id);
  1096→  const { error: updateError } = await admin
  1097→    .from("subscriptions")
  1098→    .update({ plan: "free" })
  1099→    .in("user_id", userIds);
  1100→
  1101→  if (updateError) {
  1102→    console.error("Cron: failed to downgrade:", updateError);
  1103→  } else {
  1104→    // Invalidate KV subscription caches for downgraded users
  1105→    for (const uid of userIds) {
  1106→      await invalidateSubscriptionCache(env, uid);
  1107→    }
  1108→    console.log(`Cron: downgraded ${userIds.length} expired Pro → Free`);
  1109→  }
  1110→}

  1203→// ─── API: /api/cache — KV Feature Flags, Cache & Sessions ──────
  1204→async function handleCache(req: Request, env: Env): Promise<Response> {
  1205→  const url = new URL(req.url);
  1206→  const subPath = url.pathname.replace("/api/cache", "").replace("//", "/");
  1207→
  1208→  // ── Feature Flags ──
  1209→  if (subPath === "/flags" || subPath === "/flags/") {
  1210→    if (req.method === "GET") {
  1211→      // Get all feature flags (merged with defaults)
  1212→      const flags: Record<string, string> = { ...DEFAULT_FEATURE_FLAGS };
  1213→      try {
  1214→        const list = await env.arus_kv.list({ prefix: "ff:" });
  1215→        for (const k of list.keys) {
  1216→          const val = await env.arus_kv.get(k.name);
  1217→          if (val !== null) flags[k.name] = val;
  1218→        }
  1219→      } catch {}
  1220→      return json({ flags });
  1221→    }
  1222→
  1223→    if (req.method === "PUT") {
  1224→      // Admin: set a feature flag
  1225→      const body = await getBody(req);
  1226→      const flag = body.flag as string;
  1227→      const value = body.value as string;
  1228→      if (!flag || value === undefined) return json({ error: "flag dan value required" }, 400);
  1229→      await setFeatureFlag(env, flag, value);
  1230→      return json({ ok: true, flag: `ff:${flag}`, value });
  1231→    }
  1232→  }
  1233→
  1234→  // ── Generic KV Get/Put/Delete ──
  1235→  if (req.method === "GET") {
  1236→    const key = url.searchParams.get("key");
  1237→    if (key) {
  1238→      const value = await env.arus_kv.get(key);
  1239→      return json({ key, value });
  1240→    }
  1241→    // List all keys with prefix
  1242→    const prefix = url.searchParams.get("prefix") || "";
  1243→    const list = await env.arus_kv.list({ prefix });
  1244→    return json({ keys: list.keys.map((k) => ({ name: k.name, expiration: k.expiration })) });
  1245→  }
  1246→
  1247→  if (req.method === "PUT") {
  1248→    const body = await getBody(req);
  1249→    const key = body.key as string;
  1250→    const value = body.value as string;
  1251→    const ttl = body.ttl as number | undefined;
  1252→    if (!key || value === undefined) return json({ error: "key dan value required" }, 400);
  1253→    const opts = ttl ? { expirationTtl: ttl } : undefined;
  1254→    await env.arus_kv.put(key, value, opts);
  1255→    return json({ ok: true, key });
  1256→  }
  1257→
  1258→  if (req.method === "DELETE") {
  1259→    const key = url.searchParams.get("key");
  1260→    if (!key) return json({ error: "key required" }, 400);
  1261→    await env.arus_kv.delete(key);
  1262→    return json({ ok: true });
  1263→  }
  1264→
  1265→  return json({ error: "Method not allowed" }, 405);
  1266→}
