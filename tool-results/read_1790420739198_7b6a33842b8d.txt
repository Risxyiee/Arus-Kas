    74→// ─── Admin Auth Verification (with KV session cache) ───────────
    75→async function verifyAdmin(req: Request, env: Env): Promise<{ userId: string; email: string } | null> {
    76→  const authHeader = req.headers.get("Authorization");
    77→  if (!authHeader || !authHeader.startsWith("Bearer ")) return null;
    78→  const token = authHeader.replace("Bearer ", "");
    79→
    80→  // Check KV session cache first
    81→  const cached = await getCachedSession(env, token);
    82→  if (cached) {
    83→    const adminEmail = env.ADMIN_EMAIL || "riskiakbarp123@gmail.com";
    84→    if (cached.email !== adminEmail) return null;
    85→    return cached;
    86→  }
    87→
    88→  // Verify token+cache session
    89→  const res = await fetch(`${env.SUPABASE_URL}/auth/v1/user`, {
    90→    headers: { Authorization: `Bearer ${token}`, apikey: env.SUPABASE_ANON_KEY },
    91→  });
    92→  if (!res.ok) return null;
    93→  const user = await res.json() as { id: string; email: string };
    94→
    95→  const adminEmail = env.ADMIN_EMAIL || "riskiakbarp123@gmail.com";
    96→  if (user.email !== adminEmail) return null;
    97→
    98→  // Cache the session
    99→  const sessionData = { userId: user.id, email: user.email };
   100→  await cacheSession(env, token, sessionData);
   101→
   102→  return sessionData;
   103→}

   105→// ─── API: /api/admin/* ────────────────────────────────────────
   106→async function handleAdmin(req: Request, env: Env, path: string): Promise<Response> {
   107→  // Verify admin access for all admin routes
   108→  const admin = await verifyAdmin(req, env);
   109→  if (!admin) return json({ error: "Akses ditolak. Halaman ini hanya untuk admin." }, 403);
   110→
   111→  const sb = getAdmin(env);
   112→  const url = new URL(req.url);
   113→
   114→  // GET /api/admin/stats — Dashboard overview stats
   115→  if (path === "/admin/stats" && req.method === "GET") {
   116→    const sevenDaysAgo = new Date(Date.now() - 7 * 86400000).toISOString();
   117→
   118→    const [profilesRes, subsRes, recentRes] = await Promise.all([
   119→      sb.from("profiles").select("id, created_at", { count: "exact" }),
   120→      sb.from("subscriptions").select("plan, expires_at"),
   121→      sb.from("profiles").select("id").gte("created_at", sevenDaysAgo),
   122→    ]);
   123→
   124→    if (profilesRes.error) throw profilesRes.error;
   125→    if (subsRes.error) throw subsRes.error;
   126→    if (recentRes.error) throw recentRes.error;
   127→
   128→    const totalUsers = profilesRes.count ?? (profilesRes.data || []).length;
   129→    const subs = subsRes.data || [];
   130→    const proCount = subs.filter((s: { plan: string }) => s.plan === "pro").length;
   131→    const freeCount = totalUsers - proCount;
   132→    const now = new Date();
   133→    const activePro = subs.filter((s: { plan: string; expires_at: string | null }) => s.plan === "pro" && s.expires_at && new Date(s.expires_at) >= now).length;
   134→    const expiredPro = subs.filter((s: { plan: string; expires_at: string | null }) => s.plan === "pro" && s.expires_at && new Date(s.expires_at) < now).length;
   135→    const recentSignups = (recentRes.data || []).length;
   136→    const monthlyRevenue = proCount * 29000;
   137→
   138→    return json({
   139→      data: { totalUsers, proCount, freeCount, monthlyRevenue, recentSignups, activePro, expiredPro },
   140→    });
   141→  }
   142→
   143→  // GET /api/admin/users — List all users with profile + subscription info
   144→  if (path === "/admin/users" && req.method === "GET") {
   145→    const [profilesRes, subsRes] = await Promise.all([
   146→      sb.from("profiles").select("*").order("created_at", { ascending: false }),
   147→      sb.from("subscriptions").select("user_id, plan, started_at, expires_at"),
   148→    ]);
   149→
   150→    if (profilesRes.error) throw profilesRes.error;
   151→    if (subsRes.error) throw subsRes.error;
   152→
   153→    const profiles = profilesRes.data || [];
   154→    const subMap = new Map<string, { plan: string; started_at: string | null; expires_at: string | null }>();
   155→    for (const s of (subsRes.data || [])) {
   156→      subMap.set(s.user_id, s);
   157→    }
   158→
   159→    // Try to get emails from auth.users via admin API (list users)
   160→    let emailMap = new Map<string, string>();
   161→    try {
   162→      const authRes = await fetch(`${env.SUPABASE_URL}/auth/v1/admin/users?limit=1000`, {
   163→        headers: { Authorization: `Bearer ${env.SUPABASE_SERVICE_KEY}`, apikey: env.SUPABASE_ANON_KEY },
   164→      });
   165→      if (authRes.ok) {
   166→        const authData = await authRes.json() as { users: { id: string; email: string }[] };
   167→        for (const u of (authData.users || [])) {
   168→          emailMap.set(u.id, u.email);
   169→        }
   170→      }
   171→    } catch {
   172→      // If admin API not available, proceed without emails
   173→    }
   174→
   175→    const users = profiles.map((p: { id: string; user_id?: string; name?: string; created_at: string }) => {
   176→      const uid = p.user_id || p.id;
   177→      const sub = subMap.get(uid);
   178→      return {
   179→        user_id: uid,
   180→        id: p.id,
   181→        name: p.name || "—",
   182→        email: emailMap.get(uid) || "—",
   183→        plan: sub?.plan || "free",
   184→        created_at: p.created_at,
   185→      };
   186→    });
   187→
   188→    return json({ data: users });
   189→  }
   190→
   191→  // GET /api/admin/subscriptions/update — should be POST, reject GET
   192→  // POST /api/admin/subscription/update — Manually update a user's plan
   193→  if (path === "/admin/subscription/update" && req.method === "POST") {
   194→    const body = await getBody(req);
   195→    const userId = body.user_id as string;
   196→    const plan = body.plan as string;
   197→    const expiresAt = body.expires_at as string | undefined;
   198→
   199→    if (!userId || !plan) return json({ error: "user_id dan plan required" }, 400);
   200→    if (plan !== "free" && plan !== "pro") return json({ error: "Plan harus 'free' atau 'pro'" }, 400);
   201→
   202→    // Check if subscription exists
   203→    const { data: existing } = await sb.from("subscriptions").select("id").eq("user_id", userId).single();
   204→
   205→    let result;
   206→    if (existing) {
   207→      const updateData: Record<string, unknown> = { plan, started_at: new Date().toISOString() };
   208→      if (plan === "pro") {
   209→        updateData.expires_at = expiresAt || new Date(Date.now() + 30 * 86400000).toISOString();
   210→      } else {
   211→        updateData.expires_at = expiresAt || null;
   212→      }
   213→      result = await sb.from("subscriptions").update(updateData).eq("user_id", userId).select().single();
   214→    } else {
   215→      const insertData: Record<string, unknown> = { user_id: userId, plan };
   216→      if (plan === "pro") {
   217→        insertData.expires_at = expiresAt || new Date(Date.now() + 30 * 86400000).toISOString();
   218→      }
   219→      result = await sb.from("subscriptions").insert(insertData).select().single();
   220→    }
   221→
   222→    if (result.error) throw result.error;
   223→    return json({ data: result.data });
   224→  }
   225→
   226→  // GET /api/admin/subscriptions/update — method not allowed
   227→  if (path === "/admin/subscription/update" && req.method === "GET") {
   228→    return json({ error: "Method not allowed. Gunakan POST." }, 405);
   229→  }
   230→
   231→  // GET /api/admin/subscriptions — Alias for subscription/update GET (not needed, but for completeness)
   232→
   233→  // GET /api/admin/subscriptions — List all subscriptions
   234→  if (path === "/admin/subscriptions" || path === "/admin/subscriptions/") {
   235→    if (req.method !== "GET") return json({ error: "Method not allowed" }, 405);
   236→
   237→    const { data, error } = await sb.from("subscriptions").select("*").order("started_at", { ascending: false, nullsFirst: "last" });
   238→    if (error) throw error;
   239→
   240→    // Get emails
   241→    let emailMap = new Map<string, string>();
   242→    try {
   243→      const authRes = await fetch(`${env.SUPABASE_URL}/auth/v1/admin/users?limit=1000`, {
   244→        headers: { Authorization: `Bearer ${env.SUPABASE_SERVICE_KEY}`, apikey: env.SUPABASE_ANON_KEY },
   245→      });
   246→      if (authRes.ok) {
   247→        const authData = await authRes.json() as { users: { id: string; email: string }[] };
   248→        for (const u of (authData.users || [])) {
   249→          emailMap.set(u.id, u.email);
   250→        }
   251→      }
   252→    } catch {
   253→      // proceed without emails
   254→    }
   255→
   256→    const subs = (data || []).map((s: { user_id: string; [key: string]: unknown }) => ({
   257→      ...s,
   258→      email: emailMap.get(s.user_id) || "—",
   259→    }));
   260→
   261→    return json({ data: subs });
   262→  }
   263→
   264→  // GET /api/admin/transactions — Recent transactions across all users
   265→  if (path === "/admin/transactions" && req.method === "GET") {
   266→    const limit = parseInt(url.searchParams.get("limit") || "50", 10);
   267→    const offset = parseInt(url.searchParams.get("offset") || "0", 10);
   268→    const from = url.searchParams.get("from");
   269→    const to = url.searchParams.get("to");
   270→    const type = url.searchParams.get("type");
   271→
   272→    let query = sb
   273→      .from("transactions")
   274→      .select("*")
   275→      .order("occurred_at", { ascending: false })
   276→      .range(offset, offset + limit - 1);
   277→
   278→    if (from) query = query.gte("occurred_at", from);
   279→    if (to) query = query.lte("occurred_at", to);
   280→    if (type) query = query.eq("type", type);
   281→
   282→    const { data, error } = await query;
   283→    if (error) throw error;
   284→
   285→    // Get emails for the user_ids in the transactions
   286→    const userIds = [...new Set((data || []).map((t: { user_id: string }) => t.user_id))];
   287→    let emailMap = new Map<string, string>();
   288→    if (userIds.length > 0) {
   289→      try {
   290→        const authRes = await fetch(`${env.SUPABASE_URL}/auth/v1/admin/users?limit=1000`, {
   291→          headers: { Authorization: `Bearer ${env.SUPABASE_SERVICE_KEY}`, apikey: env.SUPABASE_ANON_KEY },
   292→        });
   293→        if (authRes.ok) {
   294→          const authData = await authRes.json() as { users: { id: string; email: string }[] };
   295→          for (const u of (authData.users || [])) {
   296→            emailMap.set(u.id, u.email);
   297→          }
   298→        }
   299→      } catch {
   300→        // proceed without emails
   301→      }
   302→    }
   303→
   304→    const txns = (data || []).map((t: { user_id: string; [key: string]: unknown }) => ({
   305→      ...t,
   306→      user_email: emailMap.get(t.user_id) || "—",
   307→    }));
   308→
   309→    return json({ data: txns });
   310→  }
   311→
   312→  return json({ error: "Not found" }, 404);
   313→}

   316→async function handleWallets(req: Request, env: Env): Promise<Response> {
   317→  const url = new URL(req.url);
   318→  const admin = getAdmin(env);
   319→
   320→  if (req.method === "GET") {
   321→    const userId = url.searchParams.get("user_id");
   322→    if (!userId) return json({ error: "user_id required" }, 400);
   323→    const { data, error } = await admin
   324→      .from("wallets")
   325→      .select("*")
   326→      .eq("user_id", userId)
   327→      .order("created_at", { ascending: true });
   328→    if (error) throw error;
   329→    return json({ data });
   330→  }
   331→
   332→  if (req.method === "POST") {
   333→    const body = await getBody(req);
   334→    const userId = body.user_id as string;
   335→
   336→    // Check plan limit (with KV cache)
   337→    const isPro = await isProUser(env, userId);
   338→    if (!isPro) {
   339→      const { count } = await admin
   340→        .from("wallets")
   341→        .select("*", { count: "exact", head: true })
   342→        .eq("user_id", userId);
   343→      if ((count || 0) >= 2) {
   344→        return json(
   345→          { error: "Limit dompet tercapai (2 untuk Gratis). Upgrade ke Pro untuk unlimited.", limit: "wallets" },
   346→          403
   347→        );
   348→      }
   349→    }
   350→
   351→    const { data, error } = await admin.from("wallets").insert(body).select().single();
   352→    if (error) throw error;
   353→    return json({ data }, 201);
   354→  }
   355→
   356→  if (req.method === "DELETE") {
   357→    const id = url.searchParams.get("id");
   358→    if (!id) return json({ error: "id required" }, 400);
   359→    const { error } = await admin.from("wallets").delete().eq("id", id);
   360→    if (error) throw error;
   361→    return json({ ok: true });
   362→  }
   363→
   364→  return json({ error: "Method not allowed" }, 405);
   365→}

   505→async function handleSummary(req: Request, env: Env): Promise<Response> {

   508→  if (!userId) return json({ error: "user_id required" }, 400);
   509→
   510→  const admin = getAdmin(env);

   517→  if (debtsRes.error) throw debtsRes.error;
   518→
   519→  const transactions = txnsRes.data || [];
   520→  const debts = debtsRes.data || [];
   521→
   522→  const totalIncome = transactions.filter((t: { type: string }) => t.type === "income").reduce((s: number, t: { amount: number }) => s + t.amount, 0);
   523→  const totalExpenses = transactions.filter((t: { type: string }) => t.type === "expense").reduce((s: number, t: { amount: number }) => s + t.amount, 0);
   524→  const activeBalance = totalIncome - totalExpenses;
   525→  const totalDebt = debts.filter((d: { type: string; status: string }) => d.type === "debt" && d.status !== "paid").reduce((s: number, d: { amount: number; paid_amount: number }) => s + (d.amount - d.paid_amount), 0);
   526→  const totalReceivable = debts.filter((d: { type: string; status: string }) => d.type === "receivable" && d.status !== "paid").reduce((s: number, d: { amount: number; paid_amount: number }) => s + (d.amount - d.paid_amount), 0);
   527→
   528→  const now = new Date();
   529→  const monthlyData = [];
   530→  for (let i = 11; i >= 0; i--) {
   531→    const mStart = new Date(now.getFullYear(), now.getMonth() - i, 1);
   532→    const mEnd = new Date(now.getFullYear(), now.getMonth() - i + 1, 1);
   533→    const mTxns = transactions.filter((t: { occurred_at: string }) => { const d = new Date(t.occurred_at); return d >= mStart && d < mEnd; });
   534→    const mIncome = mTxns.filter((t: { type: string }) => t.type === "income").reduce((s: number, t: { amount: number }) => s + t.amount, 0);
   535→    const mExpenses = mTxns.filter((t: { type: string }) => t.type === "expense").reduce((s: number, t: { amount: number }) => s + t.amount, 0);
   536→    monthlyData.push({ month: mStart.toLocaleDateString("en-US", { month: "short", year: "2-digit" }), income: mIncome, expenses: mExpenses, net: mIncome - mExpenses });
   537→  }
   538→
   539→  return json({ activeBalance, totalIncome, totalExpenses, totalDebt, totalReceivable, monthlyData });
   540→}

   609→async function handleSync(req: Request, env: Env): Promise<Response> {
   610→  const admin = getAdmin(env);
   611→
   612→  if (req.method === "POST") {
   613→    const body = await getBody(req);
   614→    const userId = body.user_id as string;
   615→    const syncData = body.data as Record<string, unknown>;
   616→    if (!userId || !syncData) return json({ error: "user_id dan data required" }, 400);
   617→
   618→    // Only Pro can sync (with KV cache)
   619→    const isPro = await isProUser(env, userId);
   620→    if (!isPro) return json({ error: "Sinkronisasi cloud cuma untuk paket Pro." }, 403);
   621→
   622→    const results: Record<string, unknown> = {};
   623→
   624→    if ((syncData.wallets as unknown[])?.length) {
   625→      const wallets = (syncData.wallets as Record<string, unknown>[]).map((w) => ({
   626→        id: w.id, user_id: userId, name: w.name, color: w.color || "#C9A962", initial_balance: w.initial || 0,
   627→      }));
   628→      const { data: r, error } = await admin.from("wallets").upsert(wallets, { onConflict: "id" });
   629→      if (error) throw error;
   630→      results.wallets = r;
   631→    }
   632→
   633→    if ((syncData.transactions as unknown[])?.length) {
   634→      const txns = (syncData.transactions as Record<string, unknown>[]).map((t) => ({
   635→        id: t.id, user_id: userId, wallet_id: t.wallet_id, type: t.type, category: t.category,
   636→        description: t.description, amount: t.amount, occurred_at: t.occurred_at,
   637→        source: t.source || "manual", ref_id: t.ref_id || null, recur: t.recur || "none", recur_parent: t.recur_parent || null,
   638→      }));
   639→      const { data: r, error } = await admin.from("transactions").upsert(txns, { onConflict: "id" });
   640→      if (error) throw error;
   641→      results.transactions = r;
   642→    }
   643→
   644→    if ((syncData.debts as unknown[])?.length) {
   645→      const debts = (syncData.debts as Record<string, unknown>[]).map((d) => ({
   646→        id: d.id, user_id: userId, name: d.name, type: d.type, amount: d.amount,
   647→        paid_amount: d.paid_amount || 0, due_date: d.due_date || null, note: d.note || "",
   648→        status: d.status || "unpaid", settled_at: d.settled_at || null,
   649→      }));
   650→      const { data: r, error } = await admin.from("debts").upsert(debts, { onConflict: "id" });
   651→      if (error) throw error;
   652→      results.debts = r;
   653→    }
   654→
   655→    return json({ ok: true, synced: Object.keys(results) });
   656→  }
   657→
   658→  // GET — pull all data
   659→  if (req.method === "GET") {
   660→    const url = new URL(req.url);
   661→    const userId = url.searchParams.get("user_id");
   662→    if (!userId) return json({ error: "user_id required" }, 400);
   663→
   664→    const [wallets, transactions, debts, budgets, customCats, subscription] = await Promise.all([
   665→      admin.from("wallets").select("*").eq("user_id", userId),
   666→      admin.from("transactions").select("*").eq("user_id", userId).order("occurred_at", { ascending: false }),
   667→      admin.from("debts").select("*").eq("user_id", userId).order("created_at", { ascending: false }),
   668→      admin.from("budgets").select("*").eq("user_id", userId),
   669→      admin.from("custom_categories").select("*").eq("user_id", userId),
   670→      admin.from("subscriptions").select("*").eq("user_id", userId).single(),
   671→    ]);
   672→
   673→    return json({
   674→      data: {
   675→        wallets: wallets.data || [],
   676→        transactions: transactions.data || [],
   677→        debts: debts.data || [],
   678→        budgets: (budgets.data || []).reduce((acc: Record<string, number>, b: { category: string; amount: number }) => { acc[b.category] = b.amount; return acc; }, {}),
   679→        customCats: (customCats.data || []).map((c: { name: string; type: string; color: string; keywords: string[] }) => ({ name: c.name, type: c.type, color: c.color, kw: c.keywords })),
   680→        subscription: subscription.data || { plan: "free" },
   681→      },
   682→    });
   683→  }
   684→
   685→  return json({ error: "Method not allowed" }, 405);
   686→}

   940→async function checkRateLimit(request: Request, env: Env, limit = 60): Promise<boolean> {
   941→  try {
   942→    const ip = getClientIp(request);
   943→    const key = `rl:${ip}`;
   944→    const current = parseInt(await env.ARUS_KV.get(key) || "0", 10);
   945→    if (current >= limit) return false; // Rate limited
   946→    await env.ARUS_KV.put(key, String(current + 1), { expirationTtl: 60 }); // 1 min window
   947→    return true;
   948→  } catch {
   949→    return true; // Graceful fallback if KV fails
   950→  }
   951→}

  1020→// ─── API: /api/storage — R2 File Storage ──────────────────────
  1021→// Pro users: backup JSON, PDF reports, invoice attachments
  1022→async function handleStorage(req: Request, env: Env): Promise<Response> {
  1023→  if (!env.ARUS_STORAGE) return json({ error: "Cloud storage belum diaktifkan. Hubungi admin untuk setup R2.", r2_required: true }, 503);
  1024→  const url = new URL(req.url);
  1025→  const userId = url.searchParams.get("user_id");
  1026→  const fileId = url.searchParams.get("id");
  1027→
  1028→  if (!userId) return json({ error: "user_id required" }, 400);
  1029→
  1030→  // Check Pro plan (with KV cache)
  1031→  const pro = await isProUser(env, userId);
  1032→  if (!pro) return json({ error: "Storage cloud cuma untuk paket Pro. Upgrade ke Pro untuk backup & export PDF." }, 403);
  1033→
  1034→  const prefix = `users/${userId}/`;
  1035→
  1036→  if (req.method === "GET") {
  1037→    // List files or get single file
  1038→    if (fileId) {
  1039→      const obj = await env.ARUS_STORAGE.get(prefix + fileId);
  1040→      if (!obj) return json({ error: "File tidak ditemukan" }, 404);
  1041→      return new Response(obj.body, {
  1042→        headers: {
  1043→          "Content-Type": obj.httpMetadata?.contentType || "application/octet-stream",
  1044→          "Content-Disposition": obj.httpMetadata?.contentDisposition || `inline; filename="${fileId}"`,
  1045→          "Cache-Control": "private, max-age=3600",
  1046→        },
  1047→      });
  1048→    }
  1049→
  1050→    // List all files for this user
  1051→    const listed = await env.ARUS_STORAGE.list({ prefix });
  1052→    const files = listed.objects.map((o) => ({
  1053→      key: o.key.replace(prefix, ""),
  1054→      size: o.size,
  1055→      uploaded: o.uploaded.toISOString(),
  1056→      type: o.httpMetadata?.contentType || "",
  1057→    }));
  1058→    return json({ data: files, total: files.length, limit: MAX_FILES_PER_USER });
  1059→  }
  1060→
  1061→  if (req.method === "PUT") {
  1062→    // Upload file with validation
  1063→    if (!fileId) return json({ error: "id (filename) required" }, 400);
  1064→
  1065→    const contentType = req.headers.get("Content-Type") || "application/json";
  1066→
  1067→    // Validate content type
  1068→    if (!ALLOWED_TYPES.has(contentType)) {
  1069→      return json({ error: `Tipe file "${contentType}" tidak diizinkan. Gunakan: JSON, PDF, CSV, TXT, PNG, JPEG, WEBP.` }, 400);
  1070→    }
  1071→
  1072→    // Validate file size
  1073→    const contentLength = parseInt(req.headers.get("Content-Length") || "0", 10);
  1074→    if (contentLength > MAX_FILE_SIZE) {
  1075→      return json({ error: `File terlalu besar (max ${MAX_FILE_SIZE / 1024 / 1024}MB).` }, 413);
  1076→    }
  1077→
  1078→    // Check file count quota
  1079→    const existing = await env.ARUS_STORAGE.list({ prefix });
  1080→    if (existing.objects.length >= MAX_FILES_PER_USER) {
  1081→      return json({ error: `Kuota file penuh (max ${MAX_FILES_PER_USER} file). Hapus file lama dulu.` }, 403);
  1082→    }
  1083→
  1084→    // Upload to R2
  1085→    await env.ARUS_STORAGE.put(prefix + fileId, req.body, {
  1086→      httpMetadata: {
  1087→        contentType,
  1088→        contentDisposition: `attachment; filename="${fileId}"`,
  1089→      },
  1090→      customMetadata: {
  1091→        userId,
  1092→        uploadedAt: new Date().toISOString(),
  1093→      },
  1094→    });
  1095→
  1096→    return json({ ok: true, key: fileId, type: contentType });
  1097→  }
  1098→
  1099→  if (req.method === "DELETE") {
  1100→    if (!fileId) return json({ error: "id (filename) required" }, 400);
  1101→    await env.ARUS_STORAGE.delete(prefix + fileId);
  1102→    return json({ ok: true, deleted: fileId });
  1103→  }
  1104→
  1105→  return json({ error: "Method not allowed" }, 405);
  1106→}

  1108→// ─── API: /api/storage/backup — Full Data Backup to R2 ──────────
  1109→async function handleBackup(req: Request, env: Env): Promise<Response> {
  1110→  if (!env.ARUS_STORAGE) return json({ error: "Cloud storage belum diaktifkan. Hubungi admin untuk setup R2.", r2_required: true }, 503);
  1111→  const url = new URL(req.url);

  1115→  const pro = await isProUser(env, userId);

  1122→      admin.from("wallets").select("*").eq("user_id", userId),

  1124→      admin.from("debts").select("*").eq("user_id", userId),

  1128→

  1130→      version: "3.1",
  1131→      exported_at: new Date().toISOString(),
  1132→      user_id: userId,
  1133→      wallets: wallets.data || [],
  1134→      transactions: transactions.data || [],
  1135→      debts: debts.data || [],
  1136→      budgets: budgets.data || [],

  1153→    });

  1155→    // Also cache the last backup timestamp in KV
  1156→    await env.ARUS_KV.put(`last-backup:${userId}`, new Date().toISOString());
  1157→
  1158→    return json({
  1159→      ok: true,
  1160→      key: backupKey,
  1161→      size: JSON.stringify(backupData).length,

  1163→        wallets: (wallets.data || []).length,
  1164→        transactions: (transactions.data || []).length,
  1165→        debts: (debts.data || []).length,

  1169→    });
  1170→  }

  1172→  if (req.method === "GET") {

  1174→    const lastBackup = await env.ARUS_KV.get(`last-backup:${userId}`);
  1175→    const listed = await env.ARUS_STORAGE.list({ prefix: `users/${userId}/backup_` });

  1177→      key: o.key.replace(`users/${userId}/`, ""),
  1178→      size: o.size,
  1179→      date: o.uploaded.toISOString(),
  1180→    }));
  1181→    return json({ last_backup: lastBackup, backups });
  1182→  }
  1183→
  1184→  return json({ error: "Method not allowed" }, 405);
  1185→}

  1187→// ─── API: /api/storage/restore — Restore from R2 Backup ─────────
  1188→async function handleRestore(req: Request, env: Env): Promise<Response> {
  1189→  if (!env.ARUS_STORAGE) return json({ error: "Cloud storage belum diaktifkan. Hubungi admin untuk setup R2.", r2_required: true }, 503);
  1190→  if (req.method !== "POST") return json({ error: "Method not allowed" }, 405);
  1191→
  1192→  const body = await getBody(req);
  1193→  const userId = body.user_id as string;
  1194→  const backupKey = body.key as string;
  1195→  if (!userId || !backupKey) return json({ error: "user_id dan key required" }, 400);
  1196→
  1197→  const pro = await isProUser(env, userId);
  1198→  if (!pro) return json({ error: "Restore cuma untuk paket Pro." }, 403);
  1199→
  1200→  // Get backup from R2
  1201→  const obj = await env.ARUS_STORAGE.get(`users/${userId}/${backupKey}`);
  1202→  if (!obj) return json({ error: "Backup tidak ditemukan" }, 404);
  1203→
  1204→  const backupData = await obj.json() as {
  1205→    wallets?: unknown[];
  1206→    transactions?: unknown[];
  1207→    debts?: unknown[];
  1208→    budgets?: unknown[];
  1209→    custom_categories?: unknown[];
  1210→  };
  1211→
  1212→  const admin = getAdmin(env);
  1213→  const results: Record<string, number> = {};
  1214→
  1215→  // Restore each table (upsert)
  1216→  if (backupData.wallets?.length) {
  1217→    const { error } = await admin.from("wallets").upsert(backupData.wallets, { onConflict: "id" });
  1218→    if (!error) results.wallets = backupData.wallets.length;
  1219→  }
  1220→  if (backupData.transactions?.length) {
  1221→    const { error } = await admin.from("transactions").upsert(backupData.transactions, { onConflict: "id" });
  1222→    if (!error) results.transactions = backupData.transactions.length;
  1223→  }
  1224→  if (backupData.debts?.length) {
  1225→    const { error } = await admin.from("debts").upsert(backupData.debts, { onConflict: "id" });
  1226→    if (!error) results.debts = backupData.debts.length;
  1227→  }
  1228→  if (backupData.budgets?.length) {
  1229→    const { error } = await admin.from("budgets").upsert(backupData.budgets, { onConflict: "id" });
  1230→    if (!error) results.budgets = backupData.budgets.length;
  1231→  }
  1232→  if (backupData.custom_categories?.length) {
  1233→    const { error } = await admin.from("custom_categories").upsert(backupData.custom_categories, { onConflict: "id" });
  1234→    if (!error) results.categories = backupData.custom_categories.length;
  1235→  }
  1236→
  1237→  // Invalidate subscription cache after restore
  1238→  await invalidateSubscriptionCache(env, userId);
  1239→
  1240→  return json({ ok: true, restored: results });
  1241→}

  1243→// ─── API: /api/cache — KV Feature Flags, Cache & Sessions ──────
  1244→async function handleCache(req: Request, env: Env): Promise<Response> {
  1245→  const url = new URL(req.url);
  1246→  const subPath = url.pathname.replace("/api/cache", "").replace("//", "/");
  1247→
  1248→  // ── Feature Flags ──
  1249→  if (subPath === "/flags" || subPath === "/flags/") {
  1250→    if (req.method === "GET") {
  1251→      // Get all feature flags (merged with defaults)
  1252→      const flags: Record<string, string> = { ...DEFAULT_FEATURE_FLAGS };
  1253→      try {
  1254→        const list = await env.ARUS_KV.list({ prefix: "ff:" });
  1255→        for (const k of list.keys) {
  1256→          const val = await env.ARUS_KV.get(k.name);
  1257→          if (val !== null) flags[k.name] = val;
  1258→        }
  1259→      } catch {}
  1260→      return json({ flags });
  1261→    }
  1262→
  1263→    if (req.method === "PUT") {
  1264→      // Admin: set a feature flag
  1265→      const body = await getBody(req);
  1266→      const flag = body.flag as string;
  1267→      const value = body.value as string;
  1268→      if (!flag || value === undefined) return json({ error: "flag dan value required" }, 400);
  1269→      await setFeatureFlag(env, flag, value);
  1270→      return json({ ok: true, flag: `ff:${flag}`, value });
  1271→    }
  1272→  }
  1273→
  1274→  // ── Generic KV Get/Put/Delete ──
  1275→  if (req.method === "GET") {
  1276→    const key = url.searchParams.get("key");
  1277→    if (key) {
  1278→      const value = await env.ARUS_KV.get(key);
  1279→      return json({ key, value });
  1280→    }
  1281→    // List all keys with prefix
  1282→    const prefix = url.searchParams.get("prefix") || "";
  1283→    const list = await env.ARUS_KV.list({ prefix });
  1284→    return json({ keys: list.keys.map((k) => ({ name: k.name, expiration: k.expiration })) });
  1285→  }
  1286→
  1287→  if (req.method === "PUT") {
  1288→    const body = await getBody(req);
  1289→    const key = body.key as string;
  1290→    const value = body.value as string;
  1291→    const ttl = body.ttl as number | undefined;
  1292→    if (!key || value === undefined) return json({ error: "key dan value required" }, 400);
  1293→    const opts = ttl ? { expirationTtl: ttl } : undefined;
  1294→    await env.ARUS_KV.put(key, value, opts);
  1295→    return json({ ok: true, key });
  1296→  }
  1297→
  1298→  if (req.method === "DELETE") {
  1299→    const key = url.searchParams.get("key");
  1300→    if (!key) return json({ error: "key required" }, 400);
  1301→    await env.ARUS_KV.delete(key);
  1302→    return json({ ok: true });
  1303→  }
  1304→
  1305→  return json({ error: "Method not allowed" }, 405);
  1306→}
