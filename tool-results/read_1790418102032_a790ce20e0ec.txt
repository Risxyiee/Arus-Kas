    47→// ─── Admin Auth Verification ──────────────────────────────────
    48→async function verifyAdmin(req: Request, env: Env): Promise<{ userId: string; email: string } | null> {
    49→  const authHeader = req.headers.get("Authorization");
    50→  if (!authHeader || !authHeader.startsWith("Bearer ")) return null;
    51→  const token = authHeader.replace("Bearer ", "");
    52→
    53→  // Verify token with Supabase
    54→  const res = await fetch(`${env.SUPABASE_URL}/auth/v1/user`, {
    55→    headers: { Authorization: `Bearer ${token}`, apikey: env.SUPABASE_ANON_KEY },
    56→  });
    57→  if (!res.ok) return null;
    58→  const user = await res.json() as { id: string; email: string };
    59→
    60→  // Check admin email
    61→  const adminEmail = env.ADMIN_EMAIL || "riskiakbarp123@gmail.com";
    62→  if (user.email !== adminEmail) return null;
    63→
    64→  return { userId: user.id, email: user.email };
    65→}

    68→async function handleAdmin(req: Request, env: Env, path: string): Promise<Response> {
    69→  // Verify admin access for all admin routes
    70→  const admin = await verifyAdmin(req, env);
    71→  if (!admin) return json({ error: "Akses ditolak. Halaman ini hanya untuk admin." }, 403);
    72→
    73→  const sb = getAdmin(env);
    74→  const url = new URL(req.url);
    75→
    76→  // GET /api/admin/stats — Dashboard overview stats
    77→  if (path === "/admin/stats" && req.method === "GET") {
    78→    const sevenDaysAgo = new Date(Date.now() - 7 * 86400000).toISOString();
    79→

    81→      sb.from("profiles").select("id, created_at", { count: "exact" }),

    83→      sb.from("profiles").select("id").gte("created_at", sevenDaysAgo),
    84→    ]);
    85→
    86→    if (profilesRes.error) throw profilesRes.error;
    87→    if (subsRes.error) throw subsRes.error;
    88→    if (recentRes.error) throw recentRes.error;
    89→
    90→    const totalUsers = profilesRes.count ?? (profilesRes.data || []).length;
    91→    const subs = subsRes.data || [];
    92→    const proCount = subs.filter((s: { plan: string }) => s.plan === "pro").length;
    93→    const freeCount = totalUsers - proCount;
    94→    const now = new Date();
    95→    const activePro = subs.filter((s: { plan: string; expires_at: string | null }) => s.plan === "pro" && s.expires_at && new Date(s.expires_at) >= now).length;
    96→    const expiredPro = subs.filter((s: { plan: string; expires_at: string | null }) => s.plan === "pro" && s.expires_at && new Date(s.expires_at) < now).length;
    97→    const recentSignups = (recentRes.data || []).length;
    98→    const monthlyRevenue = proCount * 29000;
    99→
   100→    return json({
   101→      data: { totalUsers, proCount, freeCount, monthlyRevenue, recentSignups, activePro, expiredPro },
   102→    });
   103→  }
   104→
   105→  // GET /api/admin/users — List all users with profile + subscription info
   106→  if (path === "/admin/users" && req.method === "GET") {
   107→    const [profilesRes, subsRes] = await Promise.all([
   108→      sb.from("profiles").select("*").order("created_at", { ascending: false }),
   109→      sb.from("subscriptions").select("user_id, plan, started_at, expires_at"),
   110→    ]);
   111→
   112→    if (profilesRes.error) throw profilesRes.error;
   113→    if (subsRes.error) throw subsRes.error;
   114→
   115→    const profiles = profilesRes.data || [];
   116→    const subMap = new Map<string, { plan: string; started_at: string | null; expires_at: string | null }>();
   117→    for (const s of (subsRes.data || [])) {
   118→      subMap.set(s.user_id, s);
   119→    }
   120→
   121→    // Try to get emails from auth.users via admin API (list users)
   122→    let emailMap = new Map<string, string>();
   123→    try {
   124→      const authRes = await fetch(`${env.SUPABASE_URL}/auth/v1/admin/users?limit=1000`, {
   125→        headers: { Authorization: `Bearer ${env.SUPABASE_SERVICE_KEY}`, apikey: env.SUPABASE_ANON_KEY },
   126→      });
   127→      if (authRes.ok) {
   128→        const authData = await authRes.json() as { users: { id: string; email: string }[] };
   129→        for (const u of (authData.users || [])) {
   130→          emailMap.set(u.id, u.email);
   131→        }
   132→      }
   133→    } catch {
   134→      // If admin API not available, proceed without emails
   135→    }
   136→
   137→    const users = profiles.map((p: { id: string; user_id?: string; name?: string; created_at: string }) => {
   138→      const uid = p.user_id || p.id;
   139→      const sub = subMap.get(uid);
   140→      return {
   141→        user_id: uid,
   142→        id: p.id,
   143→        name: p.name || "—",
   144→        email: emailMap.get(uid) || "—",
   145→        plan: sub?.plan || "free",
   146→        created_at: p.created_at,
   147→      };
   148→    });
   149→
   150→    return json({ data: users });
   151→  }
   152→
   153→  // GET /api/admin/subscriptions/update — should be POST, reject GET
   154→  // POST /api/admin/subscription/update — Manually update a user's plan
   155→  if (path === "/admin/subscription/update" && req.method === "POST") {
   156→    const body = await getBody(req);
   157→    const userId = body.user_id as string;
   158→    const plan = body.plan as string;
   159→    const expiresAt = body.expires_at as string | undefined;
   160→
   161→    if (!userId || !plan) return json({ error: "user_id dan plan required" }, 400);
   162→    if (plan !== "free" && plan !== "pro") return json({ error: "Plan harus 'free' atau 'pro'" }, 400);
   163→
   164→    // Check if subscription exists
   165→    const { data: existing } = await sb.from("subscriptions").select("id").eq("user_id", userId).single();
   166→
   167→    let result;
   168→    if (existing) {
   169→      const updateData: Record<string, unknown> = { plan, started_at: new Date().toISOString() };
   170→      if (plan === "pro") {
   171→        updateData.expires_at = expiresAt || new Date(Date.now() + 30 * 86400000).toISOString();
   172→      } else {
   173→        updateData.expires_at = expiresAt || null;
   174→      }
   175→      result = await sb.from("subscriptions").update(updateData).eq("user_id", userId).select().single();
   176→    } else {
   177→      const insertData: Record<string, unknown> = { user_id: userId, plan };
   178→      if (plan === "pro") {
   179→        insertData.expires_at = expiresAt || new Date(Date.now() + 30 * 86400000).toISOString();
   180→      }
   181→      result = await sb.from("subscriptions").insert(insertData).select().single();
   182→    }
   183→
   184→    if (result.error) throw result.error;
   185→    return json({ data: result.data });
   186→  }
   187→
   188→  // GET /api/admin/subscriptions/update — method not allowed
   189→  if (path === "/admin/subscription/update" && req.method === "GET") {
   190→    return json({ error: "Method not allowed. Gunakan POST." }, 405);
   191→  }
   192→
   193→  // GET /api/admin/subscriptions — Alias for subscription/update GET (not needed, but for completeness)
   194→
   195→  // GET /api/admin/subscriptions — List all subscriptions
   196→  if (path === "/admin/subscriptions" || path === "/admin/subscriptions/") {
   197→    if (req.method !== "GET") return json({ error: "Method not allowed" }, 405);
   198→
   199→    const { data, error } = await sb.from("subscriptions").select("*").order("started_at", { ascending: false, nullsFirst: "last" });
   200→    if (error) throw error;
   201→
   202→    // Get emails
   203→    let emailMap = new Map<string, string>();
   204→    try {
   205→      const authRes = await fetch(`${env.SUPABASE_URL}/auth/v1/admin/users?limit=1000`, {
   206→        headers: { Authorization: `Bearer ${env.SUPABASE_SERVICE_KEY}`, apikey: env.SUPABASE_ANON_KEY },

   209→        const authData = await authRes.json() as { users: { id: string; email: string }[] };
   210→        for (const u of (authData.users || [])) {
   211→          emailMap.set(u.id, u.email);
   212→        }
   213→      }
   214→    } catch {
   215→      // proceed without emails
   216→    }
   217→
   218→    const subs = (data || []).map((s: { user_id: string; [key: string]: unknown }) => ({
   219→      ...s,
   220→      email: emailMap.get(s.user_id) || "—",
   221→    }));
   222→
   223→    return json({ data: subs });
   224→  }
   225→
   226→  // GET /api/admin/transactions — Recent transactions across all users
   227→  if (path === "/admin/transactions" && req.method === "GET") {
   228→    const limit = parseInt(url.searchParams.get("limit") || "50", 10);
   229→    const offset = parseInt(url.searchParams.get("offset") || "0", 10);
   230→    const from = url.searchParams.get("from");
   231→    const to = url.searchParams.get("to");
   232→    const type = url.searchParams.get("type");
   233→
   234→    let query = sb
   235→      .from("transactions")
   236→      .select("*")
   237→      .order("occurred_at", { ascending: false })
   238→      .range(offset, offset + limit - 1);
   239→
   240→    if (from) query = query.gte("occurred_at", from);
   241→    if (to) query = query.lte("occurred_at", to);
   242→    if (type) query = query.eq("type", type);
   243→
   244→    const { data, error } = await query;
   245→    if (error) throw error;
   246→
   247→    // Get emails for the user_ids in the transactions
   248→    const userIds = [...new Set((data || []).map((t: { user_id: string }) => t.user_id))];
   249→    let emailMap = new Map<string, string>();
   250→    if (userIds.length > 0) {
   251→      try {
   252→        const authRes = await fetch(`${env.SUPABASE_URL}/auth/v1/admin/users?limit=1000`, {
   253→          headers: { Authorization: `Bearer ${env.SUPABASE_SERVICE_KEY}`, apikey: env.SUPABASE_ANON_KEY },

   256→          const authData = await authRes.json() as { users: { id: string; email: string }[] };
   257→          for (const u of (authData.users || [])) {
   258→            emailMap.set(u.id, u.email);
   259→          }
   260→        }
   261→      } catch {
   262→        // proceed without emails
   263→      }
   264→    }
   265→
   266→    const txns = (data || []).map((t: { user_id: string; [key: string]: unknown }) => ({
   267→      ...t,
   268→      user_email: emailMap.get(t.user_id) || "—",
   269→    }));
   270→
   271→    return json({ data: txns });
   272→  }
   273→
   274→  return json({ error: "Not found" }, 404);
   275→}

   278→async function handleWallets(req: Request, env: Env): Promise<Response> {

   286→      .from("wallets")
   287→      .select("*")
   288→      .eq("user_id", userId)
   289→      .order("created_at", { ascending: true });
   290→    if (error) throw error;
   291→    return json({ data });
   292→  }
   293→
   294→  if (req.method === "POST") {
   295→    const body = await getBody(req);
   296→    const userId = body.user_id as string;
   297→
   298→    // Check plan limit
   299→    const { data: sub } = await admin
   300→      .from("subscriptions")
   301→      .select("plan")
   302→      .eq("user_id", userId)
   303→      .single();
   304→
   305→    if (sub?.plan !== "pro") {
   306→      const { count } = await admin
   307→        .from("wallets")
   308→        .select("*", { count: "exact", head: true })
   309→        .eq("user_id", userId);
   310→      if ((count || 0) >= 2) {
   311→        return json(
   312→          { error: "Limit dompet tercapai (2 untuk Gratis). Upgrade ke Pro untuk unlimited.", limit: "wallets" },
   313→          403
   314→        );
   315→      }
   316→    }
   317→
   318→    const { data, error } = await admin.from("wallets").insert(body).select().single();
   319→    if (error) throw error;
   320→    return json({ data }, 201);
   321→  }
   322→
   323→  if (req.method === "DELETE") {
   324→    const id = url.searchParams.get("id");
   325→    if (!id) return json({ error: "id required" }, 400);
   326→    const { error } = await admin.from("wallets").delete().eq("id", id);
   327→    if (error) throw error;
   328→    return json({ ok: true });
   329→  }
   330→
   331→  return json({ error: "Method not allowed" }, 405);
   332→}

   471→// ─── API: /api/summary ──────────────────────────────────────────
   472→async function handleSummary(req: Request, env: Env): Promise<Response> {
   473→  const url = new URL(req.url);
   474→  const userId = url.searchParams.get("user_id");
   475→  if (!userId) return json({ error: "user_id required" }, 400);
   476→
   477→  const admin = getAdmin(env);
   478→  const [txnsRes, debtsRes] = await Promise.all([
   479→    admin.from("transactions").select("*").eq("user_id", userId),
   480→    admin.from("debts").select("*").eq("user_id", userId),
   481→  ]);
   482→
   483→  if (txnsRes.error) throw txnsRes.error;
   484→  if (debtsRes.error) throw debtsRes.error;
   485→
   486→  const transactions = txnsRes.data || [];
   487→  const debts = debtsRes.data || [];
   488→
   489→  const totalIncome = transactions.filter((t: { type: string }) => t.type === "income").reduce((s: number, t: { amount: number }) => s + t.amount, 0);
   490→  const totalExpenses = transactions.filter((t: { type: string }) => t.type === "expense").reduce((s: number, t: { amount: number }) => s + t.amount, 0);
   491→  const activeBalance = totalIncome - totalExpenses;
   492→  const totalDebt = debts.filter((d: { type: string; status: string }) => d.type === "debt" && d.status !== "paid").reduce((s: number, d: { amount: number; paid_amount: number }) => s + (d.amount - d.paid_amount), 0);
   493→  const totalReceivable = debts.filter((d: { type: string; status: string }) => d.type === "receivable" && d.status !== "paid").reduce((s: number, d: { amount: number; paid_amount: number }) => s + (d.amount - d.paid_amount), 0);
   494→
   495→  const now = new Date();
   496→  const monthlyData = [];
   497→  for (let i = 11; i >= 0; i--) {
   498→    const mStart = new Date(now.getFullYear(), now.getMonth() - i, 1);
   499→    const mEnd = new Date(now.getFullYear(), now.getMonth() - i + 1, 1);
   500→    const mTxns = transactions.filter((t: { occurred_at: string }) => { const d = new Date(t.occurred_at); return d >= mStart && d < mEnd; });
   501→    const mIncome = mTxns.filter((t: { type: string }) => t.type === "income").reduce((s: number, t: { amount: number }) => s + t.amount, 0);
   502→    const mExpenses = mTxns.filter((t: { type: string }) => t.type === "expense").reduce((s: number, t: { amount: number }) => s + t.amount, 0);
   503→    monthlyData.push({ month: mStart.toLocaleDateString("en-US", { month: "short", year: "2-digit" }), income: mIncome, expenses: mExpenses, net: mIncome - mExpenses });
   504→  }
   505→
   506→  return json({ activeBalance, totalIncome, totalExpenses, totalDebt, totalReceivable, monthlyData });

   593→
   594→    if ((syncData.transactions as unknown[])?.length) {
   595→      const txns = (syncData.transactions as Record<string, unknown>[]).map((t) => ({
   596→        id: t.id, user_id: userId, wallet_id: t.wallet_id, type: t.type, category: t.category,

   602→      results.transactions = r;

   604→
   605→    if ((syncData.debts as unknown[])?.length) {
   606→      const debts = (syncData.debts as Record<string, unknown>[]).map((d) => ({

   636→        wallets: wallets.data || [],
   637→        transactions: transactions.data || [],

   640→        customCats: (customCats.data || []).map((c: { name: string; type: string; color: string; keywords: string[] }) => ({ name: c.name, type: c.type, color: c.color, kw: c.keywords })),
   641→        subscription: subscription.data || { plan: "free" },
   642→      },

   644→  }

   891→async function checkRateLimit(request: Request, env: Env, limit = 60): Promise<boolean> {
   892→  if (!env.ARUS_KV) return true; // KV not bound — skip rate limiting
   893→  try {
   894→    const ip = getClientIp(request);
   895→    const key = `rl:${ip}`;
   896→
   897→    const current = parseInt(await env.ARUS_KV.get(key) || "0", 10);
   898→    if (current >= limit) return false; // Rate limited
   899→
   900→    await env.ARUS_KV.put(key, String(current + 1), { expirationTtl: 60 }); // 1 min window
   901→    return true;
   902→  } catch {
   903→    return true;
   904→  }
   905→}

   909→async function handleStorage(req: Request, env: Env): Promise<Response> {
   910→  if (!env.ARUS_STORAGE) return json({ error: "Storage belum di-setup. Aktifkan R2 binding." }, 503);
   911→  const url = new URL(req.url);
   912→  const userId = url.searchParams.get("user_id");
   913→  const fileId = url.searchParams.get("id");
   914→
   915→  if (!userId) return json({ error: "user_id required" }, 400);
   916→
   917→  // Check Pro plan
   918→  const admin = getAdmin(env);
   919→  const { data: sub } = await admin.from("subscriptions").select("plan").eq("user_id", userId).single();
   920→  if (sub?.plan !== "pro") return json({ error: "Storage cloud cuma untuk paket Pro." }, 403);
   921→

   923→
   924→  if (req.method === "GET") {
   925→    // List files or get single file
   926→    if (fileId) {
   927→      const obj = await env.ARUS_STORAGE.get(prefix + fileId);
   928→      if (!obj) return json({ error: "File tidak ditemukan" }, 404);
   929→      return new Response(obj.body, {
   930→        headers: {
   931→          "Content-Type": obj.httpMetadata?.contentType || "application/octet-stream",
   932→          "Content-Disposition": obj.httpMetadata?.contentDisposition || "",
   933→        },
   934→      });
   935→    }
   936→
   937→    // List all files for this user
   938→    const listed = await env.ARUS_STORAGE.list({ prefix });
   939→    const files = listed.objects.map((o) => ({
   940→      key: o.key.replace(prefix, ""),
   941→      size: o.size,
   942→      uploaded: o.uploaded.toISOString(),
   943→      type: o.httpMetadata?.contentType || "",
   944→    }));
   945→    return json({ data: files });
   946→  }
   947→
   948→  if (req.method === "PUT") {
   949→    // Upload file
   950→    if (!fileId) return json({ error: "id (filename) required" }, 400);
   951→    const contentType = req.headers.get("Content-Type") || "application/json";
   952→    await env.ARUS_STORAGE.put(prefix + fileId, req.body, {
   953→      httpMetadata: { contentType },
   954→    });
   955→    return json({ ok: true, key: fileId });
   956→  }
   957→
   958→  if (req.method === "DELETE") {
   959→    if (!fileId) return json({ error: "id (filename) required" }, 400);
   960→    await env.ARUS_STORAGE.delete(prefix + fileId);
   961→    return json({ ok: true });
   962→  }
   963→
   964→  return json({ error: "Method not allowed" }, 405);
   965→}

   968→async function handleCache(req: Request, env: Env): Promise<Response> {
   969→  if (!env.ARUS_KV) return json({ error: "Cache belum di-setup. Aktifkan KV binding." }, 503);
   970→  const url = new URL(req.url);
   971→
   972→  if (req.method === "GET") {
   973→    // Get a cached value or feature flag
   974→    const key = url.searchParams.get("key");
   975→    if (key) {
   976→      const value = await env.ARUS_KV.get(key);
   977→      return json({ key, value });
   978→    }
   979→    // List all keys with prefix
   980→    const prefix = url.searchParams.get("prefix") || "";
   981→    const list = await env.ARUS_KV.list({ prefix });
   982→    return json({ keys: list.keys.map((k) => ({ name: k.name, expiration: k.expiration })) });
   983→  }
   984→
   985→  if (req.method === "PUT") {
   986→    const body = await getBody(req);
   987→    const key = body.key as string;
   988→    const value = body.value as string;
   989→    const ttl = body.ttl as number | undefined;
   990→    if (!key || value === undefined) return json({ error: "key dan value required" }, 400);
   991→    const opts = ttl ? { expirationTtl: ttl } : undefined;
   992→    await env.ARUS_KV.put(key, value, opts);
   993→    return json({ ok: true, key });
   994→  }
   995→
   996→  if (req.method === "DELETE") {
   997→    const key = url.searchParams.get("key");
   998→    if (!key) return json({ error: "key required" }, 400);
   999→    await env.ARUS_KV.delete(key);
  1000→    return json({ ok: true });
  1001→  }
  1002→
  1003→  return json({ error: "Method not allowed" }, 405);
  1004→}
