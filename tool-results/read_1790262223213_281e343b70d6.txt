    28→    28→function parseAmt(s: string): number {
    29→    29→  const m = s.toLowerCase().replace(/\s/g, '').match(/(\d+(?:[.,]\d{3})*(?:[.,]\d{1,2})?)(rb|k|jt|juta)?/);
    30→    30→  if (!m) return 0;
    31→    31→  let n = parseFloat(m[1].replace(/\./g, '').replace(',', '.'));
    32→    32→  if (m[2] === 'rb' || m[2] === 'k') n *= 1e3;
    33→    33→  if (m[2] === 'jt' || m[2] === 'juta') n *= 1e6;
    34→    34→  return Math.round(n);
    35→    35→}
