    47→    47→// ─── Admin Auth Verification ──────────────────────────────────
    48→    48→async function verifyAdmin(req: Request, env: Env): Promise<{ userId: string; email: string } | null> {
    49→    49→  const authHeader = req.headers.get("Authorization");
    50→    50→  if (!authHeader || !authHeader.startsWith("Bearer ")) return null;
    51→    51→  const token = authHeader.replace("Bearer ", "");
    52→    52→
    53→    53→  // Verify token with Supabase
    54→    54→  const res = await fetch(`${env.SUPABASE_URL}/auth/v1/user`, {
    55→    55→    headers: { Authorization: `Bearer ${token}`, apikey: env.SUPABASE_ANON_KEY },
    56→    56→  });
    57→    57→  if (!res.ok) return null;
    58→    58→  const user = await res.json() as { id: string; email: string };
    59→    59→
    60→    60→  // Check admin email
    61→    61→  const adminEmail = env.ADMIN_EMAIL || "riskiakbarp123@gmail.com";
    62→    62→  if (user.email !== adminEmail) return null;
    63→    63→
    64→    64→  return { userId: user.id, email: user.email };
    65→    65→}

   278→   278→async function handleWallets(req: Request, env: Env): Promise<Response> {

   287→   287→      .select("*")
   288→   288→      .eq("user_id", userId)
   289→   289→      .order("created_at", { ascending: true });
   290→   290→    if (error) throw error;
   291→   291→    return json({ data });
   292→   292→  }
   293→   293→
   294→   294→  if (req.method === "POST") {
   295→   295→    const body = await getBody(req);
   296→   296→    const userId = body.user_id as string;
   297→   297→
   298→   298→    // Check plan limit
   299→   299→    const { data: sub } = await admin
   300→   300→      .from("subscriptions")
   301→   301→      .select("plan")
   302→   302→      .eq("user_id", userId)
   303→   303→      .single();
   304→   304→
   305→   305→    if (sub?.plan !== "pro") {
   306→   306→      const { count } = await admin
   307→   307→        .from("wallets")
   308→   308→        .select("*", { count: "exact", head: true })
   309→   309→        .eq("user_id", userId);
   310→   310→      if ((count || 0) >= 2) {
   311→   311→        return json(
   312→   312→          { error: "Limit dompet tercapai (2 untuk Gratis). Upgrade ke Pro untuk unlimited.", limit: "wallets" },

   317→   317→
   318→   318→    const { data, error } = await admin.from("wallets").insert(body).select().single();
   319→   319→    if (error) throw error;
   320→   320→    return json({ data }, 201);
   321→   321→  }
   322→   322→
   323→   323→  if (req.method === "DELETE") {
   324→   324→    const id = url.searchParams.get("id");
   325→   325→    if (!id) return json({ error: "id required" }, 400);
   326→   326→    const { error } = await admin.from("wallets").delete().eq("id", id);
   327→   327→    if (error) throw error;
   328→   328→    return json({ ok: true });
   329→   329→  }
   330→   330→
   331→   331→  return json({ error: "Method not allowed" }, 405);
   332→   332→}

   849→   849→async function handleCron(env: Env): Promise<void> {
   850→   850→  const admin = getAdmin(env);
   851→   851→  const now = new Date().toISOString();
   852→   852→
   853→   853→  // Find all Pro subscriptions that have expired
   854→   854→  const { data: expired, error } = await admin
   855→   855→    .from("subscriptions")
   856→   856→    .select("user_id, expires_at")
   857→   857→    .eq("plan", "pro")
   858→   858→    .lt("expires_at", now);
   859→   859→
   860→   860→  if (error) {
   861→   861→    console.error("Cron: failed to query expired subscriptions:", error);
   862→   862→    return;
   863→   863→  }
   864→   864→
   865→   865→  if (!expired || expired.length === 0) {
   866→   866→    console.log(`Cron: no expired subscriptions at ${now}`);
   867→   867→    return;
   868→   868→  }
   869→   869→
   870→   870→  // Downgrade expired Pro users to Free
   871→   871→  const userIds = expired.map((s: { user_id: string }) => s.user_id);
   872→   872→  const { error: updateError } = await admin
   873→   873→    .from("subscriptions")
   874→   874→    .update({ plan: "free" })
   875→   875→    .in("user_id", userIds);
   876→   876→
   877→   877→  if (updateError) {
   878→   878→    console.error("Cron: failed to downgrade:", updateError);
   879→   879→  } else {
   880→   880→    console.log(`Cron: downgraded ${userIds.length} expired Pro → Free`);
   881→   881→  }
   882→   882→}

   891→   891→async function checkRateLimit(request: Request, env: Env, limit = 60): Promise<boolean> {
   892→   892→  if (!env.ARUS_KV) return true; // KV not bound — skip rate limiting
   893→   893→  try {
   894→   894→    const ip = getClientIp(request);
   895→   895→    const key = `rl:${ip}`;
   896→   896→
   897→   897→    const current = parseInt(await env.ARUS_KV.get(key) || "0", 10);
   898→   898→    if (current >= limit) return false; // Rate limited
   899→   899→
   900→   900→    await env.ARUS_KV.put(key, String(current + 1), { expirationTtl: 60 }); // 1 min window
   901→   901→    return true;
   902→   902→  } catch {
   903→   903→    return true;
   904→   904→  }
   905→   905→}

   908→   908→// Pro users: backup JSON, PDF reports, invoice attachments
   909→   909→async function handleStorage(req: Request, env: Env): Promise<Response> {
   910→   910→  if (!env.ARUS_STORAGE) return json({ error: "Storage belum di-setup. Aktifkan R2 binding." }, 503);
   911→   911→  const url = new URL(req.url);
   912→   912→  const userId = url.searchParams.get("user_id");
   913→   913→  const fileId = url.searchParams.get("id");
   914→   914→
   915→   915→  if (!userId) return json({ error: "user_id required" }, 400);
   916→   916→
   917→   917→  // Check Pro plan
   918→   918→  const admin = getAdmin(env);
   919→   919→  const { data: sub } = await admin.from("subscriptions").select("plan").eq("user_id", userId).single();
   920→   920→  if (sub?.plan !== "pro") return json({ error: "Storage cloud cuma untuk paket Pro." }, 403);
   921→   921→
   922→   922→  const prefix = `users/${userId}/`;
   923→   923→
   924→   924→  if (req.method === "GET") {
   925→   925→    // List files or get single file
   926→   926→    if (fileId) {
   927→   927→      const obj = await env.ARUS_STORAGE.get(prefix + fileId);
   928→   928→      if (!obj) return json({ error: "File tidak ditemukan" }, 404);
   929→   929→      return new Response(obj.body, {
   930→   930→        headers: {
   931→   931→          "Content-Type": obj.httpMetadata?.contentType || "application/octet-stream",
   932→   932→          "Content-Disposition": obj.httpMetadata?.contentDisposition || "",
   933→   933→        },
   934→   934→      });
   935→   935→    }
   936→   936→
   937→   937→    // List all files for this user
   938→   938→    const listed = await env.ARUS_STORAGE.list({ prefix });
   939→   939→    const files = listed.objects.map((o) => ({
   940→   940→      key: o.key.replace(prefix, ""),
   941→   941→      size: o.size,
   942→   942→      uploaded: o.uploaded.toISOString(),
   943→   943→      type: o.httpMetadata?.contentType || "",
   944→   944→    }));
   945→   945→    return json({ data: files });
   946→   946→  }
   947→   947→
   948→   948→  if (req.method === "PUT") {
   949→   949→    // Upload file
   950→   950→    if (!fileId) return json({ error: "id (filename) required" }, 400);
   951→   951→    const contentType = req.headers.get("Content-Type") || "application/json";
   952→   952→    await env.ARUS_STORAGE.put(prefix + fileId, req.body, {
   953→   953→      httpMetadata: { contentType },
   954→   954→    });
   955→   955→    return json({ ok: true, key: fileId });
   956→   956→  }
   957→   957→
   958→   958→  if (req.method === "DELETE") {
   959→   959→    if (!fileId) return json({ error: "id (filename) required" }, 400);
   960→   960→    await env.ARUS_STORAGE.delete(prefix + fileId);
   961→   961→    return json({ ok: true });
   962→   962→  }
   963→   963→
   964→   964→  return json({ error: "Method not allowed" }, 405);
   965→   965→}

   968→   968→async function handleCache(req: Request, env: Env): Promise<Response> {
   969→   969→  if (!env.ARUS_KV) return json({ error: "Cache belum di-setup. Aktifkan KV binding." }, 503);
   970→   970→  const url = new URL(req.url);
   971→   971→
   972→   972→  if (req.method === "GET") {
   973→   973→    // Get a cached value or feature flag
   974→   974→    const key = url.searchParams.get("key");
   975→   975→    if (key) {
   976→   976→      const value = await env.ARUS_KV.get(key);
   977→   977→      return json({ key, value });
   978→   978→    }
   979→   979→    // List all keys with prefix
   980→   980→    const prefix = url.searchParams.get("prefix") || "";
   981→   981→    const list = await env.ARUS_KV.list({ prefix });
   982→   982→    return json({ keys: list.keys.map((k) => ({ name: k.name, expiration: k.expiration })) });
   983→   983→  }
   984→   984→
   985→   985→  if (req.method === "PUT") {
   986→   986→    const body = await getBody(req);
   987→   987→    const key = body.key as string;
   988→   988→    const value = body.value as string;
   989→   989→    const ttl = body.ttl as number | undefined;
   990→   990→    if (!key || value === undefined) return json({ error: "key dan value required" }, 400);
   991→   991→    const opts = ttl ? { expirationTtl: ttl } : undefined;
   992→   992→    await env.ARUS_KV.put(key, value, opts);
   993→   993→    return json({ ok: true, key });
   994→   994→  }
   995→   995→
   996→   996→  if (req.method === "DELETE") {
   997→   997→    const key = url.searchParams.get("key");
   998→   998→    if (!key) return json({ error: "key required" }, 400);
   999→   999→    await env.ARUS_KV.delete(key);
  1000→  1000→    return json({ ok: true });
  1001→  1001→  }
  1002→  1002→
  1003→  1003→  return json({ error: "Method not allowed" }, 405);
  1004→  1004→}
