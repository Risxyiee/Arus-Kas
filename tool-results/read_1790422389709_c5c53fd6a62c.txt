   937→function drawChart(){
   938→ const cv=$('#cash-chart');if(!cv||!cv.clientWidth)return;
   939→ const S=monthSeries(6);
   940→ const cs=getComputedStyle(document.documentElement);
   941→ const pos=cs.getPropertyValue('--pos').trim(),neg=cs.getPropertyValue('--neg').trim(),
   942→       ink=cs.getPropertyValue('--ink').trim(),mut=cs.getPropertyValue('--mut').trim(),
   943→       line=cs.getPropertyValue('--line').trim();
   944→ const dpr=window.devicePixelRatio||1,W=cv.clientWidth,H=cv.clientHeight;
   945→ cv.width=W*dpr;cv.height=H*dpr;
   946→ const ctx=cv.getContext('2d');ctx.scale(dpr,dpr);ctx.clearRect(0,0,W,H);
   947→ const padL=58,padR=8,padT=12,padB=26,pw=W-padL-padR,ph=H-padT-padB;
   948→ const max=Math.max(1,Math.max.apply(null,S.map(s=>Math.max(s.inc,s.exp))))*1.15;
   949→ const y=v=>padT+ph-(v/max)*ph;
   950→ ctx.font='10px ui-monospace,Consolas,monospace';ctx.fillStyle=mut;ctx.textAlign='right';ctx.textBaseline='middle';
   951→ for(let i=0;i<=4;i++){const v=max*i/4,yy=y(v);
   952→  ctx.strokeStyle=line;ctx.lineWidth=1;ctx.beginPath();ctx.moveTo(padL,yy);ctx.lineTo(W-padR,yy);ctx.stroke();
   953→  ctx.fillText(shortMoney(v),padL-8,yy);}
   954→ const gw=pw/S.length,bw=Math.min(16,gw*0.26);
   955→ function bar(x,y0,h,color){h=Math.max(h,.5);ctx.fillStyle=color;ctx.beginPath();
   956→  if(ctx.roundRect)ctx.roundRect(x,y0,bw,h,[3,3,0,0]);else ctx.rect(x,y0,bw,h);ctx.fill();}

   958→ S.forEach((s,i)=>{const cx=padL+gw*i+gw/2;
   959→  bar(cx-bw-3,y(s.inc),padT+ph-y(s.inc),pos);
   960→  bar(cx+3,y(s.exp),padT+ph-y(s.exp),neg);
   961→  ctx.fillStyle=mut;ctx.fillText(s.label,cx,H-8);});
   962→ ctx.strokeStyle=ink;ctx.setLineDash([5,4]);ctx.lineWidth=1.5;ctx.beginPath();
   963→ S.forEach((s,i)=>{const cx=padL+gw*i+gw/2,yy=y(Math.max(0,s.net));if(i)ctx.lineTo(cx,yy);else ctx.moveTo(cx,yy);});
   964→ ctx.stroke();ctx.setLineDash([]);
