   518→async function api(path, opts = {}) {
   519→  const maxRetries = (opts._retries || 0);
   520→  const retryLimit = 2; // max 2 retries (3 total attempts)
   521→  try {
   522→    const res = await fetch(API + '/api' + path, {headers: apiHeaders(), ...opts});
   523→    const text = await res.text();
   524→    try { return JSON.parse(text); } catch {
   525→      return {error: 'Server error: respons bukan JSON. Pastikan API sudah berjalan.'};
   526→    }
   527→  } catch(e) {
   528→    // Network error — retry with exponential backoff
   529→    if (maxRetries < retryLimit) {
   530→      const delay = Math.pow(2, maxRetries) * 1000; // 1s, 2s, 4s
   531→      await new Promise(r => setTimeout(r, delay));
   532→      return api(path, {...opts, _retries: maxRetries + 1});
   533→    }
   534→    return {error: e.message || 'Koneksi gagal. Periksa internet Anda.'};
   535→  }
   536→}
