    67→// ─── SHA-512 helper (for Midtrans signature verification) ────────
    68→async function sha512(message: string): Promise<string> {
    69→  const data = new TextEncoder().encode(message);
    70→  const hash = await crypto.subtle.digest("SHA-512", data);
    71→  return Array.from(new Uint8Array(hash)).map(b => b.toString(16).padStart(2, "0")).join("");
    72→}

    74→// ─── Admin Auth Verification (with KV session cache) ───────────
    75→async function verifyAdmin(req: Request, env: Env): Promise<{ userId: string; email: string } | null> {
    76→  const authHeader = req.headers.get("Authorization");
    77→  if (!authHeader || !authHeader.startsWith("Bearer ")) return null;
    78→  const token = authHeader.replace("Bearer ", "");
    79→
    80→  // Check KV session cache first
    81→  const cached = await getCachedSession(env, token);
    82→  if (cached) {
    83→    const adminEmail = env.ADMIN_EMAIL || "riskiakbarp123@gmail.com";
    84→    if (cached.email !== adminEmail) return null;
    85→    return cached;
    86→  }
    87→
    88→  // Verify token+cache session
    89→  const res = await fetch(`${env.SUPABASE_URL}/auth/v1/user`, {
    90→    headers: { Authorization: `Bearer ${token}`, apikey: env.SUPABASE_ANON_KEY },
    91→  });
    92→  if (!res.ok) return null;
    93→  const user = await res.json() as { id: string; email: string };
    94→
    95→  const adminEmail = env.ADMIN_EMAIL || "riskiakbarp123@gmail.com";
    96→  if (user.email !== adminEmail) return null;
    97→
    98→  // Cache the session
    99→  const sessionData = { userId: user.id, email: user.email };
   100→  await cacheSession(env, token, sessionData);
   101→
   102→  return sessionData;
   103→}

   420→}

   423→async function handleWallets(req: Request, env: Env): Promise<Response> {
   424→  const url = new URL(req.url);
   425→  const admin = getAdmin(env);
   426→
   427→  if (req.method === "GET") {
   428→    const userId = url.searchParams.get("user_id");
   429→    if (!userId) return json({ error: "user_id required" }, 400);
   430→    const { data, error } = await admin
   431→      .from("wallets")
   432→      .select("*")
   433→      .eq("user_id", userId)
   434→      .order("created_at", { ascending: true });
   435→    if (error) throw error;
   436→    return json({ data });
   437→  }
   438→
   439→  if (req.method === "POST") {
   440→    const body = await getBody(req);
   441→    const userId = body.user_id as string;
   442→
   443→    // Check plan limit (with KV cache)
   444→    const isPro = await isProUser(env, userId);
   445→    if (!isPro) {
   446→      const { count } = await admin
   447→        .from("wallets")
   448→        .select("*", { count: "exact", head: true })
   449→        .eq("user_id", userId);
   450→      if ((count || 0) >= 2) {
   451→        return json(
   452→          { error: "Limit dompet tercapai (2 untuk Gratis). Upgrade ke Pro untuk unlimited.", limit: "wallets" },
   453→          403
   454→        );
   455→      }
   456→    }
   457→
   458→    const { data, error } = await admin.from("wallets").insert(body).select().single();
   459→    if (error) throw error;
   460→    return json({ data }, 201);
   461→  }
   462→
   463→  if (req.method === "DELETE") {
   464→    const id = url.searchParams.get("id");
   465→    if (!id) return json({ error: "id required" }, 400);
   466→    const { error } = await admin.from("wallets").delete().eq("id", id);
   467→    if (error) throw error;
   468→    return json({ ok: true });
   469→  }
   470→
   471→  return json({ error: "Method not allowed" }, 405);
   472→}

   474→// ─── API: /api/transactions ─────────────────────────────────────
   475→async function handleTransactions(req: Request, env: Env): Promise<Response> {
   476→  const url = new URL(req.url);
   477→  const admin = getAdmin(env);

   482→
   483→    let query = admin
   484→      .from("transactions")
   485→      .select("*")
   486→      .eq("user_id", userId)

   621→async function handleSummary(req: Request, env: Env): Promise<Response> {
   622→  const url = new URL(req.url);
   623→  const userId = url.searchParams.get("user_id");
   624→  if (!userId) return json({ error: "user_id required" }, 400);
   625→
   626→  const admin = getAdmin(env);
   627→  const [txnsRes, debtsRes] = await Promise.all([
   628→    admin.from("transactions").select("*").eq("user_id", userId),
   629→    admin.from("debts").select("*").eq("user_id", userId),
   630→  ]);
   631→
   632→  if (txnsRes.error) throw txnsRes.error;
   633→  if (debtsRes.error) throw debtsRes.error;
   634→
   635→  const transactions = txnsRes.data || [];
   636→  const debts = debtsRes.data || [];
   637→
   638→  const totalIncome = transactions.filter((t: { type: string }) => t.type === "income").reduce((s: number, t: { amount: number }) => s + t.amount, 0);
   639→  const totalExpenses = transactions.filter((t: { type: string }) => t.type === "expense").reduce((s: number, t: { amount: number }) => s + t.amount, 0);
   640→  const activeBalance = totalIncome - totalExpenses;
   641→  const totalDebt = debts.filter((d: { type: string; status: string }) => d.type === "debt" && d.status !== "paid").reduce((s: number, d: { amount: number; paid_amount: number }) => s + (d.amount - d.paid_amount), 0);
   642→  const totalReceivable = debts.filter((d: { type: string; status: string }) => d.type === "receivable" && d.status !== "paid").reduce((s: number, d: { amount: number; paid_amount: number }) => s + (d.amount - d.paid_amount), 0);
   643→
   644→  const now = new Date();
   645→  const monthlyData = [];
   646→  for (let i = 11; i >= 0; i--) {
   647→    const mStart = new Date(now.getFullYear(), now.getMonth() - i, 1);
   648→    const mEnd = new Date(now.getFullYear(), now.getMonth() - i + 1, 1);
   649→    const mTxns = transactions.filter((t: { occurred_at: string }) => { const d = new Date(t.occurred_at); return d >= mStart && d < mEnd; });
   650→    const mIncome = mTxns.filter((t: { type: string }) => t.type === "income").reduce((s: number, t: { amount: number }) => s + t.amount, 0);
   651→    const mExpenses = mTxns.filter((t: { type: string }) => t.type === "expense").reduce((s: number, t: { amount: number }) => s + t.amount, 0);
   652→    monthlyData.push({ month: mStart.toLocaleDateString("en-US", { month: "short", year: "2-digit" }), income: mIncome, expenses: mExpenses, net: mIncome - mExpenses });
   653→  }
   654→
   655→  return json({ activeBalance, totalIncome, totalExpenses, totalDebt, totalReceivable, monthlyData });
   656→}

   965→// ─── Midtrans: Webhook Handler ──────────────────────────────────
   966→async function handleMidtransWebhook(req: Request, env: Env): Promise<Response> {
   967→  const body = await getBody(req);
   968→
   969→  const transactionStatus = body.transaction_status as string;
   970→  const orderId = body.order_id as string;
   971→  const paymentType = body.payment_type as string;
   972→  const metadata = body.metadata as { user_id?: string; plan?: string } | undefined;
   973→  const userId = metadata?.user_id || (body.custom_field1 as string);
   974→
   975→  if (!orderId || !transactionStatus) {
   976→    return json({ error: "Invalid webhook payload" }, 400);
   977→  }
   978→
   979→  // Verify signature to prevent fake webhook attacks
   980→  const signatureKey = body.signature_key as string;
   981→  const grossAmount = body.gross_amount as string;
   982→  if (signatureKey && env.MIDTRANS_SERVER_KEY) {
   983→    const expectedSig = await sha512(orderId + transactionStatus + grossAmount + env.MIDTRANS_SERVER_KEY);
   984→    if (signatureKey !== expectedSig) {
   985→      return json({ error: "Invalid signature" }, 403);
   986→    }
   987→  }
   988→
   989→  const admin = getAdmin(env);
   990→
   991→  // Determine if payment is successful
   992→  const isSuccess =
   993→    transactionStatus === "capture" ||
   994→    transactionStatus === "settlement";
   995→
   996→  const isPending = transactionStatus === "pending";
   997→  const isFailure =
   998→    transactionStatus === "deny" ||
   999→    transactionStatus === "expire" ||
  1000→    transactionStatus === "cancel";
  1001→
  1002→  if (isSuccess && userId) {
  1003→    // Activate Pro plan — determine duration from gross_amount
  1004→    const amount = parseFloat(grossAmount || "29000");
  1005→    const isAnnual = amount >= 278400;
  1006→    const durationDays = isAnnual ? 365 : 30;
  1007→    const billingSuffix = isAnnual ? "_annual" : "_monthly";
  1008→    const expiresAt = new Date(Date.now() + durationDays * 86400000).toISOString();
  1009→    // Try update with billing column; if it doesn't exist, fall back to without
  1010→    let updateData: Record<string, unknown> = {
  1011→      plan: "pro",
  1012→      started_at: new Date().toISOString(),
  1013→      expires_at: expiresAt,
  1014→      billing: isAnnual ? "annual" : "monthly",
  1015→      payment_method: `midtrans_${paymentType}${billingSuffix}`,
  1016→      transaction_id: orderId,
  1017→    };
  1018→    let { error } = await admin
  1019→      .from("subscriptions")
  1020→      .update(updateData)
  1021→      .eq("user_id", userId);
  1022→
  1023→    // If billing column doesn't exist, retry without it
  1024→    if (error && (error.message || "").includes("billing")) {
  1025→      const { billing: _, ...fallbackData } = updateData;
  1026→      const retryResult = await admin.from("subscriptions").update(fallbackData).eq("user_id", userId);
  1027→      error = retryResult.error;
  1028→    }
  1029→
  1030→    if (error) console.error("Failed to update subscription:", error);
  1031→    else await invalidateSubscriptionCache(env, userId);
  1032→  } else if (isFailure && userId) {
  1033→    // Revert to free
  1034→    await admin
  1035→      .from("subscriptions")
  1036→      .update({ plan: "free", transaction_id: orderId })
  1037→      .eq("user_id", userId);
  1038→    await invalidateSubscriptionCache(env, userId);
  1039→  }
  1040→
  1041→  // Always return 200 to Midtrans
  1042→  return json({ ok: true });
  1043→}

  1179→    for (const uid of userIds) {

  1181→    }

  1183→  }

  1277→// ─── API: /api/cache — KV Feature Flags, Cache & Sessions ──────
  1278→async function handleCache(req: Request, env: Env): Promise<Response> {
  1279→  const url = new URL(req.url);
  1280→  const subPath = url.pathname.replace("/api/cache", "").replace("//", "/");
  1281→
  1282→  // ── Feature Flags ──
  1283→  if (subPath === "/flags" || subPath === "/flags/") {
  1284→    if (req.method === "GET") {
  1285→      // Get all feature flags (merged with defaults)
  1286→      const flags: Record<string, string> = { ...DEFAULT_FEATURE_FLAGS };
  1287→      try {
  1288→        const list = await env.arus_kv.list({ prefix: "ff:" });
  1289→        for (const k of list.keys) {
  1290→          const val = await env.arus_kv.get(k.name);
  1291→          if (val !== null) flags[k.name] = val;
  1292→        }
  1293→      } catch {}
  1294→      return json({ flags });
  1295→    }
  1296→
  1297→    if (req.method === "PUT") {
  1298→      // Admin: set a feature flag
  1299→      const body = await getBody(req);
  1300→      const flag = body.flag as string;
  1301→      const value = body.value as string;
  1302→      if (!flag || value === undefined) return json({ error: "flag dan value required" }, 400);
  1303→      await setFeatureFlag(env, flag, value);
  1304→      return json({ ok: true, flag: `ff:${flag}`, value });
  1305→    }
  1306→  }
  1307→
  1308→  // ── Generic KV Get/Put/Delete ──
  1309→  if (req.method === "GET") {
  1310→    const key = url.searchParams.get("key");
  1311→    if (key) {
  1312→      const value = await env.arus_kv.get(key);
  1313→      return json({ key, value });
  1314→    }
  1315→    // List all keys with prefix
  1316→    const prefix = url.searchParams.get("prefix") || "";
  1317→    const list = await env.arus_kv.list({ prefix });
  1318→    return json({ keys: list.keys.map((k) => ({ name: k.name, expiration: k.expiration })) });
  1319→  }
  1320→
  1321→  if (req.method === "PUT") {
  1322→    const body = await getBody(req);
  1323→    const key = body.key as string;
  1324→    const value = body.value as string;
  1325→    const ttl = body.ttl as number | undefined;
  1326→    if (!key || value === undefined) return json({ error: "key dan value required" }, 400);
  1327→    const opts = ttl ? { expirationTtl: ttl } : undefined;
  1328→    await env.arus_kv.put(key, value, opts);
  1329→    return json({ ok: true, key });
  1330→  }
  1331→
  1332→  if (req.method === "DELETE") {
  1333→    const key = url.searchParams.get("key");
  1334→    if (!key) return json({ error: "key required" }, 400);
  1335→    await env.arus_kv.delete(key);
  1336→    return json({ ok: true });
  1337→  }
  1338→
  1339→  return json({ error: "Method not allowed" }, 405);
  1340→}

  1342→// ─── Security Headers ──────────────────────────────────────────
  1343→function withSecurityHeaders(response: Response): Response {
  1344→  const headers = new Headers(response.headers);
  1345→  // HSTS — force HTTPS for 1 year (only set if on HTTPS)
  1346→  headers.set("Strict-Transport-Security", "max-age=31536000; includeSubDomains");
  1347→  // Prevent MIME type sniffing
  1348→  headers.set("X-Content-Type-Options", "nosniff");
  1349→  // Prevent clickjacking (allow same-origin framing for Midtrans Snap)
  1350→  headers.set("X-Frame-Options", "SAMEORIGIN");
  1351→  // XSS protection (legacy, but still useful for older browsers)
  1352→  headers.set("X-XSS-Protection", "1; mode=block");
  1353→  // Referrer policy
  1354→  headers.set("Referrer-Policy", "strict-origin-when-cross-origin");
  1355→  // Permissions policy — limit features
  1356→  headers.set("Permissions-Policy", "camera=(), microphone=(), geolocation=()");
  1357→  return new Response(response.body, { status: response.status, statusText: response.statusText, headers });
  1358→}
