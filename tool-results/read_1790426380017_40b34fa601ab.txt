    67→// ─── Admin Auth Verification (with KV session cache) ───────────
    68→async function verifyAdmin(req: Request, env: Env): Promise<{ userId: string; email: string } | null> {
    69→  const authHeader = req.headers.get("Authorization");
    70→  if (!authHeader || !authHeader.startsWith("Bearer ")) return null;
    71→  const token = authHeader.replace("Bearer ", "");
    72→
    73→  // Check KV session cache first
    74→  const cached = await getCachedSession(env, token);
    75→  if (cached) {
    76→    const adminEmail = env.ADMIN_EMAIL || "riskiakbarp123@gmail.com";
    77→    if (cached.email !== adminEmail) return null;
    78→    return cached;
    79→  }
    80→
    81→  // Verify token+cache session
    82→  const res = await fetch(`${env.SUPABASE_URL}/auth/v1/user`, {
    83→    headers: { Authorization: `Bearer ${token}`, apikey: env.SUPABASE_ANON_KEY },
    84→  });
    85→  if (!res.ok) return null;
    86→  const user = await res.json() as { id: string; email: string };
    87→
    88→  const adminEmail = env.ADMIN_EMAIL || "riskiakbarp123@gmail.com";
    89→  if (user.email !== adminEmail) return null;
    90→
    91→  // Cache the session
    92→  const sessionData = { userId: user.id, email: user.email };
    93→  await cacheSession(env, token, sessionData);
    94→
    95→  return sessionData;
    96→}

    98→// ─── API: /api/admin/* ────────────────────────────────────────
    99→async function handleAdmin(req: Request, env: Env, path: string): Promise<Response> {
   100→  // Verify admin access for all admin routes
   101→  const admin = await verifyAdmin(req, env);
   102→  if (!admin) return json({ error: "Akses ditolak. Halaman ini hanya untuk admin." }, 403);
   103→
   104→  const sb = getAdmin(env);
   105→  const url = new URL(req.url);
   106→
   107→  // GET /api/admin/stats — Dashboard overview stats
   108→  if (path === "/admin/stats" && req.method === "GET") {
   109→    const sevenDaysAgo = new Date(Date.now() - 7 * 86400000).toISOString();
   110→
   111→    const [profilesRes, subsRes, recentRes] = await Promise.all([
   112→      sb.from("profiles").select("id, created_at", { count: "exact" }),
   113→      sb.from("subscriptions").select("plan, expires_at"),
   114→      sb.from("profiles").select("id").gte("created_at", sevenDaysAgo),
   115→    ]);
   116→
   117→    if (profilesRes.error) throw profilesRes.error;
   118→    if (subsRes.error) throw subsRes.error;
   119→    if (recentRes.error) throw recentRes.error;
   120→
   121→    const totalUsers = profilesRes.count ?? (profilesRes.data || []).length;
   122→    const subs = subsRes.data || [];
   123→    const proCount = subs.filter((s: { plan: string }) => s.plan === "pro").length;
   124→    const freeCount = totalUsers - proCount;
   125→    const now = new Date();
   126→    const activePro = subs.filter((s: { plan: string; expires_at: string | null }) => s.plan === "pro" && s.expires_at && new Date(s.expires_at) >= now).length;
   127→    const expiredPro = subs.filter((s: { plan: string; expires_at: string | null }) => s.plan === "pro" && s.expires_at && new Date(s.expires_at) < now).length;
   128→    const recentSignups = (recentRes.data || []).length;
   129→    const monthlyRevenue = proCount * 29000;
   130→
   131→    return json({
   132→      data: { totalUsers, proCount, freeCount, monthlyRevenue, recentSignups, activePro, expiredPro },
   133→    });
   134→  }
   135→
   136→  // GET /api/admin/users — List all users with profile + subscription info
   137→  if (path === "/admin/users" && req.method === "GET") {
   138→    const [profilesRes, subsRes] = await Promise.all([
   139→      sb.from("profiles").select("*").order("created_at", { ascending: false }),
   140→      sb.from("subscriptions").select("user_id, plan, started_at, expires_at"),
   141→    ]);
   142→
   143→    if (profilesRes.error) throw profilesRes.error;
   144→    if (subsRes.error) throw subsRes.error;
   145→
   146→    const profiles = profilesRes.data || [];
   147→    const subMap = new Map<string, { plan: string; started_at: string | null; expires_at: string | null }>();
   148→    for (const s of (subsRes.data || [])) {
   149→      subMap.set(s.user_id, s);
   150→    }
   151→
   152→    // Try to get emails from auth.users via admin API (list users)
   153→    let emailMap = new Map<string, string>();
   154→    try {
   155→      const authRes = await fetch(`${env.SUPABASE_URL}/auth/v1/admin/users?limit=1000`, {
   156→        headers: { Authorization: `Bearer ${env.SUPABASE_SERVICE_KEY}`, apikey: env.SUPABASE_ANON_KEY },
   157→      });
   158→      if (authRes.ok) {
   159→        const authData = await authRes.json() as { users: { id: string; email: string }[] };
   160→        for (const u of (authData.users || [])) {
   161→          emailMap.set(u.id, u.email);
   162→        }
   163→      }
   164→    } catch {
   165→      // If admin API not available, proceed without emails
   166→    }
   167→
   168→    const users = profiles.map((p: { id: string; user_id?: string; name?: string; created_at: string }) => {
   169→      const uid = p.user_id || p.id;
   170→      const sub = subMap.get(uid);
   171→      return {
   172→        user_id: uid,
   173→        id: p.id,
   174→        name: p.name || "—",
   175→        email: emailMap.get(uid) || "—",
   176→        plan: sub?.plan || "free",
   177→        created_at: p.created_at,
   178→      };
   179→    });
   180→
   181→    return json({ data: users });
   182→  }
   183→
   184→  // GET /api/admin/subscriptions/update — should be POST, reject GET
   185→  // POST /api/admin/subscription/update — Manually update a user's plan
   186→  if (path === "/admin/subscription/update" && req.method === "POST") {
   187→    const body = await getBody(req);
   188→    const userId = body.user_id as string;
   189→    const plan = body.plan as string;
   190→    const expiresAt = body.expires_at as string | undefined;
   191→
   192→    if (!userId || !plan) return json({ error: "user_id dan plan required" }, 400);
   193→    if (plan !== "free" && plan !== "pro") return json({ error: "Plan harus 'free' atau 'pro'" }, 400);
   194→
   195→    // Check if subscription exists
   196→    const { data: existing } = await sb.from("subscriptions").select("id").eq("user_id", userId).single();
   197→
   198→    let result;
   199→    if (existing) {
   200→      const updateData: Record<string, unknown> = { plan, started_at: new Date().toISOString() };
   201→      if (plan === "pro") {
   202→        updateData.expires_at = expiresAt || new Date(Date.now() + 30 * 86400000).toISOString();
   203→      } else {
   204→        updateData.expires_at = expiresAt || null;
   205→      }
   206→      result = await sb.from("subscriptions").update(updateData).eq("user_id", userId).select().single();
   207→    } else {
   208→      const insertData: Record<string, unknown> = { user_id: userId, plan };
   209→      if (plan === "pro") {
   210→        insertData.expires_at = expiresAt || new Date(Date.now() + 30 * 86400000).toISOString();
   211→      }
   212→      result = await sb.from("subscriptions").insert(insertData).select().single();
   213→    }
   214→
   215→    if (result.error) throw result.error;
   216→    return json({ data: result.data });
   217→  }
   218→
   219→  // GET /api/admin/subscriptions/update — method not allowed
   220→  if (path === "/admin/subscription/update" && req.method === "GET") {
   221→    return json({ error: "Method not allowed. Gunakan POST." }, 405);
   222→  }
   223→
   224→  // GET /api/admin/subscriptions — Alias for subscription/update GET (not needed, but for completeness)
   225→
   226→  // GET /api/admin/subscriptions — List all subscriptions
   227→  if (path === "/admin/subscriptions" || path === "/admin/subscriptions/") {
   228→    if (req.method !== "GET") return json({ error: "Method not allowed" }, 405);
   229→
   230→    const { data, error } = await sb.from("subscriptions").select("*").order("started_at", { ascending: false, nullsFirst: "last" });
   231→    if (error) throw error;
   232→
   233→    // Get emails
   234→    let emailMap = new Map<string, string>();
   235→    try {
   236→      const authRes = await fetch(`${env.SUPABASE_URL}/auth/v1/admin/users?limit=1000`, {
   237→        headers: { Authorization: `Bearer ${env.SUPABASE_SERVICE_KEY}`, apikey: env.SUPABASE_ANON_KEY },
   238→      });
   239→      if (authRes.ok) {
   240→        const authData = await authRes.json() as { users: { id: string; email: string }[] };
   241→        for (const u of (authData.users || [])) {
   242→          emailMap.set(u.id, u.email);
   243→        }
   244→      }
   245→    } catch {
   246→      // proceed without emails
   247→    }
   248→
   249→    const subs = (data || []).map((s: { user_id: string; [key: string]: unknown }) => ({
   250→      ...s,
   251→      email: emailMap.get(s.user_id) || "—",
   252→    }));
   253→
   254→    return json({ data: subs });
   255→  }
   256→
   257→  // GET /api/admin/transactions — Recent transactions across all users
   258→  if (path === "/admin/transactions" && req.method === "GET") {
   259→    const limit = parseInt(url.searchParams.get("limit") || "50", 10);
   260→    const offset = parseInt(url.searchParams.get("offset") || "0", 10);
   261→    const from = url.searchParams.get("from");
   262→    const to = url.searchParams.get("to");
   263→    const type = url.searchParams.get("type");
   264→
   265→    let query = sb
   266→      .from("transactions")
   267→      .select("*")
   268→      .order("occurred_at", { ascending: false })
   269→      .range(offset, offset + limit - 1);
   270→
   271→    if (from) query = query.gte("occurred_at", from);
   272→    if (to) query = query.lte("occurred_at", to);
   273→    if (type) query = query.eq("type", type);
   274→
   275→    const { data, error } = await query;
   276→    if (error) throw error;
   277→
   278→    // Get emails for the user_ids in the transactions
   279→    const userIds = [...new Set((data || []).map((t: { user_id: string }) => t.user_id))];
   280→    let emailMap = new Map<string, string>();
   281→    if (userIds.length > 0) {
   282→      try {
   283→        const authRes = await fetch(`${env.SUPABASE_URL}/auth/v1/admin/users?limit=1000`, {
   284→          headers: { Authorization: `Bearer ${env.SUPABASE_SERVICE_KEY}`, apikey: env.SUPABASE_ANON_KEY },
   285→        });
   286→        if (authRes.ok) {
   287→          const authData = await authRes.json() as { users: { id: string; email: string }[] };
   288→          for (const u of (authData.users || [])) {
   289→            emailMap.set(u.id, u.email);
   290→          }
   291→        }
   292→      } catch {
   293→        // proceed without emails
   294→      }
   295→    }
   296→
   297→    const txns = (data || []).map((t: { user_id: string; [key: string]: unknown }) => ({
   298→      ...t,
   299→      user_email: emailMap.get(t.user_id) || "—",
   300→    }));
   301→
   302→    return json({ data: txns });
   303→  }
   304→
   305→  // DELETE /api/admin/user — Delete a user (auth + profile + subscriptions)
   306→  if (path === "/admin/user" && req.method === "DELETE") {
   307→    const body = await getBody(req);
   308→    const userId = body.user_id as string;
   309→    if (!userId) return json({ error: "user_id required" }, 400);
   310→    // Prevent admin from deleting themselves
   311→    if (userId === admin.userId) return json({ error: "Tidak bisa hapus akun admin sendiri" }, 400);
   312→
   313→    // Delete from auth.users via admin API
   314→    try {
   315→      await fetch(`${env.SUPABASE_URL}/auth/v1/admin/users/${userId}`, {
   316→        method: "DELETE",
   317→        headers: { Authorization: `Bearer ${env.SUPABASE_SERVICE_KEY}`, apikey: env.SUPABASE_ANON_KEY },
   318→      });
   319→    } catch { /* continue even if auth delete fails */ }
   320→
   321→    // Delete profile, subscription, and user data
   322→    await Promise.all([
   323→      sb.from("profiles").delete().eq("user_id", userId),
   324→      sb.from("subscriptions").delete().eq("user_id", userId),
   325→      sb.from("wallets").delete().eq("user_id", userId),
   326→      sb.from("transactions").delete().eq("user_id", userId),
   327→      sb.from("debts").delete().eq("user_id", userId),
   328→      sb.from("budgets").delete().eq("user_id", userId),
   329→      sb.from("custom_categories").delete().eq("user_id", userId),
   330→    ]);
   331→
   332→    // Invalidate caches
   333→    await invalidateSubscriptionCache(env, userId);
   334→    try { await env.arus_kv.delete(`last-backup:${userId}`); } catch {}
   335→
   336→    return json({ ok: true, deleted: userId });
   337→  }
   338→
   339→  // POST /api/admin/user/ban — Ban/unban a user (disable their access)
   340→  if (path === "/admin/user/ban" && req.method === "POST") {
   341→    const body = await getBody(req);
   342→    const userId = body.user_id as string;
   343→    const ban = body.ban as boolean;
   344→    if (!userId) return json({ error: "user_id required" }, 400);
   345→    if (userId === admin.userId) return json({ error: "Tidak bisa ban akun admin sendiri" }, 400);
   346→
   347→    // Update user's app_metadata to ban them
   348→    const authRes = await fetch(`${env.SUPABASE_URL}/auth/v1/admin/users/${userId}`, {
   349→      method: "PUT",
   350→      headers: { Authorization: `Bearer ${env.SUPABASE_SERVICE_KEY}`, apikey: env.SUPABASE_ANON_KEY, "Content-Type": "application/json" },
   351→      body: JSON.stringify({ app_metadata: { banned: ban } }),
   352→    });
   353→    if (!authRes.ok) {
   354→      const data = await authRes.json();
   355→      return json({ error: data.msg || "Gagal update user" }, authRes.status);
   356→    }
   357→
   358→    // Also invalidate their session cache
   359→    await invalidateSubscriptionCache(env, userId);
   360→
   361→    return json({ ok: true, user_id: userId, banned: ban });
   362→  }
   363→
   364→  // POST /api/admin/user/reset-password — Admin resets a user's password
   365→  if (path === "/admin/user/reset-password" && req.method === "POST") {
   366→    const body = await getBody(req);
   367→    const userId = body.user_id as string;
   368→    if (!userId) return json({ error: "user_id required" }, 400);
   369→
   370→    // Send password reset email via Supabase
   371→    // First get the user's email
   372→    const authRes = await fetch(`${env.SUPABASE_URL}/auth/v1/admin/users/${userId}`, {
   373→      headers: { Authorization: `Bearer ${env.SUPABASE_SERVICE_KEY}`, apikey: env.SUPABASE_ANON_KEY },
   374→    });
   375→    if (!authRes.ok) return json({ error: "User tidak ditemukan" }, 404);
   376→    const userData = await authRes.json() as { email: string };
   377→    if (!userData.email) return json({ error: "User tidak punya email" }, 400);
   378→
   379→    // Send recovery email
   380→    await fetch(`${env.SUPABASE_URL}/auth/v1/recover`, {
   381→      method: "POST",
   382→      headers: { "Content-Type": "application/json", apikey: env.SUPABASE_ANON_KEY },
   383→      body: JSON.stringify({ email: userData.email }),
   384→    });
   385→
   386→    return json({ ok: true, message: `Link reset password dikirim ke ${userData.email}` });
   387→  }
   388→
   389→  return json({ error: "Not found" }, 404);
   390→}

   393→async function handleWallets(req: Request, env: Env): Promise<Response> {
   394→  const url = new URL(req.url);
   395→  const admin = getAdmin(env);
   396→
   397→  if (req.method === "GET") {
   398→    const userId = url.searchParams.get("user_id");
   399→    if (!userId) return json({ error: "user_id required" }, 400);
   400→    const { data, error } = await admin
   401→      .from("wallets")
   402→      .select("*")
   403→      .eq("user_id", userId)
   404→      .order("created_at", { ascending: true });
   405→    if (error) throw error;
   406→    return json({ data });
   407→  }
   408→
   409→  if (req.method === "POST") {
   410→    const body = await getBody(req);
   411→    const userId = body.user_id as string;
   412→
   413→    // Check plan limit (with KV cache)
   414→    const isPro = await isProUser(env, userId);
   415→    if (!isPro) {
   416→      const { count } = await admin
   417→        .from("wallets")
   418→        .select("*", { count: "exact", head: true })
   419→        .eq("user_id", userId);
   420→      if ((count || 0) >= 2) {
   421→        return json(
   422→          { error: "Limit dompet tercapai (2 untuk Gratis). Upgrade ke Pro untuk unlimited.", limit: "wallets" },
   423→          403
   424→        );
   425→      }
   426→    }
   427→
   428→    const { data, error } = await admin.from("wallets").insert(body).select().single();
   429→    if (error) throw error;
   430→    return json({ data }, 201);
   431→  }
   432→
   433→  if (req.method === "DELETE") {
   434→    const id = url.searchParams.get("id");
   435→    if (!id) return json({ error: "id required" }, 400);
   436→    const { error } = await admin.from("wallets").delete().eq("id", id);
   437→    if (error) throw error;
   438→    return json({ ok: true });
   439→  }
   440→
   441→  return json({ error: "Method not allowed" }, 405);
   442→}

   445→async function handleTransactions(req: Request, env: Env): Promise<Response> {
   446→  const url = new URL(req.url);
   447→  const admin = getAdmin(env);
   448→
   449→  if (req.method === "GET") {
   450→    const userId = url.searchParams.get("user_id");
   451→    if (!userId) return json({ error: "user_id required" }, 400);
   452→
   453→    let query = admin
   454→      .from("transactions")
   455→      .select("*")
   456→      .eq("user_id", userId)
   457→      .order("occurred_at", { ascending: false });
   458→
   459→    const from = url.searchParams.get("from");
   460→    const to = url.searchParams.get("to");
   461→    const type = url.searchParams.get("type");
   462→    const category = url.searchParams.get("category");
   463→    const wallet = url.searchParams.get("wallet");
   464→
   465→    if (from) query = query.gte("occurred_at", from);
   466→    if (to) query = query.lte("occurred_at", to);
   467→    if (type) query = query.eq("type", type);
   468→    if (category) query = query.eq("category", category);
   469→    if (wallet) query = query.eq("wallet_id", wallet);
   470→
   471→    const { data, error } = await query;
   472→    if (error) throw error;
   473→    return json({ data });
   474→  }
   475→
   476→  if (req.method === "POST") {
   477→    const body = await getBody(req);
   478→    const { data, error } = await admin.from("transactions").insert(body).select().single();
   479→    if (error) throw error;
   480→    return json({ data }, 201);
   481→  }
   482→
   483→  if (req.method === "DELETE") {
   484→    const id = url.searchParams.get("id");
   485→    if (!id) return json({ error: "id required" }, 400);
   486→    const { error } = await admin.from("transactions").delete().eq("id", id);
   487→    if (error) throw error;
   488→    return json({ ok: true });
   489→  }
   490→
   491→  return json({ error: "Method not allowed" }, 405);
   492→}

   494→// ─── API: /api/debts ────────────────────────────────────────────

   582→async function handleSummary(req: Request, env: Env): Promise<Response> {
   583→  const url = new URL(req.url);
   584→  const userId = url.searchParams.get("user_id");
   585→  if (!userId) return json({ error: "user_id required" }, 400);
   586→
   587→  const admin = getAdmin(env);
   588→  const [txnsRes, debtsRes] = await Promise.all([
   589→    admin.from("transactions").select("*").eq("user_id", userId),
   590→    admin.from("debts").select("*").eq("user_id", userId),
   591→  ]);
   592→
   593→  if (txnsRes.error) throw txnsRes.error;
   594→  if (debtsRes.error) throw debtsRes.error;
   595→
   596→  const transactions = txnsRes.data || [];
   597→  const debts = debtsRes.data || [];
   598→
   599→  const totalIncome = transactions.filter((t: { type: string }) => t.type === "income").reduce((s: number, t: { amount: number }) => s + t.amount, 0);
   600→  const totalExpenses = transactions.filter((t: { type: string }) => t.type === "expense").reduce((s: number, t: { amount: number }) => s + t.amount, 0);
   601→  const activeBalance = totalIncome - totalExpenses;
   602→  const totalDebt = debts.filter((d: { type: string; status: string }) => d.type === "debt" && d.status !== "paid").reduce((s: number, d: { amount: number; paid_amount: number }) => s + (d.amount - d.paid_amount), 0);
   603→  const totalReceivable = debts.filter((d: { type: string; status: string }) => d.type === "receivable" && d.status !== "paid").reduce((s: number, d: { amount: number; paid_amount: number }) => s + (d.amount - d.paid_amount), 0);
   604→
   605→  const now = new Date();
   606→  const monthlyData = [];
   607→  for (let i = 11; i >= 0; i--) {
   608→    const mStart = new Date(now.getFullYear(), now.getMonth() - i, 1);
   609→    const mEnd = new Date(now.getFullYear(), now.getMonth() - i + 1, 1);
   610→    const mTxns = transactions.filter((t: { occurred_at: string }) => { const d = new Date(t.occurred_at); return d >= mStart && d < mEnd; });
   611→    const mIncome = mTxns.filter((t: { type: string }) => t.type === "income").reduce((s: number, t: { amount: number }) => s + t.amount, 0);
   612→    const mExpenses = mTxns.filter((t: { type: string }) => t.type === "expense").reduce((s: number, t: { amount: number }) => s + t.amount, 0);
   613→    monthlyData.push({ month: mStart.toLocaleDateString("en-US", { month: "short", year: "2-digit" }), income: mIncome, expenses: mExpenses, net: mIncome - mExpenses });
   614→  }
   615→
   616→  return json({ activeBalance, totalIncome, totalExpenses, totalDebt, totalReceivable, monthlyData });
   617→}

   949→  if (isSuccess && userId) {
   950→    // Activate Pro plan
   951→    const expiresAt = new Date(Date.now() + 30 * 86400000).toISOString();
   952→    const { error } = await admin
   953→      .from("subscriptions")
   954→      .update({
   955→        plan: "pro",
   956→        started_at: new Date().toISOString(),
   957→        expires_at: expiresAt,
   958→        payment_method: `midtrans_${paymentType}`,
   959→        transaction_id: orderId,
   960→      })
   961→      .eq("user_id", userId);
   962→
   963→    if (error) console.error("Failed to update subscription:", error);
   964→  } else if (isFailure && userId) {
   965→    // Revert to free
   966→    await admin
   967→      .from("subscriptions")
   968→      .update({ plan: "free", transaction_id: orderId })
   969→      .eq("user_id", userId);
   970→  }
   971→
   972→  // Always return 200 to Midtrans
   973→  return json({ ok: true });
   974→}

   992→      }, 503);
   993→    }
   994→
   995→    // Route matching

   998→    if (path === "/wallets" || path.startsWith("/wallets")) return await handleWallets(request, env);
   999→    if (path === "/transactions" || path.startsWith("/transactions")) return await handleTransactions(request, env);
  1000→    if (path === "/debts" || path.startsWith("/debts")) return await handleDebts(request, env);
  1001→    if (path === "/subscription" || path.startsWith("/subscription")) return await handleSubscription(request, env);
  1002→    if (path === "/summary" || path.startsWith("/summary")) return await handleSummary(request, env);
  1003→    if (path === "/categorize" || path.startsWith("/categorize")) return await handleCategorize(request);

  1005→    if (path === "/auth/signup") return await handleSignup(request, env);
  1006→    if (path === "/auth/me") return await handleAuthMe(request, env);

  1009→    if (path === "/auth/reset-password") return await handleResetPassword(request, env);
  1010→    if (path === "/sync" || path.startsWith("/sync")) return await handleSync(request, env);
  1011→    if (path === "/payment/create") return await handleMidtransCreate(request, env);
  1012→    if (path === "/payment/webhook") return await handleMidtransWebhook(request, env);
  1013→    // R2 storage routes — disabled (no R2 binding)
  1014→    if (path === "/storage/backup" || path.startsWith("/storage/backup")) return r2Disabled();
  1015→    if (path === "/storage/restore" || path.startsWith("/storage/restore")) return r2Disabled();
  1016→    if (path === "/storage" || path.startsWith("/storage")) return r2Disabled();
  1017→    if (path === "/cache" || path.startsWith("/cache")) return await handleCache(request, env);
  1018→
  1019→    // Admin routes
  1020→    if (path.startsWith("/admin")) return await handleAdmin(request, env, path);
  1021→
  1022→    return json({ error: "Not found" }, 404);
  1023→  } catch (err: unknown) {
  1024→    const msg = err instanceof Error ? err.message : "Server error";
  1025→    return json({ error: msg }, 500);
  1026→  }
  1027→}

  1029→// ─── Cron: Check expired Pro subscriptions ────────────────────
  1030→async function handleCron(env: Env): Promise<void> {
  1031→  const admin = getAdmin(env);
  1032→  const now = new Date().toISOString();
  1033→
  1034→  // Find all Pro subscriptions that have expired
  1035→  const { data: expired, error } = await admin
  1036→    .from("subscriptions")
  1037→    .select("user_id, expires_at")
  1038→    .eq("plan", "pro")
  1039→    .lt("expires_at", now);
  1040→
  1041→  if (error) {
  1042→    console.error("Cron: failed to query expired subscriptions:", error);
  1043→    return;
  1044→  }
  1045→
  1046→  if (!expired || expired.length === 0) {
  1047→    console.log(`Cron: no expired subscriptions at ${now}`);
  1048→    return;
  1049→  }
  1050→
  1051→  // Downgrade expired Pro users to Free
  1052→  const userIds = expired.map((s: { user_id: string }) => s.user_id);
  1053→  const { error: updateError } = await admin
  1054→    .from("subscriptions")
  1055→    .update({ plan: "free" })
  1056→    .in("user_id", userIds);
  1057→
  1058→  if (updateError) {
  1059→    console.error("Cron: failed to downgrade:", updateError);
  1060→  } else {
  1061→    // Invalidate KV subscription caches for downgraded users
  1062→    for (const uid of userIds) {
  1063→      await invalidateSubscriptionCache(env, uid);
  1064→    }
  1065→    console.log(`Cron: downgraded ${userIds.length} expired Pro → Free`);
  1066→  }
  1067→}

  1160→// ─── API: /api/cache — KV Feature Flags, Cache & Sessions ──────
  1161→async function handleCache(req: Request, env: Env): Promise<Response> {
  1162→  const url = new URL(req.url);
  1163→  const subPath = url.pathname.replace("/api/cache", "").replace("//", "/");
  1164→
  1165→  // ── Feature Flags ──
  1166→  if (subPath === "/flags" || subPath === "/flags/") {
  1167→    if (req.method === "GET") {
  1168→      // Get all feature flags (merged with defaults)
  1169→      const flags: Record<string, string> = { ...DEFAULT_FEATURE_FLAGS };
  1170→      try {
  1171→        const list = await env.arus_kv.list({ prefix: "ff:" });
  1172→        for (const k of list.keys) {
  1173→          const val = await env.arus_kv.get(k.name);
  1174→          if (val !== null) flags[k.name] = val;
  1175→        }
  1176→      } catch {}
  1177→      return json({ flags });
  1178→    }
  1179→
  1180→    if (req.method === "PUT") {
  1181→      // Admin: set a feature flag
  1182→      const body = await getBody(req);
  1183→      const flag = body.flag as string;
  1184→      const value = body.value as string;
  1185→      if (!flag || value === undefined) return json({ error: "flag dan value required" }, 400);
  1186→      await setFeatureFlag(env, flag, value);
  1187→      return json({ ok: true, flag: `ff:${flag}`, value });
  1188→    }
  1189→  }
  1190→
  1191→  // ── Generic KV Get/Put/Delete ──
  1192→  if (req.method === "GET") {
  1193→    const key = url.searchParams.get("key");
  1194→    if (key) {
  1195→      const value = await env.arus_kv.get(key);
  1196→      return json({ key, value });
  1197→    }
  1198→    // List all keys with prefix
  1199→    const prefix = url.searchParams.get("prefix") || "";
  1200→    const list = await env.arus_kv.list({ prefix });
  1201→    return json({ keys: list.keys.map((k) => ({ name: k.name, expiration: k.expiration })) });
  1202→  }
  1203→
  1204→  if (req.method === "PUT") {
  1205→    const body = await getBody(req);
  1206→    const key = body.key as string;
  1207→    const value = body.value as string;
  1208→    const ttl = body.ttl as number | undefined;
  1209→    if (!key || value === undefined) return json({ error: "key dan value required" }, 400);
  1210→    const opts = ttl ? { expirationTtl: ttl } : undefined;
  1211→    await env.arus_kv.put(key, value, opts);
  1212→    return json({ ok: true, key });
  1213→  }
  1214→
  1215→  if (req.method === "DELETE") {
  1216→    const key = url.searchParams.get("key");
  1217→    if (!key) return json({ error: "key required" }, 400);
  1218→    await env.arus_kv.delete(key);
  1219→    return json({ ok: true });
  1220→  }
  1221→
  1222→  return json({ error: "Method not allowed" }, 405);
  1223→}
