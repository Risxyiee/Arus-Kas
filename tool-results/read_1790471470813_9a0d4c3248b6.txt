   245→// ─── API helper ───────────────────────────────────────────────
   246→async function api(path, opts = {}) {
   247→  const maxRetries = (opts._retries || 0);
   248→  const retryLimit = 2;
   249→  const headers = { 'Content-Type': 'application/json', ...(opts.headers || {}) };
   250→  if (token) headers['Authorization'] = 'Bearer ' + token;
   251→  try {
   252→    const res = await fetch('/api' + path, { ...opts, headers });
   253→    const text = await res.text();
   254→    try {
   255→      const data = JSON.parse(text);
   256→      if (!res.ok) throw new Error(data.error || 'Gagal memuat data');
   257→      return data;
   258→    } catch(e) {
   259→      if (!res.ok) throw new Error('Server error (HTTP ' + res.status + ')');
   260→      throw e;
   261→    }
   262→  } catch(e) {
   263→    if (maxRetries < retryLimit) {
   264→      await new Promise(r => setTimeout(r, Math.pow(2, maxRetries) * 1000));
   265→      return api(path, {...opts, _retries: maxRetries + 1});
   266→    }
   267→    throw e;
   268→  }
   269→}

   273→function fmtDate(d) { if (!d) return '—'; return new Date(d).toLocaleDateString('id-ID', {day:'numeric',month:'short',year:'numeric'}); }
   274→function fmtRelTime(d) {
   275→  if (!d) return '—';
   276→  const diff = Date.now() - new Date(d).getTime();
   277→  const days = Math.floor(diff / 86400000);

   284→async function doLogin() {
   285→  const email = document.getElementById('loginEmail').value.trim();
   286→  const pass = document.getElementById('loginPass').value;
   287→  const errEl = document.getElementById('loginError');
   288→  const btn = document.getElementById('loginBtn');
   289→  errEl.classList.remove('show');
   290→  btn.disabled = true; btn.textContent = 'Memproses...';
   291→
   292→  try {
   293→    // Step 1: Login to get token
   294→    const loginRes = await fetch('/api/auth/login', {
   295→      method: 'POST',
   296→      headers: { 'Content-Type': 'application/json' },
   297→      body: JSON.stringify({ email, password: pass })
   298→    });
   299→    const loginData = await loginRes.json();
   300→    if (!loginRes.ok) throw new Error(loginData.error_description || loginData.error || 'Login gagal');
   301→
   302→    token = loginData.access_token;
   303→    if (!token) throw new Error('Token tidak ditemukan');
   304→
   305→    // Step 2: Verify user with /api/auth/me
   306→    const meRes = await fetch('/api/auth/me', {
   307→      headers: { 'Authorization': 'Bearer ' + token }
   308→    });
   309→    const meData = await meRes.json();
   310→    if (!meRes.ok) throw new Error('Verifikasi gagal');
   311→
   312→    // Step 3: Check admin email
   313→    const userEmail = meData.data?.email || meData.email;
   314→    if (userEmail !== ADMIN_EMAIL) {
   315→      throw new Error('Akses ditolak. Halaman ini hanya untuk admin.');
   316→    }
   317→
   318→    // Success
   319→    sessionStorage.setItem('arus_admin_token', token);
   320→    document.getElementById('adminEmail').textContent = userEmail;
   321→    document.getElementById('loginScreen').style.display = 'none';
   322→    document.getElementById('dashScreen').style.display = 'flex';
   323→    loadSection('dashboard');
   324→
   325→  } catch (e) {
   326→    errEl.textContent = e.message;
   327→    errEl.classList.add('show');
   328→    token = '';
   329→  } finally {
   330→    btn.disabled = false; btn.textContent = 'Masuk';
   331→  }
   332→}

   914→async function loadKvKeys() {
   915→  const prefix = document.getElementById('kvPrefix')?.value || '';
   916→  const list = document.getElementById('kvKeysList');
   917→  if (!list) return;
   918→  list.innerHTML = '<span class="muted">Memuat...</span>';
   919→  try {
   920→    const res = await api('/cache?prefix=' + encodeURIComponent(prefix));
   921→    const keys = res.keys || [];
   922→    if (!keys.length) {
   923→      list.innerHTML = '<span class="muted">Tidak ada keys dengan prefix "' + esc(prefix) + '"</span>';
   924→    } else {
   925→      list.innerHTML = '<table style="width:100%"><thead><tr><th>Key</th><th>Expiry</th><th>Aksi</th></tr></thead><tbody>' +
   926→        keys.slice(0, 50).map(k => `<tr><td class="mono xs">${esc(k.name)}</td><td class="xs muted">${k.expiration ? new Date(k.expiration * 1000).toLocaleString('id') : 'Permanen'}</td><td><button class="btn mini" onclick="deleteKvKey('${esc(k.name)}')">Hapus</button></td></tr>`).join('') +
   927→        '</tbody></table>';
   928→    }
   929→  } catch (e) {
   930→    list.innerHTML = '<span class="neg">Error: ' + esc(e.message) + '</span>';
   931→  }
   932→}
