   613→function drawChart(){
   614→ const cv=$('#cash-chart');if(!cv||!cv.clientWidth)return;
   615→ const S=monthSeries(6);
   616→ const cs=getComputedStyle(document.documentElement);
   617→ const pos=cs.getPropertyValue('--pos').trim(),neg=cs.getPropertyValue('--neg').trim(),
   618→       ink=cs.getPropertyValue('--ink').trim(),mut=cs.getPropertyValue('--mut').trim(),
   619→       line=cs.getPropertyValue('--line').trim();
   620→ const dpr=window.devicePixelRatio||1,W=cv.clientWidth,H=cv.clientHeight;
   621→ cv.width=W*dpr;cv.height=H*dpr;
   622→ const ctx=cv.getContext('2d');ctx.scale(dpr,dpr);ctx.clearRect(0,0,W,H);
   623→ const padL=58,padR=8,padT=12,padB=26,pw=W-padL-padR,ph=H-padT-padB;
   624→ const max=Math.max(1,Math.max.apply(null,S.map(s=>Math.max(s.inc,s.exp))))*1.15;
   625→ const y=v=>padT+ph-(v/max)*ph;
   626→ ctx.font='10px ui-monospace,Consolas,monospace';ctx.fillStyle=mut;ctx.textAlign='right';ctx.textBaseline='middle';
   627→ for(let i=0;i<=4;i++){const v=max*i/4,yy=y(v);
   628→  ctx.strokeStyle=line;ctx.lineWidth=1;ctx.beginPath();ctx.moveTo(padL,yy);ctx.lineTo(W-padR,yy);ctx.stroke();
   629→  ctx.fillText(shortMoney(v),padL-8,yy);}
   630→ const gw=pw/S.length,bw=Math.min(16,gw*0.26);
   631→ function bar(x,y0,h,color){h=Math.max(h,.5);ctx.fillStyle=color;ctx.beginPath();
   632→  if(ctx.roundRect)ctx.roundRect(x,y0,bw,h,[3,3,0,0]);else ctx.rect(x,y0,bw,h);ctx.fill();}
   633→ ctx.textAlign='center';ctx.textBaseline='alphabetic';
   634→ S.forEach((s,i)=>{const cx=padL+gw*i+gw/2;
   635→  bar(cx-bw-3,y(s.inc),padT+ph-y(s.inc),pos);
   636→  bar(cx+3,y(s.exp),padT+ph-y(s.exp),neg);
   637→  ctx.fillStyle=mut;ctx.fillText(s.label,cx,H-8);});
   638→ ctx.strokeStyle=ink;ctx.setLineDash([5,4]);ctx.lineWidth=1.5;ctx.beginPath();
   639→ S.forEach((s,i)=>{const cx=padL+gw*i+gw/2,yy=y(Math.max(0,s.net));if(i)ctx.lineTo(cx,yy);else ctx.moveTo(cx,yy);});
   640→ ctx.stroke();ctx.setLineDash([]);
   641→ S.forEach((s,i)=>{const cx=padL+gw*i+gw/2,yy=y(Math.max(0,s.net));
   642→  ctx.fillStyle=ink;ctx.beginPath();ctx.arc(cx,yy,2.5,0,7);ctx.fill();});
   643→}
