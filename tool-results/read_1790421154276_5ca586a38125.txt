    26→// ─── Supabase Auth proxy ─────────────────────────────────────────
    27→async function supabaseAuth(endpoint: string, options: RequestInit): Promise<NextResponse> {
    28→  const res = await fetch(`${SUPABASE_URL}/auth/v1${endpoint}`, {
    29→    ...options,
    30→    headers: {
    31→      'Content-Type': 'application/json',
    32→      apikey: SUPABASE_ANON_KEY,
    33→      ...(options.headers as Record<string, string> || {}),
    34→    },
    35→  })
    36→  const data = await res.json()
    37→  if (!res.ok) return json({ error: data.error_description || data.msg || data.error || 'Auth gagal' }, res.status)
    38→  return json(data)
    39→}

    42→async function verifyAdmin(req: NextRequest): Promise<{ userId: string; email: string } | null> {
    43→  const authHeader = req.headers.get('Authorization')
    44→  if (!authHeader || !authHeader.startsWith('Bearer ')) return null
    45→  const token = authHeader.replace('Bearer ', '')
    46→
    47→  const res = await fetch(`${SUPABASE_URL}/auth/v1/user`, {
    48→    headers: { Authorization: `Bearer ${token}`, apikey: SUPABASE_ANON_KEY },
    49→  })
    50→  if (!res.ok) return null
    51→  const user = await res.json() as { id: string; email: string }
    52→  if (user.email !== ADMIN_EMAIL) return null
    53→  return { userId: user.id, email: user.email }

    66→async function sbSelect(table: string, query: string, filters?: Record<string, string>): Promise<{ data: unknown[] | null; error: string | null; count?: number }> {
    67→  let url = `${SUPABASE_URL}/rest/v1/${table}?select=${query}`
    68→  if (filters) {
    69→    for (const [k, v] of Object.entries(filters)) {
    70→      url += `&${k}=${encodeURIComponent(v)}`
    71→    }
    72→  }
    73→  const res = await fetch(url, { headers: adminHeaders() })
    74→  const data = await res.json()
    75→  if (!res.ok) return { data: null, error: data.message || data.error || 'Query failed' }
    76→  return { data: data as unknown[], error: null, count: (data as unknown[]).length }
    77→}

    80→  const res = await fetch(`${SUPABASE_URL}/rest/v1/${table}`, {
    81→    method: 'POST',
    82→    headers: { ...adminHeaders(), Prefer: 'return=representation' },
    83→    body: JSON.stringify(body),
    84→  })
    85→  const data = await res.json()
    86→  if (!res.ok) return { data: null, error: data.message || data.error || 'Insert failed' }
    87→  return { data, error: null }

    91→  let url = `${SUPABASE_URL}/rest/v1/${table}?`
    92→  for (const [k, v] of Object.entries(filters)) {
    93→    url += `${k}=${encodeURIComponent(v)}&`
    94→  }
    95→  const res = await fetch(url, {
    96→    method: 'PATCH',
    97→    headers: { ...adminHeaders(), Prefer: 'return=representation' },
    98→    body: JSON.stringify(body),
    99→  })
   100→  const data = await res.json()
   101→  if (!res.ok) return { data: null, error: data.message || data.error || 'Update failed' }
   102→  return { data, error: null }

   110→    const res = await fetch(`${SUPABASE_URL}/auth/v1/admin/users?limit=1000`, {

   112→    })
   113→    if (res.ok) {
   114→      const authData = await res.json() as { users: { id: string; email: string }[] }
   115→      for (const u of (authData.users || [])) {
   116→        emailMap.set(u.id, u.email)
   117→      }
   118→    }
   119→  } catch { /* proceed without emails */ }
   120→  return emailMap

   211→  const recentSignups = (recentRes.data || []).length
   212→
   213→  return json({

   262→// ─── Admin: Subscription Update ──────────────────────────────────
   263→async function handleAdminSubscriptionUpdate(req: NextRequest): Promise<NextResponse> {
   264→  try {
   265→    const body = await req.json()
   266→    const userId = body.user_id as string
   267→    const plan = body.plan as string
   268→    const expiresAt = body.expires_at as string | undefined
   269→
   270→    if (!userId || !plan) return json({ error: 'user_id dan plan required' }, 400)
   271→    if (plan !== 'free' && plan !== 'pro') return json({ error: "Plan harus 'free' atau 'pro'" }, 400)
   272→
   273→    // Check existing
   274→    const { data: existing } = await sbSelect('subscriptions', 'id', { user_id: `eq.${userId}` })
   275→
   276→    if (existing && existing.length > 0) {
   277→      const updateData: Record<string, unknown> = { plan, started_at: new Date().toISOString() }
   278→      if (plan === 'pro') {
   279→        updateData.expires_at = expiresAt || new Date(Date.now() + 30 * 86400000).toISOString()
   280→      } else {
   281→        updateData.expires_at = null
   282→      }
   283→      const result = await sbUpdate('subscriptions', updateData, { user_id: `eq.${userId}` })
   284→      if (result.error) return json({ error: result.error }, 500)
   285→      return json({ data: result.data })
   286→    } else {
   287→      const insertData: Record<string, unknown> = { user_id: userId, plan }
   288→      if (plan === 'pro') {
   289→        insertData.expires_at = expiresAt || new Date(Date.now() + 30 * 86400000).toISOString()
   290→      }
   291→      const result = await sbInsert('subscriptions', insertData)
   292→      if (result.error) return json({ error: result.error }, 500)
   293→      return json({ data: result.data }, 201)
   294→    }
   295→  } catch {
   296→    return json({ error: 'Request body tidak valid' }, 400)
   297→  }
   298→}

   463→// ─── Subscription ────────────────────────────────────────────────
   464→async function handleSubscription(req: NextRequest): Promise<NextResponse> {
   465→  const url = new URL(req.url)
   466→
   467→  if (req.method === 'GET') {
   468→    const userId = url.searchParams.get('user_id')
   469→    if (!userId) return json({ error: 'user_id required' }, 400)
   470→    const { data, error } = await sbSelect('subscriptions', '*', { user_id: `eq.${userId}` })
   471→    if (error) return json({ error }, 500)
   472→    if (!data || data.length === 0) {
   473→      return json({ data: { plan: 'free', started_at: null, expires_at: null } })
   474→    }
   475→    return json({ data: data[0] })
   476→  }
   477→
   478→  if (req.method === 'POST') {
   479→    try {
   480→      const body = await req.json()
   481→      const userId = body.user_id as string
   482→      const plan = body.plan as string
   483→      if (!userId || !plan) return json({ error: 'user_id dan plan required' }, 400)
   484→
   485→      const { data: existing } = await sbSelect('subscriptions', 'id', { user_id: `eq.${userId}` })
   486→      const expiresAt = plan === 'pro' ? new Date(Date.now() + 30 * 86400000).toISOString() : null
   487→
   488→      if (existing && existing.length > 0) {
   489→        const result = await sbUpdate('subscriptions', { plan, started_at: new Date().toISOString(), expires_at: expiresAt }, { user_id: `eq.${userId}` })
   490→        if (result.error) return json({ error: result.error }, 500)
   491→        return json({ data: result.data })
   492→      } else {
   493→        const result = await sbInsert('subscriptions', { user_id: userId, plan, expires_at: expiresAt })
   494→        if (result.error) return json({ error: result.error }, 500)
   495→        return json({ data: result.data }, 201)
   496→      }
   497→    } catch {
   498→      return json({ error: 'Request body tidak valid' }, 400)
   499→    }
   500→  }
   501→
   502→  return json({ error: 'Method not allowed' }, 405)
   503→}

   506→async function handleSummary(req: NextRequest): Promise<NextResponse> {
   507→  const url = new URL(req.url)
   508→  const userId = url.searchParams.get('user_id')
   509→  if (!userId) return json({ error: 'user_id required' }, 400)
   510→
   511→  const [txnsRes, debtsRes] = await Promise.all([
   512→    sbSelect('transactions', '*', { user_id: `eq.${userId}` }),
   513→    sbSelect('debts', '*', { user_id: `eq.${userId}` }),
   514→  ])
   515→
   516→  const transactions = (txnsRes.data || []) as { type: string; amount: number; occurred_at: string }[]
   517→  const debts = (debtsRes.data || []) as { type: string; status: string; amount: number; paid_amount: number }[]
   518→
   519→  const totalIncome = transactions.filter(t => t.type === 'income').reduce((s, t) => s + t.amount, 0)
   520→  const totalExpenses = transactions.filter(t => t.type === 'expense').reduce((s, t) => s + t.amount, 0)
   521→  const activeBalance = totalIncome - totalExpenses
   522→  const totalDebt = debts.filter(d => d.type === 'debt' && d.status !== 'paid').reduce((s, d) => s + (d.amount - d.paid_amount), 0)
   523→  const totalReceivable = debts.filter(d => d.type === 'receivable' && d.status !== 'paid').reduce((s, d) => s + (d.amount - d.paid_amount), 0)
   524→
   525→  return json({ activeBalance, totalIncome, totalExpenses, totalDebt, totalReceivable })
   526→}

   528→// ─── Categorize ──────────────────────────────────────────────────
   529→const CATEGORY_RULES: { keywords: string[]; category: string; type: 'income' | 'expense' }[] = [
   530→  { keywords: ['food', 'lunch', 'dinner', 'breakfast', 'cafe', 'coffee', 'grocery', 'makan', 'beli', 'restaurant'], category: 'F&B', type: 'expense' },
   531→  { keywords: ['gas', 'fuel', 'uber', 'taxi', 'bus', 'train', 'parking', 'transport', 'ojek', 'bensin'], category: 'Transportation', type: 'expense' },
   532→  { keywords: ['rent', 'mortgage', 'electricity', 'water', 'utility', 'internet', 'listrik', 'sewa'], category: 'Housing', type: 'expense' },
   533→  { keywords: ['movie', 'netflix', 'spotify', 'game', 'entertainment', 'hiburan'], category: 'Entertainment', type: 'expense' },
   534→  { keywords: ['shop', 'shopping', 'clothes', 'belanja', 'purchase'], category: 'Shopping', type: 'expense' },
   535→  { keywords: ['doctor', 'hospital', 'medicine', 'health', 'dokter', 'obat'], category: 'Healthcare', type: 'expense' },
   536→  { keywords: ['salary', 'wage', 'paycheck', 'bonus', 'freelance', 'income', 'gaji', 'upah'], category: 'Income', type: 'income' },
   537→  { keywords: ['transfer', 'deposit', 'withdrawal'], category: 'Transfer', type: 'expense' },
   538→  { keywords: ['gift', 'donation', 'hadiah', 'sedekah'], category: 'Gift', type: 'expense' },
   539→]
   540→
   541→function autoCategorize(desc: string): { category: string; type: 'income' | 'expense' } {
   542→  const lower = desc.toLowerCase()
   543→  for (const rule of CATEGORY_RULES) {
   544→    for (const kw of rule.keywords) {
   545→      if (lower.includes(kw)) return { category: rule.category, type: rule.type }
   546→    }
   547→  }
   548→  return { category: 'Other', type: 'expense' }
   549→}

   562→// ─── Sync ────────────────────────────────────────────────────────
   563→async function handleSync(req: NextRequest): Promise<NextResponse> {
   564→  const url = new URL(req.url)
   565→
   566→  if (req.method === 'GET') {
   567→    const userId = url.searchParams.get('user_id')
   568→    if (!userId) return json({ error: 'user_id required' }, 400)
   569→
   570→    const [wallets, transactions, debts, budgets, customCats, subscription] = await Promise.all([
   571→      sbSelect('wallets', '*', { user_id: `eq.${userId}` }),
   572→      sbSelect('transactions', '*', { user_id: `eq.${userId}` }),
   573→      sbSelect('debts', '*', { user_id: `eq.${userId}` }),
   574→      sbSelect('budgets', '*', { user_id: `eq.${userId}` }),
   575→      sbSelect('custom_categories', '*', { user_id: `eq.${userId}` }),
   576→      sbSelect('subscriptions', '*', { user_id: `eq.${userId}` }),
   577→    ])
   578→
   579→    return json({
   580→      data: {
   581→        wallets: wallets.data || [],
   582→        transactions: transactions.data || [],
   583→        debts: debts.data || [],
   584→        budgets: (budgets.data || []).reduce((acc: Record<string, number>, b: Record<string, unknown>) => { acc[b.category as string] = b.amount as number; return acc }, {}),
   585→        customCats: (customCats.data || []).map((c: Record<string, unknown>) => ({ name: c.name, type: c.type, color: c.color, kw: c.keywords })),
   586→        subscription: (subscription.data || [])[0] || { plan: 'free' },
   587→      },
   588→    })
   589→  }
   590→
   591→  if (req.method === 'POST') {
   592→    try {
   593→      const body = await req.json()
   594→      const userId = body.user_id as string
   595→      const syncData = body.data as Record<string, unknown>
   596→      if (!userId || !syncData) return json({ error: 'user_id dan data required' }, 400)
   597→
   598→      // Check Pro plan
   599→      const { data: subs } = await sbSelect('subscriptions', 'plan', { user_id: `eq.${userId}` })
   600→      const sub = (subs || [])[0] as { plan: string } | undefined
   601→      if (sub?.plan !== 'pro') return json({ error: 'Sinkronisasi cloud cuma untuk paket Pro.' }, 403)
   602→
   603→      const results: Record<string, unknown> = {}
   604→
   605→      // Upsert wallets
   606→      if (Array.isArray(syncData.wallets) && syncData.wallets.length > 0) {
   607→        const wallets = (syncData.wallets as Record<string, unknown>[]).map(w => ({
   608→          id: w.id, user_id: userId, name: w.name, color: w.color || '#C9A962', initial_balance: w.initial || 0,
   609→        }))
   610→        const res = await fetch(`${SUPABASE_URL}/rest/v1/wallets`, {
   611→          method: 'POST',
   612→          headers: { ...adminHeaders(), Prefer: 'resolution=merge-duplicates' },
   613→          body: JSON.stringify(wallets),
   614→        })
   615→        if (res.ok) results.wallets = await res.json()
   616→      }
   617→
   618→      // Upsert transactions
   619→      if (Array.isArray(syncData.transactions) && syncData.transactions.length > 0) {
   620→        const txns = (syncData.transactions as Record<string, unknown>[]).map(t => ({
   621→          id: t.id, user_id: userId, wallet_id: t.wallet_id, type: t.type, category: t.category,
   622→          description: t.description, amount: t.amount, occurred_at: t.occurred_at,
   623→          source: t.source || 'manual', ref_id: t.ref_id || null, recur: t.recur || 'none', recur_parent: t.recur_parent || null,
   624→        }))
   625→        const res = await fetch(`${SUPABASE_URL}/rest/v1/transactions`, {
   626→          method: 'POST',
   627→          headers: { ...adminHeaders(), Prefer: 'resolution=merge-duplicates' },
   628→          body: JSON.stringify(txns),
   629→        })
   630→        if (res.ok) results.transactions = await res.json()
   631→      }
   632→
   633→      // Upsert debts
   634→      if (Array.isArray(syncData.debts) && syncData.debts.length > 0) {
   635→        const debts = (syncData.debts as Record<string, unknown>[]).map(d => ({
   636→          id: d.id, user_id: userId, name: d.name, type: d.type, amount: d.amount,
   637→          paid_amount: d.paid_amount || 0, due_date: d.due_date || null, note: d.note || '',
   638→          status: d.status || 'unpaid', settled_at: d.settled_at || null,
   639→        }))
   640→        const res = await fetch(`${SUPABASE_URL}/rest/v1/debts`, {
   641→          method: 'POST',
   642→          headers: { ...adminHeaders(), Prefer: 'resolution=merge-duplicates' },
   643→          body: JSON.stringify(debts),
   644→        })
   645→        if (res.ok) results.debts = await res.json()
   646→      }
   647→
   648→      return json({ ok: true, synced: Object.keys(results) })
   649→    } catch {
   650→      return json({ error: 'Request body tidak valid' }, 400)
   651→    }
   652→  }
   653→
   654→  return json({ error: 'Method not allowed' }, 405)
   655→}

   658→async function handlePaymentCreate(req: NextRequest): Promise<NextResponse> {
   659→  try {
   660→    const body = await req.json()
   661→    const userId = body.user_id as string
   662→    const email = body.email as string
   663→    const name = (body.name as string) || 'Saya'
   664→    if (!userId) return json({ error: 'user_id required' }, 400)
   665→
   666→    const midtransKey = getEnv('MIDTRANS_SERVER_KEY')
   667→    if (!midtransKey || midtransKey.includes('placeholder')) {
   668→      return json({ error: 'Midtrans belum dikonfigurasi. Set MIDTRANS_SERVER_KEY di .env' }, 503)
   669→    }
   670→
   671→    const isProd = getEnv('MIDTRANS_IS_PRODUCTION') === 'true'
   672→    const baseUrl = isProd
   673→      ? 'https://app.midtrans.com/snap/v1/transactions'
   674→      : 'https://app.sandbox.midtrans.com/snap/v1/transactions'
   675→
   676→    const orderId = `ARUS-PRO-${userId.slice(0, 8)}-${Date.now()}`
   677→    const payload = {
   678→      transaction_details: { order_id: orderId, gross_amount: 29000 },
   679→      item_details: [{ id: 'pro-monthly', price: 29000, quantity: 1, name: 'Arus Pro — Bulanan', category: 'Subscription' }],
   680→      customer_details: { first_name: name, email: email || undefined },
   681→      metadata: { user_id: userId, plan: 'pro' },
   682→    }
   683→
   684→    const authKey = Buffer.from(midtransKey + ':').toString('base64')
   685→    const res = await fetch(baseUrl, {
   686→      method: 'POST',
   687→      headers: { 'Content-Type': 'application/json', Accept: 'application/json', Authorization: `Basic ${authKey}` },
   688→      body: JSON.stringify(payload),
   689→    })
   690→    const data = await res.json()
   691→    if (!res.ok) return json({ error: data.error_messages?.[0] || 'Midtrans error' }, 400)
   692→
   693→    // Save pending order
   694→    await fetch(`${SUPABASE_URL}/rest/v1/subscriptions`, {
   695→      method: 'POST',
   696→      headers: { ...adminHeaders(), Prefer: 'resolution=merge-duplicates' },
   697→      body: JSON.stringify({ user_id: userId, plan: 'free', payment_method: 'midtrans', transaction_id: orderId }),
   698→    })
   699→
   700→    return json({ token: data.token, redirect_url: data.redirect_url, order_id: orderId })
   701→  } catch (e) {
   702→    return json({ error: 'Gagal membuat pembayaran' }, 500)
   703→  }
   704→}

   730→async function handleRequest(req: NextRequest): Promise<NextResponse> {
   731→  // Check if Supabase is configured
   732→  if (!isConfigured()) {
   733→    return json({
   734→      error: 'Supabase belum dikonfigurasi. Tambahkan NEXT_PUBLIC_SUPABASE_URL dan NEXT_PUBLIC_SUPABASE_ANON_KEY di file .env',
   735→      setup: true,
   736→    }, 503)
   737→  }
   738→
   739→  const url = new URL(req.url)
   740→  const pathParts = url.pathname.replace('/api/', '').split('/')
   741→  const path = '/' + pathParts.join('/')
   742→
   743→  try {
   744→    // ─── Auth routes ───
   745→    if (path === '/auth/login') return await handleLogin(req)
   746→    if (path === '/auth/signup') return await handleSignup(req)
   747→    if (path === '/auth/me') return await handleAuthMe(req)
   748→    if (path === '/auth/logout') return await handleLogout(req)
   749→
   750→    // ─── Admin routes (require admin verification) ───
   751→    if (path.startsWith('/admin')) {
   752→      const admin = await verifyAdmin(req)
   753→      if (!admin) return json({ error: 'Akses ditolak. Halaman ini hanya untuk admin.' }, 403)
   754→
   755→      if (path === '/admin/stats') return await handleAdminStats()
   756→      if (path === '/admin/users') return await handleAdminUsers()
   757→      if (path === '/admin/subscriptions') return await handleAdminSubscriptions()
   758→      if (path === '/admin/subscription/update') return await handleAdminSubscriptionUpdate(req)
   759→      if (path === '/admin/transactions') return await handleAdminTransactions(req)
   760→      return json({ error: 'Admin endpoint not found' }, 404)
   761→    }
   762→
   763→    // ─── Data routes ───
   764→    if (path === '/wallets' || path.startsWith('/wallets')) return await handleWallets(req)
   765→    if (path === '/transactions' || path.startsWith('/transactions')) return await handleTransactions(req)
   766→    if (path === '/debts' || path.startsWith('/debts')) return await handleDebts(req)
   767→    if (path === '/subscription') return await handleSubscription(req)
   768→    if (path === '/summary') return await handleSummary(req)
   769→    if (path === '/categorize') return await handleCategorize(req)
   770→    if (path === '/sync') return await handleSync(req)
   771→    if (path === '/payment/create') return await handlePaymentCreate(req)
   772→
   773→    // ─── Root ───
   774→    if (path === '/' || path === '') return json({ message: 'Arus Kas API v3.1', status: 'ok' })
   775→
   776→    return json({ error: 'Not found' }, 404)
   777→  } catch (err: unknown) {
   778→    const msg = err instanceof Error ? err.message : 'Server error'
   779→    return json({ error: msg }, 500)
   780→  }
   781→}
